Example 15099

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@meaty-swing.org"
  subject: "Cubitum tepesco placeat vestigium pecus."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Cultellus brevis basium causa ars accedo color ara perspiciatis clarus. Cognatus vix somniculosus tibi amet distinctio adiuvo. Appositus ater itaque vomer uberrime ascit.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Corrigo amiculum cervus deripio utrum tribuo tolero conor. Veritatis fuga demens suppellex arceo curiositas vilis absque caput. Bellum admoveo delectatio tres adsuesco coruscus aequus cicuta accusantium cinis.

Visit: {{url}}
"""
}
