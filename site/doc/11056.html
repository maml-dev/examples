DNS zone record entries.

{
  zone: "realistic-futon.com"
  records: [
    {
      type: "MX"
      name: "www"
      value: "black-and-white-tuber.info"
      ttl: 86400
    }

    {
      type: "MX"
      name: "www"
      value: "wavy-redesign.org"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "staging"
      value: "149.96.93.118"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "*"
      value: "respectful-tenant.name"
      ttl: 300
    }

    {
      type: "A"
      name: "cdn"
      value: "private-premise.org"
      ttl: 3600
    }
  ]
}
