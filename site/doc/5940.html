Example 5940

CI Pipeline

Continuous integration pipeline definition.

{
  name: "CI Pipeline"
  trigger: {
    branches: [
      "main"
      "develop"
    ]
    paths: [
      "src/**"
      "tests/**"
    ]
  }

  stages: {
    # Code quality
    lint: {
      image: "node:18"
      command: "npx eslint ."
    }
    test: {
      image: "node:18"
      command: "npm test"
      coverage: true
    }
    build: {
      image: "node:22"
      command: "npm run build"
      artifacts: [
        "dist/**"
        "build/**"
      ]
    }
  }
}
