Example 11853

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vim"
  shortcuts: [
    {
      key: "Ctrl+Shift+P"
      action: "select_all_occurrences"
      when: "terminal"
    }

    {
      key: "Ctrl+Shift+P"
      action: "toggle_sidebar"
      when: "editor"
    }

    {
      key: "Alt+Up"
      action: "toggle_terminal"
      when: "terminal"
    }

    {
      key: "Ctrl+S"
      action: "command_palette"
      when: "editor"
    }

    {
      key: "Ctrl+D"
      action: "search_project"
      when: "terminal"
    }
  ]
}
