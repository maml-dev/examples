Example 15242

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vscode"
  shortcuts: [
    {
      key: "Ctrl+S"
      action: "debug_run"
      when: "editor"
    }

    {
      key: "Cmd+Shift+L"
      action: "find_references"
      when: "debug"
    }

    {
      key: "Ctrl+D"
      action: "search_project"
      when: "terminal"
    }
  ]
}
