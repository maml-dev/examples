Example 145

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "emacs"
  shortcuts: [
    {
      key: "Ctrl+B"
      action: "select_all_occurrences"
      when: "terminal"
    }

    {
      key: "Ctrl+`"
      action: "command_palette"
      when: "editor"
    }

    {
      key: "Ctrl+S"
      action: "toggle_comment"
      when: "terminal"
    }
  ]
}
