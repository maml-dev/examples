Example 1837

Queue Config

Message queue and worker configuration.

{
  broker: "sqs"
  connection: {
    host: "mq.internal"
    port: 9092
    vhost: "/production"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 5
      retry: 1
      timeout: 22445 # ms
    }

    priority: {
      concurrency: 16
      retry: 9
      timeout: 8261 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 5
    queue: "dead-letter"
  }
}
