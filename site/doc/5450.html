Example 5450

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vscode"
  shortcuts: [
    {
      key: "Ctrl+Shift+P"
      action: "toggle_terminal"
      when: "terminal"
    }

    {
      key: "Ctrl+`"
      action: "select_all_occurrences"
      when: "global"
    }

    {
      key: "F5"
      action: "save"
      when: "terminal"
    }
  ]
}
