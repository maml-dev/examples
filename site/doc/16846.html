Example 16846

API Routes

REST API route definitions.

{
  base_url: "/api/v3"
  routes: [
    {
      method: "DELETE"
      path: "/products/:id"
      handler: "createOrders"
      auth: true
    }

    {
      method: "DELETE"
      path: "/orders"
      handler: "createPosts"
      auth: true
    }

    {
      method: "GET"
      path: "/orders"
      handler: "updatePosts"
      auth: false
    }

    {
      method: "PATCH"
      path: "/users/:id"
      handler: "deleteUsers"
      auth: true
    }

    {
      method: "PATCH"
      path: "/users/:id"
      handler: "deleteProducts"
      auth: true
    }
  ]
}
