Example 14056

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "sublime"
  shortcuts: [
    {
      key: "Ctrl+S"
      action: "toggle_comment"
      when: "debug"
    }

    {
      key: "Ctrl+Shift+F"
      action: "format"
      when: "debug"
    }

    {
      key: "Ctrl+S"
      action: "toggle_comment"
      when: "editor"
    }

    {
      key: "Cmd+K"
      action: "toggle_terminal"
      when: "terminal"
    }
  ]
}
