Example 16587

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@magnificent-devastation.com"
  subject: "Curis sollers volup collum tribuo."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Amaritudo crastinus sustineo. Aurum aequitas nostrum tepidus cumque aperiam solus vestigium. Quibusdam cariosus adipisci culpa conatus decet calco.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Sollicito voco vulticulus approbo temporibus deripio subnecto tollo textus. Veritas consequuntur arcesso. Cerno vigilo absens ascit vigilo aurum quisquam.

Visit: {{url}}
"""
}
