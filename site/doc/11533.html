Example 11533

K8s Deployment

Kubernetes deployment specification.

{
  apiVersion: "apps/v1"
  kind: "Deployment"
  metadata: {
    name: "window-api"
    namespace: "default"
    labels: {
      app: "consistency"
      tier: "frontend"
    }
  }

  # Deployment spec
  spec: {
    replicas: 4
    container: {
      image: "meadow:4.15.9"
      port: 9090
      resources: {
        cpu_limit: "1000m"
        memory_limit: "512Mi"
      }
    }

    health_check: {
      path: "/ready"
      interval: 10
      timeout: 5
    }
  }
}
