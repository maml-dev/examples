Example 6421

Email Routing

Email routing and filtering rules.

{
  domain: "showy-amendment.org"
  inbound: {
    mx_records: [
      "mx1.showy-amendment.org"
      "mx2.showy-amendment.org"
    ]
    spam_filter: {
      enabled: true
      threshold: 7
      action: "reject"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 2048
    }
  }

  outbound: {
    relay: "smtp.showy-amendment.org"
    port: 25
    tls: true
    rate_limit: {
      per_minute: 205
      per_hour: 5157
    }
  }
}
