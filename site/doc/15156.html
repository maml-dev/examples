Example 15156

API Routes

REST API route definitions.

{
  base_url: "/api/v2"
  routes: [
    {
      method: "PATCH"
      path: "/products"
      handler: "getPosts"
      auth: false
    }

    {
      method: "DELETE"
      path: "/posts/:id"
      handler: "createProducts"
      auth: false
    }

    {
      method: "POST"
      path: "/posts"
      handler: "deleteProducts"
      auth: true
    }

    {
      method: "GET"
      path: "/orders/:id"
      handler: "listProducts"
      auth: true
    }

    {
      method: "GET"
      path: "/orders/:id"
      handler: "getPosts"
      auth: true
    }

    {
      method: "GET"
      path: "/orders/:id"
      handler: "listUsers"
      auth: true
    }
  ]
}
