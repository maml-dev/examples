Example 16295

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@burly-unique.net"
  subject: "Vereor voco consequatur."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Accusamus curso textilis subiungo alii ut trado cohaero. Candidus ullam patior bardus cauda velociter. Theca theatrum bibo dignissimos teneo denuncio.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Ago defungo coadunatio vulticulus quod. Approbo velociter absorbeo auctus. Causa neque laborum aegrus cado depono vereor ascisco credo.

Visit: {{url}}
"""
}
