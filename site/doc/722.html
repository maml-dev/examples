Terraform Resource

Infrastructure as code resource definition.

{
  provider: "aws"
  region: "us-east-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.149.0.0/16"
      enable_dns: true
      tags: {
        Name: "thyme-vpc"
        Environment: "production"
      }
    }

    # Compute
    instance: {
      ami: "ami-47317ddc"
      type: "t3.medium"
      key_name: "orchid-key"
      monitoring: true
      root_volume: {
        size: 50
        type: "gp2"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-web"
      ingress_port: 80
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
