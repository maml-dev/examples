Example 3630

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.bony-follower.biz"
    port: 443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "1m"
    max_requests: 3638
    by: "user"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://whispered-fledgling.net"
      "https://app.known-provision.name"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 7200
  }

  cache: {
    enabled: true
    ttl: 1842
    store: "redis"
  }
}
