Terraform Resource

Infrastructure as code resource definition.

{
  provider: "google"
  region: "asia-southeast1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.159.0.0/16"
      enable_dns: true
      tags: {
        Name: "lotion-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-f28ea933"
      type: "t3.medium"
      key_name: "median-key"
      monitoring: true
      root_volume: {
        size: 100
        type: "io1"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-ssh"
      ingress_port: 8080
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
