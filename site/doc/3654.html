Example 3654

Cron Jobs

Scheduled task definitions.

{
  timezone: "America/New_York"
  jobs: [
    {
      name: "refresh-cache"
      schedule: "*/15 * * * *"
      command: "node scripts/cleanup.js"
      enabled: true
    }

    {
      name: "backup-db"
      schedule: "0 2 * * 0"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "cleanup-temp"
      schedule: "*/15 * * * *"
      command: "python manage.py report"
      enabled: true
    }

    {
      name: "send-reports"
      schedule: "*/15 * * * *"
      command: "node scripts/cleanup.js"
      enabled: false
    }

    {
      name: "backup-db"
      schedule: "0 * * * *"
      command: "node scripts/cleanup.js"
      enabled: true
    }

    {
      name: "sync-data"
      schedule: "0 2 * * 0"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }
  ]
}
