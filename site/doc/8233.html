Example 8233

Backup Strategy

Backup schedule and retention policy.

{
  source: {
    type: "mongodb"
    host: "localhost"
    databases: [
      "halt"
      "pleasure"
    ]
  }

  # Backup schedule
  schedule: {
    full: "0 2 * * 0" # weekly Sunday 2am
    incremental: "0 2 * * 1-6" # daily except Sunday
    snapshot: "0 */4 * * *" # every 4 hours
  }

  retention: {
    daily: 8
    weekly: 5
    monthly: 19
  }

  destination: {
    type: "local"
    path: "/mnt/backups/anticodon"
    encryption: true
    compression: "zstd"
  }
}
