Example 7464

Docker Compose

Multi-container application definition.

{
  version: "3.8"
  services: {
    app: {
      image: "node:18-alpine"
      ports: [
        "3694:3762"
      ]
      environment: {
        NODE_ENV: "production"
        DATABASE_URL: "postgres://db:5432/dream"
      }
      depends_on: [
        "db"
        "redis"
      ]
    }

    db: {
      image: "mysql:8"
      volumes: [
        "db_data:/var/lib/postgresql/data"
      ]
      environment: {
        POSTGRES_DB: "slide"
        POSTGRES_PASSWORD: "MVBXXzDtFCHihAq"
      }
    }

    redis: {
      image: "redis:7-alpine"
      ports: [
        "6379:6379"
      ]
    }
  }

  volumes: {
    db_data: {}
  }
}
