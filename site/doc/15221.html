Example 15221

K8s Deployment

Kubernetes deployment specification.

{
  apiVersion: "apps/v1"
  kind: "Deployment"
  metadata: {
    name: "tectonics-worker"
    namespace: "production"
    labels: {
      app: "scale"
      tier: "backend"
    }
  }

  # Deployment spec
  spec: {
    replicas: 5
    container: {
      image: "carboxyl:5.17.12"
      port: 8000
      resources: {
        cpu_limit: "500m"
        memory_limit: "1Gi"
      }
    }

    health_check: {
      path: "/health"
      interval: 15
      timeout: 5
    }
  }
}
