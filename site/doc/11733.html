CI Pipeline

Continuous integration pipeline definition.

{
  name: "Build & Test"
  trigger: {
    branches: [
      "main"
      "develop"
    ]
    paths: [
      "src/**"
      "tests/**"
    ]
  }

  stages: {
    # Code quality
    lint: {
      image: "node:22"
      command: "npx biome check"
    }
    test: {
      image: "node:18"
      command: "npx vitest"
      coverage: true
    }
    build: {
      image: "node:20"
      command: "npm run build"
      artifacts: [
        "dist/**"
        "build/**"
      ]
    }
  }
}
