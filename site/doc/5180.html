Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Oberbrunner - Halvorson Dashboard"
  refresh_interval: 60 # seconds
  widgets: [
    {
      type: "chart"
      title: "Signups"
      source: "/api/metrics/revenue"
      chart_type: "line"
      period: "30d"
    }

    {
      type: "stat"
      title: "Uptime"
      source: "/api/stats/users"
      format: "number"
    }

    {
      type: "table"
      title: "Top Pages"
      source: "/api/recent/orders"
      limit: 23
    }
  ]

  theme: "light"
}
