Example 6736

Queue Config

Message queue and worker configuration.

{
  broker: "sqs"
  connection: {
    host: "localhost"
    port: 5672
    vhost: "/production"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 5
      retry: 1
      timeout: 25336 # ms
    }

    priority: {
      concurrency: 20
      retry: 9
      timeout: 3412 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 8
    queue: "dead-letter"
  }
}
