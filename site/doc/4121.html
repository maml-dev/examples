Example 4121

K8s Deployment

Kubernetes deployment specification.

{
  apiVersion: "apps/v1"
  kind: "Deployment"
  metadata: {
    name: "mortise-api"
    namespace: "staging"
    labels: {
      app: "guidance"
      tier: "data"
    }
  }

  # Deployment spec
  spec: {
    replicas: 2
    container: {
      image: "freezing:2.10.16"
      port: 8000
      resources: {
        cpu_limit: "500m"
        memory_limit: "512Mi"
      }
    }

    health_check: {
      path: "/health"
      interval: 15
      timeout: 3
    }
  }
}
