Example 1739

Email Routing

Email routing and filtering rules.

{
  domain: "compassionate-concentration.name"
  inbound: {
    mx_records: [
      "mx1.compassionate-concentration.name"
      "mx2.compassionate-concentration.name"
    ]
    spam_filter: {
      enabled: true
      threshold: 7
      action: "reject"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.compassionate-concentration.name"
    port: 465
    tls: true
    rate_limit: {
      per_minute: 164
      per_hour: 4572
    }
  }
}
