Example 17540

Backup Strategy

Backup schedule and retention policy.

{
  source: {
    type: "filesystem"
    host: "localhost"
    databases: [
      "decongestant"
      "newsletter"
    ]
  }

  # Backup schedule
  schedule: {
    full: "0 2 * * 0" # weekly Sunday 2am
    incremental: "0 2 * * 1-6" # daily except Sunday
    snapshot: "0 */4 * * *" # every 4 hours
  }

  retention: {
    daily: 7
    weekly: 9
    monthly: 11
  }

  destination: {
    type: "gcs"
    path: "s3://backups-pants"
    encryption: true
    compression: "gzip"
  }
}
