Example 17451

API Routes

REST API route definitions.

{
  base_url: "/api/v2"
  routes: [
    {
      method: "POST"
      path: "/orders/:id"
      handler: "listOrders"
      auth: false
    }

    {
      method: "POST"
      path: "/comments"
      handler: "deleteOrders"
      auth: true
    }

    {
      method: "DELETE"
      path: "/products"
      handler: "getProducts"
      auth: true
    }
  ]
}
