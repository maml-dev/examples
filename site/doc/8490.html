Example 8490

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@oblong-tomography.name"
  subject: "Crur acies defendo sponte ars."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Pectus adulescens delibero aurum demo. Correptius verbum itaque maxime aspernatur testimonium quae defero. Deludo cariosus patior acervus adamo theologus ater cruentus tunc.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Pariatur ventosus suadeo abstergo vel ullus addo adnuo succedo. Turpis velit et perferendis venustas considero. Molestiae vergo depono asporto arcesso comminor veritatis vulgus.

Visit: {{url}}
"""
}
