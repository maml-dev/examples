Example 14645

Cron Jobs

Scheduled task definitions.

{
  timezone: "UTC"
  jobs: [
    {
      name: "refresh-cache"
      schedule: "0 2 * * 0"
      command: "./bin/sync --all"
      enabled: true
    }

    {
      name: "sync-data"
      schedule: "0 * * * *"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "backup-db"
      schedule: "0 2 * * 0"
      command: "python manage.py report"
      enabled: true
    }
  ]
}
