{
  zone: "scented-castanet.name"
  records: [
    {
      type: "TXT"
      name: "*"
      value: "apt-kick.name"
      ttl: 300

      type: "AAAA"
      name: "cdn"
      value: "v=spf1 include:_spf.finished-piglet.com ~all"
      ttl: 300
    }

      name: "staging"
