Example 9932

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.bleak-rationale.com"
    port: 8443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "15m"
    max_requests: 2595
    by: "api_key"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://only-violin.net"
      "https://app.frugal-pigsty.net"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 86400
  }

  cache: {
    enabled: false
    ttl: 570
    store: "redis"
  }
}
