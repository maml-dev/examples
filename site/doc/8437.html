Example 8437

Email Routing

Email routing and filtering rules.

{
  domain: "blond-role.biz"
  inbound: {
    mx_records: [
      "mx1.blond-role.biz"
      "mx2.blond-role.biz"
    ]
    spam_filter: {
      enabled: true
      threshold: 8.5
      action: "tag"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.blond-role.biz"
    port: 465
    tls: true
    rate_limit: {
      per_minute: 312
      per_hour: 7088
    }
  }
}
