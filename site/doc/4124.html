Terraform Resource

Infrastructure as code resource definition.

{
  provider: "digitalocean"
  region: "asia-southeast1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.143.0.0/16"
      enable_dns: true
      tags: {
        Name: "boyfriend-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-3bb538c7"
      type: "t3.medium"
      key_name: "cornet-key"
      monitoring: true
      root_volume: {
        size: 50
        type: "gp2"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-ssh"
      ingress_port: 22
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
