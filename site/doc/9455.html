Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "delete"
        "view_analytics"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "manage_billing"
        "write"
      ]
      max_seats: 27
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "manage_users"
        "manage_billing"
        "read"
        "write"
      ]
      max_seats: 22
    }

    {
      name: "admin"
      permissions: [
        "view_analytics"
        "manage_users"
        "delete"
      ]
      max_seats: 6
    }

      ]
      max_seats: null
