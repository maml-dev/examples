Test Suite

Test runner configuration with coverage settings.

{
  runner: "mocha"
  root: "./src/__tests__"
  include: [
    "**/*.test.ts"
    "**/*.spec.ts"
  ]
  exclude: [
    "node_modules"
    "dist"
    "**/*.e2e.ts"
  ]

  # Coverage
  coverage: {
    enabled: true
    provider: "v8"
    thresholds: {
      lines: 87
      branches: 62
      functions: 85
    }
    reporter: [
      "text"
      "lcov"
      "html"
    ]
  }
  globals: true
  timeout: 20815 # ms
}
