API Routes

REST API route definitions.

{
  base_url: "/api/v1"
  routes: [
    {
      method: "PATCH"
      path: "/comments"
      handler: "createProducts"
      auth: false
    }

    {
      method: "GET"
      path: "/products"
      handler: "createUsers"
      auth: false
    }

    {
      method: "PUT"
      path: "/orders"
      handler: "createPosts"
      auth: true
    }

    {
      method: "DELETE"
      path: "/products"
      handler: "createPosts"
      auth: false
    }
  ]
}
