Example 6826

Email Routing

Email routing and filtering rules.

{
  domain: "big-devastation.name"
  inbound: {
    mx_records: [
      "mx1.big-devastation.name"
      "mx2.big-devastation.name"
    ]
    spam_filter: {
      enabled: true
      threshold: 8.5
      action: "reject"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.big-devastation.name"
    port: 25
    tls: true
    rate_limit: {
      per_minute: 466
      per_hour: 4298
    }
  }
}
