Example 10869

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Hauck LLC Dashboard"
  refresh_interval: 60 # seconds
  widgets: [
    {
      type: "chart"
      title: "Active Users"
      source: "/api/metrics/traffic"
      chart_type: "bar"
      period: "30d"
    }

    {
      type: "stat"
      title: "Error Rate"
      source: "/api/stats/revenue"
      format: "percent"
    }

    {
      type: "table"
      title: "Latest Signups"
      source: "/api/recent/signups"
      limit: 23
    }
  ]

  theme: "light"
}
