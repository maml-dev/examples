Example 3828

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.pleasant-replacement.net"
    port: 8443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "5m"
    max_requests: 4536
    by: "api_key"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://snoopy-baseboard.org"
      "https://app.qualified-omelet.com"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 86400
  }

  cache: {
    enabled: true
    ttl: 1219
    store: "memory"
  }
}
