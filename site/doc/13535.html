Example 13535

Backup Strategy

Backup schedule and retention policy.

{
  source: {
    type: "mongodb"
    host: "primary.db.internal"
    databases: [
      "derby"
      "eyeliner"
    ]
  }

  # Backup schedule
  schedule: {
    full: "0 2 * * 0" # weekly Sunday 2am
    incremental: "0 2 * * 1-6" # daily except Sunday
    snapshot: "0 */4 * * *" # every 4 hours
  }

  retention: {
    daily: 8
    weekly: 8
    monthly: 24
  }

  destination: {
    type: "sftp"
    path: "s3://backups-forage"
    encryption: true
    compression: "gzip"
  }
}
