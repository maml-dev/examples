Example 3335

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.distant-director.com"
    port: 443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "1m"
    max_requests: 3467
    by: "api_key"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://fond-possession.com"
      "https://app.negative-necklace.com"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 86400
  }

  cache: {
    enabled: false
    ttl: 537
    store: "redis"
  }
}
