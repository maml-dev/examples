Example 9125

K8s Deployment

Kubernetes deployment specification.

{
  apiVersion: "apps/v1"
  kind: "Deployment"
  metadata: {
    name: "papa-web"
    namespace: "staging"
    labels: {
      app: "dusk"
      tier: "data"
    }
  }

  # Deployment spec
  spec: {
    replicas: 1
    container: {
      image: "assist:7.1.2"
      port: 3000
      resources: {
        cpu_limit: "1000m"
        memory_limit: "1Gi"
      }
    }

    health_check: {
      path: "/healthz"
      interval: 30
      timeout: 10
    }
  }
}
