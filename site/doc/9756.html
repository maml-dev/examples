Example 9756

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vim"
  shortcuts: [
    {
      key: "Cmd+Shift+L"
      action: "toggle_sidebar"
      when: "global"
    }

    {
      key: "Ctrl+Shift+F"
      action: "command_palette"
      when: "global"
    }

    {
      key: "Ctrl+D"
      action: "save"
      when: "global"
    }

    {
      key: "Cmd+K"
      action: "save"
      when: "debug"
    }

    {
      key: "Ctrl+Shift+F"
      action: "toggle_comment"
      when: "editor"
    }

    {
      key: "Ctrl+/"
      action: "save"
      when: "editor"
    }
  ]
}
