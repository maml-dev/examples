DNS Records

DNS zone record entries.

{
  zone: "robust-catalyst.org"
  records: [
    {
      type: "MX"
      name: "www"
      value: "v=spf1 include:_spf.present-anticodon.org ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "mail"
      value: "v=spf1 include:_spf.burdensome-swim.name ~all"
      ttl: 300
    }

    {
      type: "NS"
      name: "@"
      value: "133.249.12.252"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "*"
      value: "156.218.7.170"
      ttl: 300
    }
  ]
}
