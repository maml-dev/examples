Example 7884

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@blue-suspension.net"
  subject: "Acsi laboriosam canonicus venustas atrox."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Confugo admiratio clamo comprehendo harum viriliter stipes debitis una suppellex. Amet aqua cunctatio. Alter patria dens virga amiculum adnuo delectatio.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Victus sumptus voluptatum. Theologus saepe corrigo delectatio votum theatrum basium cervus vomer admitto. Deputo ustilo admiratio.

Visit: {{url}}
"""
}
