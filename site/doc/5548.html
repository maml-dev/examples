Example 5548

Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "viewer"
      permissions: [
        "read"
        "delete"
        "manage_billing"
        "view_analytics"
      ]
      max_seats: 44
    }

    {
      name: "viewer"
      permissions: [
        "manage_users"
        "write"
        "read"
        "export_data"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "manage_billing"
      ]
      max_seats: null
    }
  ]
}
