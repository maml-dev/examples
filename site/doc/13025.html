Example 13025

Email Routing

Email routing and filtering rules.

{
  domain: "poor-flint.biz"
  inbound: {
    mx_records: [
      "mx1.poor-flint.biz"
      "mx2.poor-flint.biz"
    ]
    spam_filter: {
      enabled: true
      threshold: 8.5
      action: "tag"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.poor-flint.biz"
    port: 465
    tls: true
    rate_limit: {
      per_minute: 223
      per_hour: 2125
    }
  }
}
