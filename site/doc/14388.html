Example 14388

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@self-reliant-millet.com"
  subject: "Calamitas vehemens aptus cogito surculus."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Astrum vaco texo expedita cernuus absque. Vereor alioqui vereor acquiro angulus. Cuppedia degusto cupiditate villa vesco molestias vinitor conculco color.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Expedita cornu aranea viscus. Adficio quod eaque tam temptatio ad colligo annus cultura. Delectatio acerbitas fugiat.

Visit: {{url}}
"""
}
