Example 11929

Backup Strategy

Backup schedule and retention policy.

{
  source: {
    type: "postgresql"
    host: "primary.db.internal"
    databases: [
      "drug"
      "depot"
    ]
  }

  # Backup schedule
  schedule: {
    full: "0 2 * * 0" # weekly Sunday 2am
    incremental: "0 2 * * 1-6" # daily except Sunday
    snapshot: "0 */4 * * *" # every 4 hours
  }

  retention: {
    daily: 14
    weekly: 10
    monthly: 17
  }

  destination: {
    type: "s3"
    path: "s3://backups-deployment"
    encryption: true
    compression: "zstd"
  }
}
