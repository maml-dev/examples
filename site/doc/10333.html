Example 10333

Email Routing

Email routing and filtering rules.

{
  domain: "avaricious-pilot.name"
  inbound: {
    mx_records: [
      "mx1.avaricious-pilot.name"
      "mx2.avaricious-pilot.name"
    ]
    spam_filter: {
      enabled: true
      threshold: 8.5
      action: "tag"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.avaricious-pilot.name"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 416
      per_hour: 2212
    }
  }
}
