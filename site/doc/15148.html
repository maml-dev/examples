Example 15148

Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "billing"
      permissions: [
        "delete"
        "read"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 6
    }

    {
      name: "admin"
      permissions: [
        "delete"
        "write"
        "read"
      ]
      max_seats: 41
    }

    {
      name: "billing"
      permissions: [
        "manage_billing"
        "read"
        "view_analytics"
        "export_data"
        "delete"
      ]
      max_seats: 38
    }
  ]
}
