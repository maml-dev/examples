Example 1730

Email Routing

Email routing and filtering rules.

{
  domain: "bowed-climb.name"
  inbound: {
    mx_records: [
      "mx1.bowed-climb.name"
      "mx2.bowed-climb.name"
    ]
    spam_filter: {
      enabled: true
      threshold: 5
      action: "reject"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.bowed-climb.name"
    port: 25
    tls: true
    rate_limit: {
      per_minute: 347
      per_hour: 3859
    }
  }
}
