Example 17546

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vscode"
  shortcuts: [
    {
      key: "Ctrl+`"
      action: "select_all_occurrences"
      when: "terminal"
    }

    {
      key: "Ctrl+/"
      action: "format"
      when: "debug"
    }

    {
      key: "Ctrl+B"
      action: "select_all_occurrences"
      when: "global"
    }
  ]
}
