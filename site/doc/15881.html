Example 15881

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@some-mythology.net"
  subject: "Ulterius iusto umbra arcesso tardus."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Aetas adnuo vinum animadverto maiores qui agnitio pauper considero. Thermae beneficium cupiditate vilis complectus tenax. Tantillus cometes concido crapula viduo excepturi sum animadverto tristis.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Blandior conservo quidem tollo stips cito commodi pax sulum recusandae. Vomer aperiam arbitro ullus cura velit vorago casso. Cresco nostrum vorago custodia.

Visit: {{url}}
"""
}
