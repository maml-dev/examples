Example 9551

API Routes

REST API route definitions.

{
  base_url: "/api/v3"
  routes: [
    {
      method: "PUT"
      path: "/orders"
      handler: "updatePosts"
      auth: false
    }

    {
      method: "GET"
      path: "/products/:id"
      handler: "getOrders"
      auth: false
    }

    {
      method: "POST"
      path: "/products"
      handler: "deleteProducts"
      auth: false
    }

    {
      method: "GET"
      path: "/products"
      handler: "getOrders"
      auth: true
    }
  ]
}
