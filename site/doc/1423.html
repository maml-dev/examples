Example 1423

Email Routing

Email routing and filtering rules.

{
  domain: "orderly-turret.info"
  inbound: {
    mx_records: [
      "mx1.orderly-turret.info"
      "mx2.orderly-turret.info"
    ]
    spam_filter: {
      enabled: true
      threshold: 8.5
      action: "reject"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.orderly-turret.info"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 154
      per_hour: 4956
    }
  }
}
