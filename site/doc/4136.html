Example 4136

Backup Strategy

Backup schedule and retention policy.

{
  source: {
    type: "mysql"
    host: "localhost"
    databases: [
      "grandson"
      "convection"
    ]
  }

  # Backup schedule
  schedule: {
    full: "0 2 * * 0" # weekly Sunday 2am
    incremental: "0 2 * * 1-6" # daily except Sunday
    snapshot: "0 */4 * * *" # every 4 hours
  }

  retention: {
    daily: 14
    weekly: 7
    monthly: 12
  }

  destination: {
    type: "sftp"
    path: "s3://backups-swim"
    encryption: true
    compression: "gzip"
  }
}
