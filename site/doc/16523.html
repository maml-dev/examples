Example 16523

Queue Config

Message queue and worker configuration.

{
  broker: "kafka"
  connection: {
    host: "localhost"
    port: 9092
    vhost: "/staging"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 9
      retry: 3
      timeout: 26319 # ms
    }

    priority: {
      concurrency: 12
      retry: 5
      timeout: 2761 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 10
    queue: "dead-letter"
  }
}
