Docker Compose

Multi-container application definition.

{
  version: "3.8"
  services: {
    app: {
      image: "node:22"
      ports: [
        "3381:3186"
      ]
      environment: {
        NODE_ENV: "production"
        DATABASE_URL: "postgres://db:5432/seagull"
      }
      depends_on: [
        "db"
        "redis"
      ]
    }

    db: {
      image: "mariadb:11"
      volumes: [
        "db_data:/var/lib/postgresql/data"
      ]
      environment: {
        POSTGRES_DB: "annual"
        POSTGRES_PASSWORD: "0agZSf8nVMN079D"
      }
    }

    redis: {
      image: "redis:7-alpine"
      ports: [
        "6379:6379"
      ]
    }
  }

  volumes: {
    db_data: {}
  }
}
