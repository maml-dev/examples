Example 3942

API Routes

REST API route definitions.

{
  base_url: "/api/v3"
  routes: [
    {
      method: "GET"
      path: "/posts/:id"
      handler: "updateUsers"
      auth: false
    }

    {
      method: "PUT"
      path: "/users/:id"
      handler: "createProducts"
      auth: false
    }

    {
      method: "DELETE"
      path: "/posts"
      handler: "updateProducts"
      auth: true
    }
  ]
}
