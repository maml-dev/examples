Email Routing

Email routing and filtering rules.

{
  domain: "enchanting-mousse.name"
  inbound: {
    mx_records: [
      "mx1.enchanting-mousse.name"
      "mx2.enchanting-mousse.name"
    ]
    spam_filter: {
      enabled: true
      threshold: 7
      action: "reject"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 2048
    }
  }

  outbound: {
    relay: "smtp.enchanting-mousse.name"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 393
      per_hour: 4506
    }
  }
}
