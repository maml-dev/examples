Example 9952

Cron Jobs

Scheduled task definitions.

{
  timezone: "America/New_York"
  jobs: [
    {
      name: "send-reports"
      schedule: "30 6 * * 1-5"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "backup-db"
      schedule: "0 0 * * *"
      command: "python manage.py report"
      enabled: true
    }

    {
      name: "rotate-logs"
      schedule: "30 6 * * 1-5"
      command: "python manage.py report"
      enabled: true
    }

    {
      name: "sync-data"
      schedule: "*/15 * * * *"
      command: "node scripts/cleanup.js"
      enabled: true
    }

    {
      name: "rotate-logs"
      schedule: "*/15 * * * *"
      command: "./bin/sync --all"
      enabled: false
    }
  ]
}
