Example 3238

Monitoring Config

Application monitoring and alerting setup.

{
  service: "experiences"
  # Prometheus metrics
  metrics: {
    endpoint: "/metrics"
    interval: 60 # seconds
    labels: {
      env: "staging"
      region: "us-east"
      team: "home"
    }
  }

  # Alert thresholds
  alerts: {
    cpu_threshold: 94
    memory_threshold: 91
    latency_p99_ms: 1098
    error_rate: 0.05
    notification: {
      channel: "#incidents"
      severity: "warning"
    }
  }
}
