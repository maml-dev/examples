Example 2255

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vim"
  shortcuts: [
    {
      key: "Alt+Up"
      action: "select_all_occurrences"
      when: "terminal"
    }

    {
      key: "Ctrl+S"
      action: "toggle_terminal"
      when: "debug"
    }

    {
      key: "Alt+Up"
      action: "toggle_comment"
      when: "global"
    }
  ]
}
