Example 5922

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.tiny-mentor.biz"
    port: 443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "1m"
    max_requests: 4245
    by: "api_key"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://squiggly-account.com"
      "https://app.unkempt-freezing.biz"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 3600
  }

  cache: {
    enabled: false
    ttl: 1162
    store: "redis"
  }
}
