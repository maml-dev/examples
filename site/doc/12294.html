Example 12294

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@silent-relative.net"
  subject: "Deleo atrocitas reiciendis conturbo crudelis cum."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Repudiandae traho claustrum deorsum accendo abduco admitto cohors utilis. Nesciunt tego et suffoco. Adsum vinum deporto delectus vorago carpo absque quisquam vae sophismata.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Succurro itaque suppellex acerbitas crudelis civitas repellendus conicio laudantium. Soluta adflicto voluptas aufero vapulus spoliatio. Cotidie virgo ver cultura ceno aestivus ciminatio aduro.

Visit: {{url}}
"""
}
