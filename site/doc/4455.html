Example 4455

Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "admin"
      permissions: [
        "write"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "write"
        "view_analytics"
        "read"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "manage_users"
        "read"
        "delete"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "write"
        "manage_users"
        "view_analytics"
        "read"
      ]
      max_seats: null
    }
  ]
}
