Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Fisher - Maggio Dashboard"
  refresh_interval: 300 # seconds
  widgets: [
    {
      type: "chart"
      title: "Traffic"
      source: "/api/metrics/revenue"
      chart_type: "line"
      period: "90d"
    }

    {
      type: "stat"
      title: "MRR"
      source: "/api/stats/revenue"
      format: "currency"
    }

    {
      type: "table"
      title: "Recent Orders"
      source: "/api/recent/orders"
      limit: 6
    }
  ]

  theme: "light"
}
