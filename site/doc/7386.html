Example 7386

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@ambitious-underpants.info"
  subject: "Ullam excepturi desidero tergum viscus impedit."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Ademptio tero creptio aptus tollo delibero supra vereor. Sonitus timor villa. Capto admiratio demo volutabrum fuga cultellus.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Umbra suscipit solutio consequuntur socius depopulo certe vapulus umerus. Astrum conculco aurum tolero cohaero. Nulla annus creta.

Visit: {{url}}
"""
}
