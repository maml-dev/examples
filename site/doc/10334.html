Example 10334

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.spotless-governance.org"
    port: 443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "15m"
    max_requests: 6624
    by: "user"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://guilty-role.org"
      "https://app.chubby-membership.info"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 3600
  }

  cache: {
    enabled: false
    ttl: 1573
    store: "redis"
  }
}
