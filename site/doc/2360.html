Example 2360

API Routes

REST API route definitions.

{
  base_url: "/api/v3"
  routes: [
    {
      method: "GET"
      path: "/products"
      handler: "updateOrders"
      auth: true
    }

    {
      method: "DELETE"
      path: "/comments/:id"
      handler: "getOrders"
      auth: false
    }

    {
      method: "POST"
      path: "/orders/:id"
      handler: "getProducts"
      auth: true
    }

    {
      method: "PUT"
      path: "/products/:id"
      handler: "createPosts"
      auth: false
    }
  ]
}
