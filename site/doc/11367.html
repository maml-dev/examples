Example 11367

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Lehner, Leannon and Kemmer Dashboard"
  refresh_interval: 30 # seconds
  widgets: [
    {
      type: "chart"
      title: "Active Users"
      source: "/api/metrics/traffic"
      chart_type: "bar"
      period: "30d"
    }

    {
      type: "stat"
      title: "Total Users"
      source: "/api/stats/users"
      format: "percent"
    }

    {
      type: "table"
      title: "Top Pages"
      source: "/api/recent/signups"
      limit: 19
    }
  ]

  theme: "dark"
}
