Test Suite

Test runner configuration with coverage settings.

{
  runner: "mocha"
  root: "./src/__tests__"
  include: [
    "**/*.test.ts"
    "**/*.spec.ts"
  ]
  exclude: [
    "node_modules"
    "dist"
    "**/*.e2e.ts"
  ]

  # Coverage
  coverage: {
    enabled: true
    provider: "istanbul"
    thresholds: {
      lines: 77
      branches: 70
      functions: 81
    }
    reporter: [
      "text"
      "lcov"
      "html"
    ]
  }
  globals: true
  timeout: 20417 # ms
}
