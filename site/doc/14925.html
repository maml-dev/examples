Example 14925

Queue Config

Message queue and worker configuration.

{
  broker: "redis"
  connection: {
    host: "localhost"
    port: 5672
    vhost: "/production"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 10
      retry: 4
      timeout: 57431 # ms
    }

    priority: {
      concurrency: 8
      retry: 5
      timeout: 8442 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 7
    queue: "dead-letter"
  }
}
