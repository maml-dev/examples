Example 2423

Email Routing

Email routing and filtering rules.

{
  domain: "athletic-annual.com"
  inbound: {
    mx_records: [
      "mx1.athletic-annual.com"
      "mx2.athletic-annual.com"
    ]
    spam_filter: {
      enabled: true
      threshold: 8.5
      action: "reject"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.athletic-annual.com"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 322
      per_hour: 5843
    }
  }
}
