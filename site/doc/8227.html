Example 8227

K8s Deployment

Kubernetes deployment specification.

{
  apiVersion: "apps/v1"
  kind: "Deployment"
  metadata: {
    name: "draw-worker"
    namespace: "default"
    labels: {
      app: "league"
      tier: "data"
    }
  }

  # Deployment spec
  spec: {
    replicas: 5
    container: {
      image: "swim:0.2.5"
      port: 8080
      resources: {
        cpu_limit: "250m"
        memory_limit: "256Mi"
      }
    }

    health_check: {
      path: "/health"
      interval: 10
      timeout: 5
    }
  }
}
