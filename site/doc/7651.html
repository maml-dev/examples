Example 7651

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "emacs"
  shortcuts: [
    {
      key: "Alt+Up"
      action: "find_references"
      when: "global"
    }

    {
      key: "Alt+Up"
      action: "save"
      when: "global"
    }

    {
      key: "Ctrl+Shift+F"
      action: "toggle_comment"
      when: "global"
    }

    {
      key: "Ctrl+B"
      action: "format"
      when: "editor"
    }

    {
      key: "Cmd+K"
      action: "select_all_occurrences"
      when: "global"
    }
  ]
}
