Example 1221

Terraform Resource

Infrastructure as code resource definition.

{
  provider: "azurerm"
  region: "asia-southeast1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.207.0.0/16"
      enable_dns: true
      tags: {
        Name: "fax-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-e3b06c5d"
      type: "m5.large"
      key_name: "coliseum-key"
      monitoring: true
      root_volume: {
        size: 100
        type: "gp3"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-ssh"
      ingress_port: 443
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
