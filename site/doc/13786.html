Example 13786

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@each-sand.net"
  subject: "Corroboro eveniet commodi saepe theca quibusdam."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Usque abbas venustas viscus aut tenetur condico conduco sodalitas. Civitas succurro thalassinus autus conitor capio vomica viriliter soleo colo. Calcar suscipit ultio temeritas tergum usus ante.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Ancilla decor patior ex audacia denuncio in. Cupio cruentus vicissitudo pecto campana chirographum carmen timidus. Bestia est tabella coniuratio stips voluptatem uredo.

Visit: {{url}}
"""
}
