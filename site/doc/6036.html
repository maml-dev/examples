Example 6036

CI Pipeline

Continuous integration pipeline definition.

{
  name: "Main Pipeline"
  trigger: {
    branches: [
      "main"
      "develop"
    ]
    paths: [
      "src/**"
      "tests/**"
    ]
  }

  stages: {
    # Code quality
    lint: {
      image: "node:18"
      command: "npx biome check"
    }
    test: {
      image: "node:20"
      command: "npm test"
      coverage: true
    }
    build: {
      image: "node:22"
      command: "npm run build"
      artifacts: [
        "dist/**"
        "build/**"
      ]
    }
  }
}
