Example 11373

Test Suite

Test runner configuration with coverage settings.

{
  runner: "mocha"
  root: "./src/__tests__"
  include: [
    "**/*.test.ts"
    "**/*.spec.ts"
  ]
  exclude: [
    "node_modules"
    "dist"
    "**/*.e2e.ts"
  ]

  # Coverage
  coverage: {
    enabled: true
    provider: "istanbul"
    thresholds: {
      lines: 78
      branches: 61
      functions: 92
    }
    reporter: [
      "text"
      "lcov"
      "html"
    ]
  }
  globals: true
  timeout: 5565 # ms
}
