Example 11425

Monitoring Config

Application monitoring and alerting setup.

{
  service: "blockchains"
  # Prometheus metrics
  metrics: {
    endpoint: "/metrics"
    interval: 30 # seconds
    labels: {
      env: "staging"
      region: "eu-west"
      team: "shoes"
    }
  }

  # Alert thresholds
  alerts: {
    cpu_threshold: 75
    memory_threshold: 87
    latency_p99_ms: 1949
    error_rate: 0.01
    notification: {
      channel: "#ops-alerts"
      severity: "info"
    }
  }
}
