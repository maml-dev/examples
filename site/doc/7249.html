Example 7249

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "sublime"
  shortcuts: [
    {
      key: "Cmd+Shift+L"
      action: "toggle_sidebar"
      when: "terminal"
    }

    {
      key: "Cmd+K"
      action: "search_project"
      when: "global"
    }

    {
      key: "Alt+F7"
      action: "find_references"
      when: "global"
    }

    {
      key: "Ctrl+/"
      action: "find_references"
      when: "editor"
    }

    {
      key: "Ctrl+/"
      action: "toggle_terminal"
      when: "editor"
    }

    {
      key: "Alt+Up"
      action: "find_references"
      when: "global"
    }
  ]
}
