Terraform Resource

Infrastructure as code resource definition.

{
  provider: "aws"
  region: "eu-west-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.104.0.0/16"
      enable_dns: true
      tags: {
        Name: "fen-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-26baeaeb"
      type: "t3.medium"
      key_name: "governance-key"
      monitoring: true
      root_volume: {
        size: 20
        type: "gp2"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-web"
      ingress_port: 22
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
