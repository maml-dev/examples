Example 7021

Terraform Resource

Infrastructure as code resource definition.

{
  provider: "azurerm"
  region: "eu-west-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.184.0.0/16"
      enable_dns: true
      tags: {
        Name: "typewriter-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-7d5f4bda"
      type: "t3.small"
      key_name: "forager-key"
      monitoring: true
      root_volume: {
        size: 100
        type: "gp2"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-all"
      ingress_port: 8080
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
