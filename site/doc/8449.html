Example 8449

Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "billing"
      permissions: [
        "read"
        "export_data"
        "write"
        "view_analytics"
        "manage_users"
      ]
      max_seats: 37
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "manage_users"
      ]
      max_seats: 48
    }

    {
      name: "editor"
      permissions: [
        "write"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "delete"
        "manage_users"
        "view_analytics"
        "export_data"
        "write"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "read"
        "manage_users"
        "delete"
        "write"
      ]
      max_seats: 20
    }
  ]
}
