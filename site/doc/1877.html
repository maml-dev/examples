Example 1877

Test Suite

Test runner configuration with coverage settings.

{
  runner: "vitest"
  root: "./tests"
  include: [
    "**/*.test.ts"
    "**/*.spec.ts"
  ]
  exclude: [
    "node_modules"
    "dist"
    "**/*.e2e.ts"
  ]

  # Coverage
  coverage: {
    enabled: true
    provider: "v8"
    thresholds: {
      lines: 84
      branches: 66
      functions: 85
    }
    reporter: [
      "text"
      "lcov"
      "html"
    ]
  }
  globals: false
  timeout: 13580 # ms
}
