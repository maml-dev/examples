Example 8330

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.genuine-fold.org"
    port: 8443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "5m"
    max_requests: 3667
    by: "user"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://deadly-pile.biz"
      "https://app.good-natured-thongs.net"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 7200
  }

  cache: {
    enabled: false
    ttl: 767
    store: "redis"
  }
}
