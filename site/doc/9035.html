Example 9035

Terraform Resource

Infrastructure as code resource definition.

{
  provider: "digitalocean"
  region: "asia-southeast1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.234.0.0/16"
      enable_dns: true
      tags: {
        Name: "bowler-vpc"
        Environment: "production"
      }
    }

    # Compute
    instance: {
      ami: "ami-98bebcf1"
      type: "t3.micro"
      key_name: "inspection-key"
      monitoring: true
      root_volume: {
        size: 100
        type: "gp3"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-all"
      ingress_port: 80
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
