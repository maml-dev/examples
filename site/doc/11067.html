Example 11067

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Goldner Group Dashboard"
  refresh_interval: 300 # seconds
  widgets: [
    {
      type: "chart"
      title: "Active Users"
      source: "/api/metrics/revenue"
      chart_type: "bar"
      period: "7d"
    }

    {
      type: "stat"
      title: "Error Rate"
      source: "/api/stats/users"
      format: "number"
    }

    {
      type: "table"
      title: "Top Pages"
      source: "/api/recent/signups"
      limit: 21
    }
  ]

  theme: "auto"
}
