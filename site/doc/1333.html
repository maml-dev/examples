Example 1333

Queue Config

Message queue and worker configuration.

{
  broker: "redis"
  connection: {
    host: "mq.internal"
    port: 5672
    vhost: "/"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 10
      retry: 4
      timeout: 56305 # ms
    }

    priority: {
      concurrency: 8
      retry: 5
      timeout: 6020 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 9
    queue: "dead-letter"
  }
}
