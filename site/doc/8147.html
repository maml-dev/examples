Example 8147

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vscode"
  shortcuts: [
    {
      key: "F5"
      action: "search_project"
      when: "debug"
    }

    {
      key: "Ctrl+B"
      action: "search_project"
      when: "terminal"
    }

    {
      key: "Ctrl+Shift+P"
      action: "save"
      when: "editor"
    }
  ]
}
