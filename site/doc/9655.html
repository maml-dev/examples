Example 9655

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "sublime"
  shortcuts: [
    {
      key: "Cmd+K"
      action: "format"
      when: "terminal"
    }

    {
      key: "F5"
      action: "toggle_sidebar"
      when: "terminal"
    }

    {
      key: "Ctrl+Shift+F"
      action: "save"
      when: "terminal"
    }

    {
      key: "Ctrl+`"
      action: "toggle_sidebar"
      when: "editor"
    }

    {
      key: "Ctrl+D"
      action: "toggle_comment"
      when: "global"
    }

    {
      key: "Ctrl+`"
      action: "toggle_terminal"
      when: "debug"
    }

    {
      key: "Ctrl+B"
      action: "toggle_sidebar"
      when: "global"
    }
  ]
}
