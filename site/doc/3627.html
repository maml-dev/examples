Example 3627

Queue Config

Message queue and worker configuration.

{
  broker: "redis"
  connection: {
    host: "mq.internal"
    port: 5672
    vhost: "/production"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 3
      retry: 3
      timeout: 59939 # ms
    }

    priority: {
      concurrency: 7
      retry: 9
      timeout: 4134 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 9
    queue: "dead-letter"
  }
}
