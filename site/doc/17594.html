Example 17594

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@thin-atrium.biz"
  subject: "Complectus velum adopto defendo torqueo."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Cado pauper conicio sum auctus aro. Aequitas deleniti curto verecundia perferendis sollers artificiose. Antepono demum talis confugo.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Tribuo conturbo iusto suffoco. Cattus arbustum sordeo thymum conatus amplus sumo antea decipio ad. Nostrum stella dapifer curatio terminatio.

Visit: {{url}}
"""
}
