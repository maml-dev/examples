Example 8030

Queue Config

Message queue and worker configuration.

{
  broker: "sqs"
  connection: {
    host: "localhost"
    port: 6379
    vhost: "/staging"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 8
      retry: 2
      timeout: 36989 # ms
    }

    priority: {
      concurrency: 10
      retry: 8
      timeout: 6044 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 3
    queue: "dead-letter"
  }
}
