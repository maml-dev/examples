Example 5941

Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "write"
        "view_analytics"
      ]
      max_seats: 50
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "view_analytics"
        "delete"
        "read"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "delete"
        "export_data"
      ]
      max_seats: 32
    }
  ]
}
