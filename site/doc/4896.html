Example 4896

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@new-millet.name"
  subject: "Deserunt argentum adopto beneficium tonsor."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Conforto necessitatibus soluta communis. Centum arguo turbo denuo. Vapulus validus iure aspernatur tero.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Patria triduana aeneus solium corpus angelus thesis. Absque crebro pecus aestivus vigilo beatae confido crebro deripio sum. Enim depromo corroboro creo contabesco addo capillus aperio.

Visit: {{url}}
"""
}
