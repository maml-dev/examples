Example 9662

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Klocko LLC Dashboard"
  refresh_interval: 300 # seconds
  widgets: [
    {
      type: "chart"
      title: "Traffic"
      source: "/api/metrics/revenue"
      chart_type: "area"
      period: "30d"
    }

    {
      type: "stat"
      title: "MRR"
      source: "/api/stats/revenue"
      format: "number"
    }

    {
      type: "table"
      title: "Top Pages"
      source: "/api/recent/orders"
      limit: 11
    }
  ]

  theme: "auto"
}
