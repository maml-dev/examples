Example 7046

Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "moderator"
      permissions: [
        "export_data"
        "delete"
        "read"
      ]
      max_seats: 42
    }

    {
      name: "viewer"
      permissions: [
        "manage_billing"
        "view_analytics"
        "export_data"
        "manage_users"
        "read"
      ]
      max_seats: 41
    }

    {
      name: "billing"
      permissions: [
        "read"
        "view_analytics"
      ]
      max_seats: 14
    }
  ]
}
