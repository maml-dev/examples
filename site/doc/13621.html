Example 13621

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.dependable-venom.info"
    port: 8443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "1m"
    max_requests: 4873
    by: "user"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://bossy-unique.biz"
      "https://app.murky-valley.com"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 3600
  }

  cache: {
    enabled: true
    ttl: 2425
    store: "memory"
  }
}
