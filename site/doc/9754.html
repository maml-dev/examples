Example 9754

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "emacs"
  shortcuts: [
    {
      key: "F5"
      action: "find_references"
      when: "terminal"
    }

    {
      key: "Ctrl+/"
      action: "search_project"
      when: "editor"
    }

    {
      key: "Alt+F7"
      action: "toggle_terminal"
      when: "debug"
    }

    {
      key: "Alt+Up"
      action: "command_palette"
      when: "terminal"
    }

    {
      key: "Ctrl+S"
      action: "debug_run"
      when: "global"
    }
  ]
}
