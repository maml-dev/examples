Example 13132

Queue Config

Message queue and worker configuration.

{
  broker: "sqs"
  connection: {
    host: "localhost"
    port: 5672
    vhost: "/staging"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 1
      retry: 2
      timeout: 13572 # ms
    }

    priority: {
      concurrency: 17
      retry: 10
      timeout: 9333 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 4
    queue: "dead-letter"
  }
}
