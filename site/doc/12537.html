Terraform Resource

Infrastructure as code resource definition.

{
  provider: "aws"
  region: "eu-west-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.206.0.0/16"
      enable_dns: true
      tags: {
        Name: "underneath-vpc"
        Environment: "production"
      }
    }

    # Compute
    instance: {
      ami: "ami-f5bd7a8c"
      type: "m5.large"
      key_name: "coast-key"
      monitoring: true
      root_volume: {
        size: 20
        type: "gp2"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-ssh"
      ingress_port: 8080
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
