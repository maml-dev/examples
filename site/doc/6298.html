Example 6298

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@surprised-diver.com"
  subject: "Deduco clementia claro."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Utrum vir spiculum degero doloremque. Victoria studio appello. Succedo volaticus virga nobis umquam celer valens.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Decretum cupiditate desino vulnus delinquo stultus curiositas desidero. Utor comis corroboro inventore incidunt. Crepusculum utique minus ater laborum cubo adstringo.

Visit: {{url}}
"""
}
