Example 6689

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@aggravating-hope.info"
  subject: "Ars culpo porro aperiam ea."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Virga tergum basium quisquam volubilis vestrum adnuo. Carbo vulticulus suscipio sublime usque. Uredo numquam officia.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Eveniet cornu bonus degenero earum curia. Suffoco qui tristis comptus avaritia. Audax quae neque denuncio summopere tolero conservo cilicium ullam vapulus.

Visit: {{url}}
"""
}
