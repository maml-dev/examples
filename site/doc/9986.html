Example 9986

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@inconsequential-mortise.com"
  subject: "Infit angustus cauda distinctio amor ullus."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Depulso coaegresco caute depereo deripio confero totus tolero. Cum colligo cursus cursus iste bardus amplexus bonus ascisco. Nesciunt tenuis depulso stella ratione optio depopulo quo cuppedia.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Carcer comptus deserunt suasoria nihil. Titulus auctus vesco. Accedo usque pecto.

Visit: {{url}}
"""
}
