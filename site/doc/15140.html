Example 15140

Queue Config

Message queue and worker configuration.

{
  broker: "redis"
  connection: {
    host: "localhost"
    port: 6379
    vhost: "/"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 8
      retry: 1
      timeout: 30452 # ms
    }

    priority: {
      concurrency: 5
      retry: 7
      timeout: 1595 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 3
    queue: "dead-letter"
  }
}
