Example 10921

Email Routing

Email routing and filtering rules.

{
  domain: "courageous-season.name"
  inbound: {
    mx_records: [
      "mx1.courageous-season.name"
      "mx2.courageous-season.name"
    ]
    spam_filter: {
      enabled: true
      threshold: 8.5
      action: "quarantine"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 2048
    }
  }

  outbound: {
    relay: "smtp.courageous-season.name"
    port: 465
    tls: true
    rate_limit: {
      per_minute: 203
      per_hour: 9861
    }
  }
}
