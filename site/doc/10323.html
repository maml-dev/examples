Example 10323

Email Routing

Email routing and filtering rules.

{
  domain: "whopping-brief.biz"
  inbound: {
    mx_records: [
      "mx1.whopping-brief.biz"
      "mx2.whopping-brief.biz"
    ]
    spam_filter: {
      enabled: true
      threshold: 8.5
      action: "reject"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.whopping-brief.biz"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 79
      per_hour: 6988
    }
  }
}
