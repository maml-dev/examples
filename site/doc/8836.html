Example 8836

Terraform Resource

Infrastructure as code resource definition.

{
  provider: "digitalocean"
  region: "asia-southeast1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.11.0.0/16"
      enable_dns: true
      tags: {
        Name: "dime-vpc"
        Environment: "production"
      }
    }

    # Compute
    instance: {
      ami: "ami-47beb3ca"
      type: "m5.large"
      key_name: "squid-key"
      monitoring: true
      root_volume: {
        size: 20
        type: "io1"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-all"
      ingress_port: 80
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
