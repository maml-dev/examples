DNS Records

DNS zone record entries.

{
  zone: "winding-forager.com"
  records: [
    {
      type: "AAAA"
      name: "@"
      value: "snarling-sand.info"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "api"
      value: "closed-worth.net"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "www"
      value: "228.231.21.151"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "cdn"
      value: "v=spf1 include:_spf.bulky-bidet.info ~all"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "mail"
      value: "v=spf1 include:_spf.calculating-nun.biz ~all"
      ttl: 3600
    }
  ]
}
