Example 821

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.focused-scratch.info"
    port: 443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "1m"
    max_requests: 5948
    by: "ip"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://dull-colonialism.net"
      "https://app.affectionate-clavicle.org"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 3600
  }

  cache: {
    enabled: true
    ttl: 1459
    store: "memory"
  }
}
