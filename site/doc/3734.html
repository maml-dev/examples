Example 3734

Email Routing

Email routing and filtering rules.

{
  domain: "unfortunate-litter.info"
  inbound: {
    mx_records: [
      "mx1.unfortunate-litter.info"
      "mx2.unfortunate-litter.info"
    ]
    spam_filter: {
      enabled: true
      threshold: 5
      action: "tag"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.unfortunate-litter.info"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 243
      per_hour: 1308
    }
  }
}
