Example 12840

Backup Strategy

Backup schedule and retention policy.

{
  source: {
    type: "mongodb"
    host: "db.internal"
    databases: [
      "mallard"
      "department"
    ]
  }

  # Backup schedule
  schedule: {
    full: "0 2 * * 0" # weekly Sunday 2am
    incremental: "0 2 * * 1-6" # daily except Sunday
    snapshot: "0 */4 * * *" # every 4 hours
  }

  retention: {
    daily: 11
    weekly: 10
    monthly: 21
  }

  destination: {
    type: "sftp"
    path: "/mnt/backups/zebra"
    encryption: true
    compression: "zstd"
  }
}
