Example 17044

API Routes

REST API route definitions.

{
  base_url: "/api/v2"
  routes: [
    {
      method: "POST"
      path: "/products"
      handler: "listProducts"
      auth: true
    }

    {
      method: "PATCH"
      path: "/users"
      handler: "deleteProducts"
      auth: true
    }

    {
      method: "DELETE"
      path: "/users"
      handler: "listProducts"
      auth: true
    }

    {
      method: "POST"
      path: "/products/:id"
      handler: "getProducts"
      auth: false
    }
  ]
}
