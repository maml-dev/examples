Terraform Resource

Infrastructure as code resource definition.

{
  provider: "google"
  region: "eu-west-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.170.0.0/16"
      enable_dns: true
      tags: {
        Name: "sunbeam-vpc"
        Environment: "production"
      }
    }

    # Compute
    instance: {
      ami: "ami-65dcfade"
      type: "t3.small"
      key_name: "avalanche-key"
      monitoring: true
      root_volume: {
        size: 100
        type: "io1"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-all"
      ingress_port: 8080
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
