Example 10969

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Kihn, Waelchi and Torp Dashboard"
  refresh_interval: 60 # seconds
  widgets: [
    {
      type: "chart"
      title: "Revenue"
      source: "/api/metrics/traffic"
      chart_type: "area"
      period: "7d"
    }

    {
      type: "stat"
      title: "Uptime"
      source: "/api/stats/users"
      format: "number"
    }

    {
      type: "table"
      title: "Latest Signups"
      source: "/api/recent/orders"
      limit: 20
    }
  ]

  theme: "auto"
}
