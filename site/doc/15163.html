Example 15163

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Tromp LLC Dashboard"
  refresh_interval: 60 # seconds
  widgets: [
    {
      type: "chart"
      title: "Revenue"
      source: "/api/metrics/traffic"
      chart_type: "line"
      period: "30d"
    }

    {
      type: "stat"
      title: "Total Users"
      source: "/api/stats/revenue"
      format: "currency"
    }

    {
      type: "table"
      title: "Top Pages"
      source: "/api/recent/orders"
      limit: 6
    }
  ]

  theme: "dark"
}
