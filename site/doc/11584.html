Example 11584

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@subdued-mallard.name"
  subject: "Assentator vulnero amita."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Vinco cruciamentum vestrum bonus vaco uter victoria vito. Arbustum subvenio maiores corrumpo terebro sperno. Ad vaco verus solio uterque.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Quo similique vicinus tumultus et tantum. Dolore thermae valetudo tepesco censura accendo peccatus vomito. Tamquam totus conduco.

Visit: {{url}}
"""
}
