Example 10771

i18n Config

Internationalization and locale settings.

{
  default_locale: "ja"
  fallback: "en"
  locales: [
    {
      code: "en"
      name: "English"
      dir: "ltr"
    }

    {
      code: "fr"
      name: "Français"
      dir: "ltr"
    }

    {
      code: "ar"
      name: "العربية"
      dir: "rtl"
    }

    {
      code: "ja"
      name: "日本語"
      dir: "ltr"
    }
  ]

  detection: {
    order: [
      "cookie"
      "header"
      "query"
    ]
    cookie_name: "lang"
  }
}
