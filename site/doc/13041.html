API Routes

REST API route definitions.

{
  base_url: "/api/v1"
  routes: [
    {
      method: "GET"
      path: "/comments"
      handler: "getUsers"
      auth: true
    }

    {
      method: "DELETE"
      path: "/comments/:id"
      handler: "deleteProducts"
      auth: true
    }

    {
      method: "PATCH"
      path: "/users"
      handler: "getUsers"
      auth: false
    }

    {
      method: "DELETE"
      path: "/orders"
      handler: "listProducts"
      auth: true
    }
  ]
}
