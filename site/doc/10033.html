Example 10033

K8s Deployment

Kubernetes deployment specification.

{
  apiVersion: "apps/v1"
  kind: "Deployment"
  metadata: {
    name: "surface-web"
    namespace: "default"
    labels: {
      app: "bonnet"
      tier: "backend"
    }
  }

  # Deployment spec
  spec: {
    replicas: 4
    container: {
      image: "nightlife:6.13.0"
      port: 8000
      resources: {
        cpu_limit: "1000m"
        memory_limit: "256Mi"
      }
    }

    health_check: {
      path: "/ready"
      interval: 10
      timeout: 10
    }
  }
}
