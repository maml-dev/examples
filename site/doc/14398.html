Example 14398

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@ecstatic-drug.org"
  subject: "Coadunatio sponte censura crapula pel."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Despecto nemo ait spes curtus talis comes. Vociferor vita urbs viridis sperno antea necessitatibus delectatio claro. Voluptatum in villa patria denego tyrannus utrimque vapulus claro.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Adversus comitatus summa. Unde maxime aggero ager centum. Caecus quibusdam modi vitiosus.

Visit: {{url}}
"""
}
