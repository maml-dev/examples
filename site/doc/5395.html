Example 5395

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@lively-making.net"
  subject: "Ipsum deprimo velum cubitum sint comprehendo."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Conor vulpes blanditiis decet. Ciminatio creber valetudo voluptatum ex facilis aranea. Vilicus cursus modi aqua.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Cum canto sed suppono sopor carcer argentum arx. Desparatus conicio denuncio veritas vox conturbo vesica delectatio. Velum capitulus vomer ait.

Visit: {{url}}
"""
}
