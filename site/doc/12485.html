Example 12485

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@unconscious-yeast.org"
  subject: "Derideo iusto arcesso virgo cattus amoveo."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Ultio thymbra abundans thesaurus traho utor. Territo compono umquam xiphias aveho antepono advoco. Vesica modi timor tabgo tamisium.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Culpo aperio bene vulticulus currus. Succedo laborum ait caelum ter aeneus amplus quasi aegrotatio asporto. Maxime voluntarius natus illum denuo admiratio crapula vociferor.

Visit: {{url}}
"""
}
