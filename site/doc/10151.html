Example 10151

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vscode"
  shortcuts: [
    {
      key: "F5"
      action: "debug_run"
      when: "debug"
    }

    {
      key: "Ctrl+D"
      action: "toggle_comment"
      when: "debug"
    }

    {
      key: "Cmd+Shift+L"
      action: "debug_run"
      when: "debug"
    }
  ]
}
