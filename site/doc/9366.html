Docker Compose

Multi-container application definition.

{
  version: "3.8"
  services: {
    app: {
      image: "node:18-alpine"
      ports: [
        "3105:3746"
      ]
      environment: {
        NODE_ENV: "production"
        DATABASE_URL: "postgres://db:5432/foodstuffs"
      }
      depends_on: [
        "db"
        "redis"
      ]
    }

    db: {
      image: "postgres:16"
      volumes: [
        "db_data:/var/lib/postgresql/data"
      ]
      environment: {
        POSTGRES_DB: "obligation"
        POSTGRES_PASSWORD: "uBhR6RQ8h_9oGLi"
      }
    }

    redis: {
      image: "redis:7-alpine"
      ports: [
        "6379:6379"
      ]
    }
  }

  volumes: {
    db_data: {}
  }
}
