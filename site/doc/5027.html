Example 5027

K8s Deployment

Kubernetes deployment specification.

{
  apiVersion: "apps/v1"
  kind: "Deployment"
  metadata: {
    name: "yeast-api"
    namespace: "production"
    labels: {
      app: "recommendation"
      tier: "backend"
    }
  }

  # Deployment spec
  spec: {
    replicas: 4
    container: {
      image: "sonar:9.14.13"
      port: 8000
      resources: {
        cpu_limit: "500m"
        memory_limit: "256Mi"
      }
    }

    health_check: {
      path: "/health"
      interval: 10
      timeout: 5
    }
  }
}
