Example 1222

Email Routing

Email routing and filtering rules.

{
  domain: "judicious-recommendation.name"
  inbound: {
    mx_records: [
      "mx1.judicious-recommendation.name"
      "mx2.judicious-recommendation.name"
    ]
    spam_filter: {
      enabled: true
      threshold: 7
      action: "tag"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 2048
    }
  }

  outbound: {
    relay: "smtp.judicious-recommendation.name"
    port: 25
    tls: true
    rate_limit: {
      per_minute: 180
      per_hour: 8272
    }
  }
}
