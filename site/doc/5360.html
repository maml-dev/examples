Example 5360

Cron Jobs

Scheduled task definitions.

{
  timezone: "America/New_York"
  jobs: [
    {
      name: "cleanup-temp"
      schedule: "30 6 * * 1-5"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "refresh-cache"
      schedule: "0 0 * * *"
      command: "python manage.py report"
      enabled: false
    }

    {
      name: "backup-db"
      schedule: "30 6 * * 1-5"
      command: "node scripts/cleanup.js"
      enabled: false
    }
  ]
}
