Example 14022

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.breakable-tomb.net"
    port: 443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "5m"
    max_requests: 3560
    by: "ip"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://uncommon-pearl.net"
      "https://app.impure-gloom.com"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 3600
  }

  cache: {
    enabled: false
    ttl: 2403
    store: "redis"
  }
}
