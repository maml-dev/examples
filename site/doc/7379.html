Example 7379

Test Suite

Test runner configuration with coverage settings.

{
  runner: "vitest"
  root: "./test"
  include: [
    "**/*.test.ts"
    "**/*.spec.ts"
  ]
  exclude: [
    "node_modules"
    "dist"
    "**/*.e2e.ts"
  ]

  # Coverage
  coverage: {
    enabled: true
    provider: "c8"
    thresholds: {
      lines: 90
      branches: 79
      functions: 94
    }
    reporter: [
      "text"
      "lcov"
      "html"
    ]
  }
  globals: true
  timeout: 20027 # ms
}
