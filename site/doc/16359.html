Example 16359

API Routes

REST API route definitions.

{
  base_url: "/api/v3"
  routes: [
    {
      method: "PATCH"
      path: "/posts/:id"
      handler: "deleteUsers"
      auth: true
    }

    {
      method: "GET"
      path: "/products"
      handler: "deleteOrders"
      auth: true
    }

    {
      method: "PATCH"
      path: "/posts/:id"
      handler: "createProducts"
      auth: true
    }

    {
      method: "POST"
      path: "/products"
      handler: "getOrders"
      auth: true
    }

    {
      method: "POST"
      path: "/products/:id"
      handler: "listUsers"
      auth: true
    }
  ]
}
