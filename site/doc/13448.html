Example 13448

Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "admin"
      permissions: [
        "export_data"
        "write"
        "manage_billing"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "write"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "manage_users"
        "read"
      ]
      max_seats: 12
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "read"
        "manage_users"
        "delete"
        "write"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "read"
        "write"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "export_data"
        "delete"
        "view_analytics"
      ]
      max_seats: 16
    }
  ]
}
