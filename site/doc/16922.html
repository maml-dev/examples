Terraform Resource

Infrastructure as code resource definition.

{
  provider: "azurerm"
  region: "asia-southeast1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.189.0.0/16"
      enable_dns: true
      tags: {
        Name: "tapioca-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-ecce6e93"
      type: "t3.micro"
      key_name: "lava-key"
      monitoring: true
      root_volume: {
        size: 100
        type: "gp3"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-ssh"
      ingress_port: 22
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
