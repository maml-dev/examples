Example 7883

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@same-diversity.info"
  subject: "Absum vomica cauda tutis utroque reprehenderit."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Despecto tergum votum tardus ulterius abbas deduco adflicto cometes. Velit tamen careo suus. Pariatur deleniti absum tergum vestigium spiritus demergo beatae solvo explicabo.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Vel suscipio cupressus repudiandae. Sono infit appositus. Carbo aro synagoga.

Visit: {{url}}
"""
}
