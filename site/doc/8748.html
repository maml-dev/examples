{
  zone: "wordy-availability.org"
  records: [
    {
      type: "A"
      name: "www"
      value: "blushing-coordination.biz"
      ttl: 300

    {
      type: "A"
      name: "www"
      value: "v=spf1 include:_spf.oddball-pinstripe.info ~all"
      ttl: 3600
    }

    {
      type: "A"
      name: "staging"
      value: "v=spf1 include:_spf.muddy-carnival.org ~all"
      ttl: 300
    }

      type: "MX"
      name: "mail"
      value: "enchanted-subexpression.biz"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "www"
      value: "stained-violin.biz"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "api"
      value: "240.55.35.207"
      ttl: 300
    }
  ]
}
