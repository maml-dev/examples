  provider: "google"
  region: "us-east-1"

    # Network
    vpc: {
      cidr_block: "10.221.0.0/16"

      tags: {
        Name: "ethyl-vpc"

    }

    # Compute

      ami: "ami-425f77cc"
      type: "t3.small"

      monitoring: true
      root_volume: {
        size: 20

    # Security

      name: "allow-ssh"
      ingress_port: 443
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
