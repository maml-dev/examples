Docker Compose

Multi-container application definition.

{
  version: "3.8"
  services: {
    app: {
      image: "node:20-slim"
      ports: [
        "3693:3571"
      ]
      environment: {
        NODE_ENV: "production"
        DATABASE_URL: "postgres://db:5432/lotion"
      }
      depends_on: [
        "db"
        "redis"
      ]
    }

    db: {
      image: "postgres:16"
      volumes: [
        "db_data:/var/lib/postgresql/data"
      ]
      environment: {
        POSTGRES_DB: "outlaw"
        POSTGRES_PASSWORD: "D5nOPpL_myReOWc"
      }
    }

    redis: {
      image: "redis:7-alpine"
      ports: [
        "6379:6379"
      ]
    }
  }

  volumes: {
    db_data: {}
  }
}
