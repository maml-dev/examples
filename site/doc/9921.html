Example 9921

Terraform Resource

Infrastructure as code resource definition.

{
  provider: "azurerm"
  region: "eu-west-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.198.0.0/16"
      enable_dns: true
      tags: {
        Name: "manner-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-8fe1383e"
      type: "t3.micro"
      key_name: "pension-key"
      monitoring: true
      root_volume: {
        size: 50
        type: "gp3"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-ssh"
      ingress_port: 8080
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
