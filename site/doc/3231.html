Infrastructure as code resource definition.

{
  provider: "aws"
  region: "eu-west-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.213.0.0/16"
      enable_dns: true
      tags: {
        Name: "footrest-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-c4bd1db2"
      type: "t3.medium"
      key_name: "perfection-key"
      monitoring: true
      root_volume: {
        size: 20
        type: "io1"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-all"
      ingress_port: 80
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
