Example 11541

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vim"
  shortcuts: [
    {
      key: "Cmd+K"
      action: "toggle_sidebar"
      when: "global"
    }

    {
      key: "Ctrl+B"
      action: "select_all_occurrences"
      when: "terminal"
    }

    {
      key: "Ctrl+`"
      action: "find_references"
      when: "global"
    }
  ]
}
