Example 083

SQL Queries

Database queries stored as raw strings.

# Database query definitions
{
  name: "user-reports"
  version: 1

  queries: {
    # Find all active users
    find_active: """
SELECT u.id, u.name, u.email
FROM users u
WHERE u.active = true
  AND u.created_at > '2024-01-01'
ORDER BY u.name
"""

    # Monthly aggregation
    monthly_stats: """
SELECT
  DATE_TRUNC('month', created_at) AS month,
  COUNT(*) AS total,
  SUM(amount) AS revenue
FROM orders
GROUP BY month
ORDER BY month DESC
"""
  }
}
