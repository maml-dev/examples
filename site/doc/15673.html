Example 15673

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Pfeffer and Sons Dashboard"
  refresh_interval: 300 # seconds
  widgets: [
    {
      type: "chart"
      title: "Active Users"
      source: "/api/metrics/traffic"
      chart_type: "line"
      period: "90d"
    }

    {
      type: "stat"
      title: "Error Rate"
      source: "/api/stats/revenue"
      format: "currency"
    }

    {
      type: "table"
      title: "Recent Orders"
      source: "/api/recent/signups"
      limit: 14
    }
  ]

  theme: "auto"
}
