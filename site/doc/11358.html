Example 11358

Cron Jobs

Scheduled task definitions.

{
  timezone: "America/New_York"
  jobs: [
    {
      name: "sync-data"
      schedule: "*/15 * * * *"
      command: "python manage.py report"
      enabled: false
    }

    {
      name: "send-reports"
      schedule: "0 0 * * *"
      command: "./bin/sync --all"
      enabled: true
    }

    {
      name: "send-reports"
      schedule: "0 0 * * *"
      command: "python manage.py report"
      enabled: false
    }
  ]
}
