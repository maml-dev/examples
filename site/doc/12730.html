Example 12730

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.sweet-almighty.org"
    port: 443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "5m"
    max_requests: 1334
    by: "ip"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://lone-compromise.name"
      "https://app.blank-plumber.net"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 86400
  }

  cache: {
    enabled: true
    ttl: 1125
    store: "redis"
  }
}
