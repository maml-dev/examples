Example 16525

Queue Config

Message queue and worker configuration.

{
  broker: "rabbitmq"
  connection: {
    host: "localhost"
    port: 9092
    vhost: "/staging"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 10
      retry: 1
      timeout: 8665 # ms
    }

    priority: {
      concurrency: 20
      retry: 6
      timeout: 5520 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 10
    queue: "dead-letter"
  }
}
