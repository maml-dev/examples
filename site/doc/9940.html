Example 9940

Monitoring Config

Application monitoring and alerting setup.

{
  service: "channels"
  # Prometheus metrics
  metrics: {
    endpoint: "/metrics"
    interval: 10 # seconds
    labels: {
      env: "prod"
      region: "ap-south"
      team: "shoes"
    }
  }

  # Alert thresholds
  alerts: {
    cpu_threshold: 70
    memory_threshold: 77
    latency_p99_ms: 708
    error_rate: 0.02
    notification: {
      channel: "#ops-alerts"
      severity: "critical"
    }
  }
}
