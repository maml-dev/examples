Example 4237

Email Routing

Email routing and filtering rules.

{
  domain: "wee-safe.info"
  inbound: {
    mx_records: [
      "mx1.wee-safe.info"
      "mx2.wee-safe.info"
    ]
    spam_filter: {
      enabled: true
      threshold: 7
      action: "quarantine"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 2048
    }
  }

  outbound: {
    relay: "smtp.wee-safe.info"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 239
      per_hour: 9654
    }
  }
}
