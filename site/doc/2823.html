Example 2823

Terraform Resource

Infrastructure as code resource definition.

{
  provider: "azurerm"
  region: "eu-west-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.200.0.0/16"
      enable_dns: true
      tags: {
        Name: "lieu-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-9aebe5b6"
      type: "m5.large"
      key_name: "object-key"
      monitoring: true
      root_volume: {
        size: 50
        type: "gp3"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-ssh"
      ingress_port: 8080
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
