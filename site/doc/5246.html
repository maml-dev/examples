Example 5246

Cron Jobs

Scheduled task definitions.

{
  timezone: "Europe/London"
  jobs: [
    {
      name: "send-reports"
      schedule: "0 * * * *"
      command: "node scripts/cleanup.js"
      enabled: false
    }

    {
      name: "rotate-logs"
      schedule: "0 0 * * *"
      command: "./bin/sync --all"
      enabled: true
    }

    {
      name: "send-reports"
      schedule: "30 6 * * 1-5"
      command: "python manage.py report"
      enabled: false
    }

    {
      name: "rotate-logs"
      schedule: "*/15 * * * *"
      command: "./bin/sync --all"
      enabled: false
    }

    {
      name: "sync-data"
      schedule: "0 0 * * *"
      command: "./bin/sync --all"
      enabled: false
    }
  ]
}
