Example 8231

Terraform Resource

Infrastructure as code resource definition.

{
  provider: "aws"
  region: "eu-west-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.52.0.0/16"
      enable_dns: true
      tags: {
        Name: "fellow-vpc"
        Environment: "production"
      }
    }

    # Compute
    instance: {
      ami: "ami-3e35e5de"
      type: "t3.small"
      key_name: "pressure-key"
      monitoring: true
      root_volume: {
        size: 20
        type: "io1"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-all"
      ingress_port: 443
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
