Example 16632

Email Routing

Email routing and filtering rules.

{
  domain: "innocent-pigpen.org"
  inbound: {
    mx_records: [
      "mx1.innocent-pigpen.org"
      "mx2.innocent-pigpen.org"
    ]
    spam_filter: {
      enabled: true
      threshold: 7
      action: "reject"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 2048
    }
  }

  outbound: {
    relay: "smtp.innocent-pigpen.org"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 368
      per_hour: 1201
    }
  }
}
