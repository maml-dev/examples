Example 2854

Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "editor"
      permissions: [
        "manage_users"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "read"
        "manage_billing"
        "delete"
      ]
      max_seats: 46
    }

    {
      name: "support"
      permissions: [
        "read"
        "delete"
      ]
      max_seats: null
    }
  ]
}
