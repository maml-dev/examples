Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Mante and Sons Dashboard"
  refresh_interval: 300 # seconds
  widgets: [
    {
      type: "chart"
      title: "Revenue"
      source: "/api/metrics/traffic"
      chart_type: "area"
      period: "7d"
    }

    {
      type: "stat"
      title: "MRR"
      source: "/api/stats/revenue"
      format: "currency"
    }

    {
      type: "table"
      title: "Recent Orders"
      source: "/api/recent/signups"
      limit: 10
    }
  ]

  theme: "auto"
}
