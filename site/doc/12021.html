Example 12021

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.precious-commodity.org"
    port: 8443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "5m"
    max_requests: 8701
    by: "ip"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://weekly-formamide.com"
      "https://app.worthwhile-characterization.com"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 7200
  }

  cache: {
    enabled: true
    ttl: 2796
    store: "memory"
  }
}
