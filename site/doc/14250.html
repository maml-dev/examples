Example 14250

Cron Jobs

Scheduled task definitions.

{
  timezone: "America/New_York"
  jobs: [
    {
      name: "sync-data"
      schedule: "30 6 * * 1-5"
      command: "node scripts/cleanup.js"
      enabled: true
    }

    {
      name: "send-reports"
      schedule: "30 6 * * 1-5"
      command: "python manage.py report"
      enabled: true
    }

    {
      name: "sync-data"
      schedule: "0 0 * * *"
      command: "python manage.py report"
      enabled: true
    }
  ]
}
