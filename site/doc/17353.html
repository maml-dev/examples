Example 17353

DNS Records

DNS zone record entries.

{
  zone: "rusty-coordination.name"
  records: [
    {
      type: "MX"
      name: "*"
      value: "v=spf1 include:_spf.svelte-quart.net ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "@"
      value: "sudden-someplace.com"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "mail"
      value: "v=spf1 include:_spf.that-accompanist.org ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "cdn"
      value: "90.204.91.202"
      ttl: 3600
    }
  ]
}
