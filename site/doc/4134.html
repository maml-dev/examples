Queue Config

Message queue and worker configuration.

{
  broker: "redis"
  connection: {
    host: "localhost"
    port: 9092
    vhost: "/production"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 5
      retry: 1
      timeout: 43020 # ms
    }

    priority: {
      concurrency: 14
      retry: 9
      timeout: 6867 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 3
    queue: "dead-letter"
  }
}
