Example 16670

Test Suite

Test runner configuration with coverage settings.

{
  runner: "vitest"
  root: "./test"
  include: [
    "**/*.test.ts"
    "**/*.spec.ts"
  ]
  exclude: [
    "node_modules"
    "dist"
    "**/*.e2e.ts"
  ]

  # Coverage
  coverage: {
    enabled: true
    provider: "c8"
    thresholds: {
      lines: 79
      branches: 60
      functions: 82
    }
    reporter: [
      "text"
      "lcov"
      "html"
    ]
  }
  globals: true
  timeout: 17175 # ms
}
