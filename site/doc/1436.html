Example 1436

Queue Config

Message queue and worker configuration.

{
  broker: "rabbitmq"
  connection: {
    host: "localhost"
    port: 5672
    vhost: "/staging"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 6
      retry: 1
      timeout: 33326 # ms
    }

    priority: {
      concurrency: 16
      retry: 8
      timeout: 5846 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 8
    queue: "dead-letter"
  }
}
