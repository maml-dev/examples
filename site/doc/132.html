Example 132

K8s Deployment

Kubernetes deployment specification.

{
  apiVersion: "apps/v1"
  kind: "Deployment"
  metadata: {
    name: "information-web"
    namespace: "production"
    labels: {
      app: "ruin"
      tier: "data"
    }
  }

  # Deployment spec
  spec: {
    replicas: 4
    container: {
      image: "meal:7.3.16"
      port: 9090
      resources: {
        cpu_limit: "1000m"
        memory_limit: "512Mi"
      }
    }

    health_check: {
      path: "/ready"
      interval: 15
      timeout: 10
    }
  }
}
