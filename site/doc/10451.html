DNS Records

DNS zone record entries.

{
  zone: "stale-illusion.name"
  records: [
    {
      type: "AAAA"
      name: "mail"
      value: "146.44.93.28"
      ttl: 86400
    }

    {
      type: "MX"
      name: "@"
      value: "v=spf1 include:_spf.ruddy-editor.name ~all"
      ttl: 300
    }

    {
      type: "A"
      name: "*"
      value: "207.255.5.98"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "staging"
      value: "163.113.152.83"
      ttl: 300
    }
  ]
}
