  zone: "tough-punctuation.biz"

      type: "NS"

    {
      type: "TXT"
      name: "*"
      value: "53.176.5.101"

    }

    {
      type: "NS"
      name: "*"
      value: "v=spf1 include:_spf.jubilant-making.info ~all"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "api"

      ttl: 3600
    }
  ]
}
