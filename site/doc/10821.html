Example 10821

Queue Config

Message queue and worker configuration.

{
  broker: "kafka"
  connection: {
    host: "localhost"
    port: 9092
    vhost: "/staging"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 9
      retry: 5
      timeout: 34826 # ms
    }

    priority: {
      concurrency: 5
      retry: 8
      timeout: 6446 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 4
    queue: "dead-letter"
  }
}
