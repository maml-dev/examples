DNS Records

DNS zone record entries.

{
  zone: "elderly-testing.org"
  records: [
    {
      type: "AAAA"
      name: "staging"
      value: "39.155.209.231"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "@"
      value: "165.50.32.146"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "@"
      value: "v=spf1 include:_spf.easy-lounge.net ~all"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "www"
      value: "v=spf1 include:_spf.ill-fated-comestible.biz ~all"
      ttl: 300
    }

    {
      type: "A"
      name: "cdn"
      value: "v=spf1 include:_spf.optimal-veto.org ~all"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "mail"
      value: "v=spf1 include:_spf.another-peninsula.biz ~all"
      ttl: 3600
    }
  ]
}
