Example 12737

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.tame-airmail.name"
    port: 8443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "1m"
    max_requests: 2050
    by: "user"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://inexperienced-clavicle.info"
      "https://app.gray-utilization.org"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 3600
  }

  cache: {
    enabled: true
    ttl: 2136
    store: "redis"
  }
}
