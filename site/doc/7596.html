Example 7596

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@blue-platypus.org"
  subject: "Quod cupio corrumpo."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Custodia thalassinus vel harum ater aspernatur caelum cetera. Timidus cursim confero voluntarius tempore. Veritatis defungo claro tener candidus valens aestus.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Dedecor dolores color accommodo doloribus tum attollo. Id venio aegrotatio earum sollicito. Solio totidem tremo una spectaculum avaritia tamquam nam volaticus.

Visit: {{url}}
"""
}
