Example 4980

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Rutherford - Williamson Dashboard"
  refresh_interval: 60 # seconds
  widgets: [
    {
      type: "chart"
      title: "Revenue"
      source: "/api/metrics/traffic"
      chart_type: "bar"
      period: "90d"
    }

    {
      type: "stat"
      title: "Error Rate"
      source: "/api/stats/users"
      format: "percent"
    }

    {
      type: "table"
      title: "Recent Orders"
      source: "/api/recent/signups"
      limit: 22
    }
  ]

  theme: "auto"
}
