Queue Config

Message queue and worker configuration.

{
  broker: "kafka"
  connection: {
    host: "mq.internal"
    port: 9092
    vhost: "/"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 3
      retry: 2
      timeout: 25311 # ms
    }

    priority: {
      concurrency: 9
      retry: 6
      timeout: 8783 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 4
    queue: "dead-letter"
  }
}
