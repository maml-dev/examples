Example 8597

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@brisk-fledgling.biz"
  subject: "Acies combibo urbs clamo."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Aegre atrocitas vestrum. Tutis centum suppono uberrime cornu subseco desipio. Taedium triumphus astrum.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Ars cursus calculus ademptio taceo confugo depromo convoco accommodo conculco. Caelestis defessus color abbas apud odio tot deripio carmen. Defluo conor comptus vitiosus territo ipsam sint.

Visit: {{url}}
"""
}
