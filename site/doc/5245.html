DNS Records

DNS zone record entries.

{
  zone: "grouchy-colonialism.net"
  records: [
    {
      type: "AAAA"
      name: "api"
      value: "28.106.129.25"
      ttl: 300
    }

    {
      type: "MX"
      name: "cdn"
      value: "deadly-instructor.com"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "*"
      value: "silver-puppet.biz"
      ttl: 300
    }

    {
      type: "TXT"
      name: "@"
      value: "v=spf1 include:_spf.rewarding-luck.biz ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "mail"
      value: "v=spf1 include:_spf.phony-shore.name ~all"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "@"
      value: "112.62.147.113"
      ttl: 300
    }
  ]
}
