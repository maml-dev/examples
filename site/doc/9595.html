Example 9595

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@international-representation.org"
  subject: "Sursum aequus conculco atqui admiratio catena."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Vaco stultus sollicito cimentarius suspendo audio. Uter denuo avarus auxilium cunae terminatio supellex. Temptatio balbus ter vivo deinde uterque subvenio vere culpo depereo.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Vicinus acidus vetus. Texo sapiente tabernus ascit temptatio adamo ad omnis soluta. Vigilo studio undique rem aggredior nesciunt.

Visit: {{url}}
"""
}
