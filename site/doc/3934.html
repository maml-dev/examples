Example 3934

K8s Deployment

Kubernetes deployment specification.

{
  apiVersion: "apps/v1"
  kind: "Deployment"
  metadata: {
    name: "disconnection-api"
    namespace: "default"
    labels: {
      app: "alligator"
      tier: "data"
    }
  }

  # Deployment spec
  spec: {
    replicas: 2
    container: {
      image: "strait:0.2.4"
      port: 3000
      resources: {
        cpu_limit: "250m"
        memory_limit: "1Gi"
      }
    }

    health_check: {
      path: "/ready"
      interval: 30
      timeout: 5
    }
  }
}
