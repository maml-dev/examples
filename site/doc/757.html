Example 757

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vim"
  shortcuts: [
    {
      key: "Ctrl+`"
      action: "save"
      when: "debug"
    }

    {
      key: "Alt+F7"
      action: "search_project"
      when: "terminal"
    }

    {
      key: "Ctrl+S"
      action: "select_all_occurrences"
      when: "terminal"
    }

    {
      key: "Ctrl+Shift+P"
      action: "debug_run"
      when: "editor"
    }

    {
      key: "Ctrl+S"
      action: "find_references"
      when: "debug"
    }
  ]
}
