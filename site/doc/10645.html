Example 10645

Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "write"
      ]
      max_seats: 41
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "manage_users"
        "write"
      ]
      max_seats: 20
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "manage_users"
        "manage_billing"
        "view_analytics"
      ]
      max_seats: 37
    }
  ]
}
