Example 17372

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Boyer, Ortiz and Will Dashboard"
  refresh_interval: 60 # seconds
  widgets: [
    {
      type: "chart"
      title: "Revenue"
      source: "/api/metrics/revenue"
      chart_type: "area"
      period: "90d"
    }

    {
      type: "stat"
      title: "Uptime"
      source: "/api/stats/users"
      format: "number"
    }

    {
      type: "table"
      title: "Top Pages"
      source: "/api/recent/orders"
      limit: 20
    }
  ]

  theme: "dark"
}
