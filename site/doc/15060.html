Cron Jobs

Scheduled task definitions.

{
  timezone: "America/New_York"
  jobs: [
    {
      name: "send-reports"
      schedule: "0 2 * * 0"
      command: "node scripts/cleanup.js"
      enabled: false
    }

    {
      name: "sync-data"
      schedule: "0 * * * *"
      command: "node scripts/cleanup.js"
      enabled: false
    }

    {
      name: "backup-db"
      schedule: "0 0 * * *"
      command: "node scripts/cleanup.js"
      enabled: false
    }
  ]
}
