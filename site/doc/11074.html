Example 11074

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Cole and Sons Dashboard"
  refresh_interval: 30 # seconds
  widgets: [
    {
      type: "chart"
      title: "Signups"
      source: "/api/metrics/traffic"
      chart_type: "area"
      period: "90d"
    }

    {
      type: "stat"
      title: "Error Rate"
      source: "/api/stats/users"
      format: "number"
    }

    {
      type: "table"
      title: "Top Pages"
      source: "/api/recent/signups"
      limit: 22
    }
  ]

  theme: "light"
}
