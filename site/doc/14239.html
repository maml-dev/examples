Example 14239

Email Routing

Email routing and filtering rules.

{
  domain: "sure-footed-luck.org"
  inbound: {
    mx_records: [
      "mx1.sure-footed-luck.org"
      "mx2.sure-footed-luck.org"
    ]
    spam_filter: {
      enabled: true
      threshold: 7
      action: "quarantine"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.sure-footed-luck.org"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 171
      per_hour: 3014
    }
  }
}
