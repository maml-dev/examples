Example 6193

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@coordinated-colonialism.biz"
  subject: "Solio architecto cohibeo defetiscor volubilis."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Distinctio tristis talio dens culpo cilicium ocer decretum conculco attonbitus. Comptus corrumpo solvo verbera repellat conatus expedita consuasor toties tondeo. Desolo arcus anser adeo clam cursus vir supplanto.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Vester natus architecto. Suspendo spectaculum stella minus cubitum sono. Volaticus demonstro eligendi cursim statua depono synagoga aestus cunae solio.

Visit: {{url}}
"""
}
