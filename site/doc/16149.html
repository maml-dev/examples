Example 16149

API Routes

REST API route definitions.

{
  base_url: "/api/v2"
  routes: [
    {
      method: "PATCH"
      path: "/orders/:id"
      handler: "deletePosts"
      auth: true
    }

    {
      method: "PUT"
      path: "/orders/:id"
      handler: "deletePosts"
      auth: true
    }

    {
      method: "PATCH"
      path: "/comments"
      handler: "listPosts"
      auth: false
    }

    {
      method: "POST"
      path: "/orders/:id"
      handler: "updateUsers"
      auth: true
    }

    {
      method: "DELETE"
      path: "/orders"
      handler: "listUsers"
      auth: true
    }

    {
      method: "POST"
      path: "/users"
      handler: "getPosts"
      auth: true
    }

    {
      method: "DELETE"
      path: "/products"
      handler: "deleteProducts"
      auth: false
    }
  ]
}
