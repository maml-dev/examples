Example 1474

Docker Compose

Multi-container application definition.

{
  version: "3.8"
  services: {
    app: {
      image: "node:20-slim"
      ports: [
        "3941:3890"
      ]
      environment: {
        NODE_ENV: "production"
        DATABASE_URL: "postgres://db:5432/tune-up"
      }
      depends_on: [
        "db"
        "redis"
      ]
    }

    db: {
      image: "mysql:8"
      volumes: [
        "db_data:/var/lib/postgresql/data"
      ]
      environment: {
        POSTGRES_DB: "formation"
        POSTGRES_PASSWORD: "7QP84v5ig7Kj5Sq"
      }
    }

    redis: {
      image: "redis:7-alpine"
      ports: [
        "6379:6379"
      ]
    }
  }

  volumes: {
    db_data: {}
  }
}
