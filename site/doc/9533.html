Example 9533

Queue Config

Message queue and worker configuration.

{
  broker: "redis"
  connection: {
    host: "mq.internal"
    port: 6379
    vhost: "/production"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 5
      retry: 3
      timeout: 20230 # ms
    }

    priority: {
      concurrency: 16
      retry: 9
      timeout: 8424 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 4
    queue: "dead-letter"
  }
}
