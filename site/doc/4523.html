Example 4523

API Gateway

API gateway routing and middleware configuration.

{
  gateway: {
    host: "api.tattered-procurement.name"
    port: 8443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "1m"
    max_requests: 209
    by: "ip"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://wiggly-pigsty.org"
      "https://app.unlucky-pocket-watch.org"
    ]
    methods: [
      "GET"
      "POST"
      "PUT"
      "DELETE"
    ]
    max_age: 7200
  }

  cache: {
    enabled: false
    ttl: 1868
    store: "redis"
  }
}
