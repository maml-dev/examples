Example 9138

Email Routing

Email routing and filtering rules.

{
  domain: "impartial-deck.net"
  inbound: {
    mx_records: [
      "mx1.impartial-deck.net"
      "mx2.impartial-deck.net"
    ]
    spam_filter: {
      enabled: true
      threshold: 5
      action: "tag"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.impartial-deck.net"
    port: 465
    tls: true
    rate_limit: {
      per_minute: 203
      per_hour: 4829
    }
  }
}
