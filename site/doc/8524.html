Terraform Resource

Infrastructure as code resource definition.

{
  provider: "azurerm"
  region: "us-east-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.214.0.0/16"
      enable_dns: true
      tags: {
        Name: "challenge-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-dde4ce48"
      type: "t3.medium"
      key_name: "descendant-key"
      monitoring: true
      root_volume: {
        size: 20
        type: "gp3"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-all"
      ingress_port: 8080
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
