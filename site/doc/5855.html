DNS Records

DNS zone record entries.

{
  zone: "early-eternity.com"
  records: [
    {
      type: "CNAME"
      name: "api"
      value: "v=spf1 include:_spf.jam-packed-dead.biz ~all"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "api"
      value: "v=spf1 include:_spf.bright-tooth.net ~all"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "@"
      value: "255.207.253.45"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "www"
      value: "56.60.136.231"
      ttl: 3600
    }

    {
      type: "NS"
      name: "@"
      value: "handsome-secret.org"
      ttl: 86400
    }
  ]
}
