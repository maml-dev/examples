API Routes

REST API route definitions.

{
  base_url: "/api/v2"
  routes: [
    {
      method: "PATCH"
      path: "/products/:id"
      handler: "deleteOrders"
      auth: true
    }

    {
      method: "DELETE"
      path: "/orders/:id"
      handler: "listOrders"
      auth: true
    }

    {
      method: "PATCH"
      path: "/orders"
      handler: "updateProducts"
      auth: true
    }

    {
      method: "GET"
      path: "/posts"
      handler: "deleteOrders"
      auth: true
    }

    {
      method: "DELETE"
      path: "/orders/:id"
      handler: "createProducts"
      auth: true
    }

    {
      method: "PUT"
      path: "/products"
      handler: "updateProducts"
      auth: false
    }

    {
      method: "PUT"
      path: "/orders"
      handler: "updateProducts"
      auth: true
    }
  ]
}
