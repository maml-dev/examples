Example 15530

Queue Config

Message queue and worker configuration.

{
  broker: "redis"
  connection: {
    host: "mq.internal"
    port: 5672
    vhost: "/staging"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 10
      retry: 4
      timeout: 41386 # ms
    }

    priority: {
      concurrency: 20
      retry: 10
      timeout: 7544 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 9
    queue: "dead-letter"
  }
}
