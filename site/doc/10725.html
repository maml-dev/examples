Example 10725

Queue Config

Message queue and worker configuration.

{
  broker: "redis"
  connection: {
    host: "mq.internal"
    port: 6379
    vhost: "/production"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 10
      retry: 3
      timeout: 19360 # ms
    }

    priority: {
      concurrency: 14
      retry: 9
      timeout: 9736 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 4
    queue: "dead-letter"
  }
}
