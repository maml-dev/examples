Example 9023

K8s Deployment

Kubernetes deployment specification.

{
  apiVersion: "apps/v1"
  kind: "Deployment"
  metadata: {
    name: "antelope-worker"
    namespace: "production"
    labels: {
      app: "fax"
      tier: "backend"
    }
  }

  # Deployment spec
  spec: {
    replicas: 1
    container: {
      image: "cork:8.15.5"
      port: 9090
      resources: {
        cpu_limit: "250m"
        memory_limit: "512Mi"
      }
    }

    health_check: {
      path: "/health"
      interval: 30
      timeout: 5
    }
  }
}
