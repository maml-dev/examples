Example 9041

API Routes

REST API route definitions.

{
  base_url: "/api/v3"
  routes: [
    {
      method: "PUT"
      path: "/posts/:id"
      handler: "listPosts"
      auth: false
    }

    {
      method: "POST"
      path: "/products"
      handler: "deletePosts"
      auth: false
    }

    {
      method: "DELETE"
      path: "/posts/:id"
      handler: "deletePosts"
      auth: true
    }
  ]
}
