API Routes

REST API route definitions.

{
  base_url: "/api/v1"
  routes: [
    {
      method: "DELETE"
      path: "/orders"
      handler: "createProducts"
      auth: true
    }

    {
      method: "PATCH"
      path: "/comments"
      handler: "createOrders"
      auth: false
    }

    {
      method: "POST"
      path: "/posts/:id"
      handler: "deletePosts"
      auth: false
    }

    {
      method: "PUT"
      path: "/orders"
      handler: "createOrders"
      auth: false
    }

    {
      method: "DELETE"
      path: "/comments"
      handler: "deleteProducts"
      auth: true
    }
  ]
}
