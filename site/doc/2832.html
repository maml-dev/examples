Example 2832

Monitoring Config

Application monitoring and alerting setup.

{
  service: "relationships"
  # Prometheus metrics
  metrics: {
    endpoint: "/metrics"
    interval: 15 # seconds
    labels: {
      env: "prod"
      region: "ap-south"
      team: "jewelry"
    }
  }

  # Alert thresholds
  alerts: {
    cpu_threshold: 74
    memory_threshold: 79
    latency_p99_ms: 1746
    error_rate: 0.05
    notification: {
      channel: "#oncall"
      severity: "info"
    }
  }
}
