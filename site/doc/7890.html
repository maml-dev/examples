Example 7890

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@cheerful-rosemary.com"
  subject: "Dolorum aurum depopulo amitto."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Contra tempora cumque tum theca neque. Tepidus urbanus conculco tenax. Aspicio sortitus quam cursus curtus adeo.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Amet subseco armarium strues color capio. Dapifer uberrime thymum bonus cerno theca tergiversatio. Caritas tempus summopere explicabo cena quo adduco.

Visit: {{url}}
"""
}
