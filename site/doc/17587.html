Example 17587

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@wretched-possession.net"
  subject: "Temperantia totam minima cubicularis."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Venustas error demens advenio voluptatem. Paens canto paens agnitio capto atrox synagoga tollo virtus valeo. Conqueror cultellus officiis tumultus validus vulticulus temporibus.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Vinculum vallum allatus tam corpus conservo socius. Creta creptio excepturi subnecto voluptates terra vero. Aggredior coerceo vestrum.

Visit: {{url}}
"""
}
