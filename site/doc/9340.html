Example 9340

Monitoring Config

Application monitoring and alerting setup.

{
  service: "methodologies"
  # Prometheus metrics
  metrics: {
    endpoint: "/metrics"
    interval: 60 # seconds
    labels: {
      env: "prod"
      region: "us-east"
      team: "baby"
    }
  }

  # Alert thresholds
  alerts: {
    cpu_threshold: 80
    memory_threshold: 95
    latency_p99_ms: 1674
    error_rate: 0.02
    notification: {
      channel: "#oncall"
      severity: "critical"
    }
  }
}
