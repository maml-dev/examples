Docker Compose

Multi-container application definition.

{
  version: "3.8"
  services: {
    app: {
      image: "node:20-slim"
      ports: [
        "3322:3185"
      ]
      environment: {
        NODE_ENV: "production"
        DATABASE_URL: "postgres://db:5432/passport"
      }
      depends_on: [
        "db"
        "redis"
      ]
    }

    db: {
      image: "postgres:16"
      volumes: [
        "db_data:/var/lib/postgresql/data"
      ]
      environment: {
        POSTGRES_DB: "hose"
        POSTGRES_PASSWORD: "PX9qGHNS7HZ6B4C"
      }
    }

    redis: {
      image: "redis:7-alpine"
      ports: [
        "6379:6379"
      ]
    }
  }

  volumes: {
    db_data: {}
  }
}
