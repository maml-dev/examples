Example 1948

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "emacs"
  shortcuts: [
    {
      key: "Ctrl+`"
      action: "debug_run"
      when: "terminal"
    }

    {
      key: "Ctrl+/"
      action: "select_all_occurrences"
      when: "global"
    }

    {
      key: "Alt+Up"
      action: "save"
      when: "editor"
    }
  ]
}
