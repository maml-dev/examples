Example 7467

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Auer - Weimann Dashboard"
  refresh_interval: 30 # seconds
  widgets: [
    {
      type: "chart"
      title: "Active Users"
      source: "/api/metrics/traffic"
      chart_type: "bar"
      period: "30d"
    }

    {
      type: "stat"
      title: "MRR"
      source: "/api/stats/revenue"
      format: "number"
    }

    {
      type: "table"
      title: "Latest Signups"
      source: "/api/recent/signups"
      limit: 7
    }
  ]

  theme: "auto"
}
