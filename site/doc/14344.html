API Routes

REST API route definitions.

{
  base_url: "/api/v2"
  routes: [
    {
      method: "DELETE"
      path: "/products/:id"
      handler: "getOrders"
      auth: true
    }

    {
      method: "PATCH"
      path: "/posts/:id"
      handler: "listOrders"
      auth: true
    }

    {
      method: "PATCH"
      path: "/posts/:id"
      handler: "getPosts"
      auth: true
    }

    {
      method: "DELETE"
      path: "/orders"
      handler: "listUsers"
      auth: true
    }

    {
      method: "PUT"
      path: "/users"
      handler: "listPosts"
      auth: true
    }
  ]
}
