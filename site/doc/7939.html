Example 7939

Terraform Resource

Infrastructure as code resource definition.

{
  provider: "digitalocean"
  region: "asia-southeast1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.130.0.0/16"
      enable_dns: true
      tags: {
        Name: "loyalty-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-10ba9ef7"
      type: "m5.large"
      key_name: "presume-key"
      monitoring: true
      root_volume: {
        size: 50
        type: "io1"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-ssh"
      ingress_port: 80
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
