Example 6660

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "sublime"
  shortcuts: [
    {
      key: "Ctrl+S"
      action: "find_references"
      when: "editor"
    }

    {
      key: "Ctrl+S"
      action: "debug_run"
      when: "global"
    }

    {
      key: "Ctrl+D"
      action: "toggle_comment"
      when: "debug"
    }

    {
      key: "F5"
      action: "toggle_sidebar"
      when: "global"
    }

    {
      key: "Ctrl+D"
      action: "save"
      when: "editor"
    }
  ]
}
