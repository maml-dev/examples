Example 5559

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vim"
  shortcuts: [
    {
      key: "Ctrl+S"
      action: "save"
      when: "editor"
    }

    {
      key: "Ctrl+/"
      action: "debug_run"
      when: "terminal"
    }

    {
      key: "Ctrl+Shift+F"
      action: "select_all_occurrences"
      when: "global"
    }

    {
      key: "Ctrl+`"
      action: "toggle_sidebar"
      when: "terminal"
    }

    {
      key: "Ctrl+D"
      action: "save"
      when: "editor"
    }
  ]
}
