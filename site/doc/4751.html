Example 4751

Keyboard Shortcuts

Editor keyboard shortcut bindings.

{
  editor: "vscode"
  shortcuts: [
    {
      key: "Ctrl+Shift+P"
      action: "find_references"
      when: "editor"
    }

    {
      key: "F5"
      action: "debug_run"
      when: "editor"
    }

    {
      key: "Cmd+K"
      action: "command_palette"
      when: "editor"
    }

    {
      key: "Alt+F7"
      action: "format"
      when: "debug"
    }

    {
      key: "Ctrl+/"
      action: "command_palette"
      when: "editor"
    }

    {
      key: "Ctrl+S"
      action: "toggle_terminal"
      when: "editor"
    }

    {
      key: "Alt+Up"
      action: "format"
      when: "terminal"
    }
  ]
}
