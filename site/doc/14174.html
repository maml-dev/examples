Docker Compose

Multi-container application definition.

{
  version: "3.8"
  services: {
    app: {
      image: "node:18-alpine"
      ports: [
        "3402:3735"
      ]
      environment: {
        NODE_ENV: "production"
        DATABASE_URL: "postgres://db:5432/nucleotidase"
      }
      depends_on: [
        "db"
        "redis"
      ]
    }

    db: {
      image: "postgres:16"
      volumes: [
        "db_data:/var/lib/postgresql/data"
      ]
      environment: {
        POSTGRES_DB: "governance"
        POSTGRES_PASSWORD: "cPsPXMXlzo0Mbqp"
      }
    }

    redis: {
      image: "redis:7-alpine"
      ports: [
        "6379:6379"
      ]
    }
  }

  volumes: {
    db_data: {}
  }
}
