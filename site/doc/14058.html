DNS Records

DNS zone record entries.

{
  zone: "eminent-outlaw.net"
  records: [
    {
      type: "A"
      name: "cdn"
      value: "old-fashioned-commercial.biz"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "*"
      value: "166.133.187.223"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "mail"
      value: "31.114.20.98"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "@"
      value: "v=spf1 include:_spf.crushing-gym.name ~all"
      ttl: 86400
    }
  ]
}
