Example 1034

Monitoring Config

Application monitoring and alerting setup.

{
  service: "ai"
  # Prometheus metrics
  metrics: {
    endpoint: "/metrics"
    interval: 10 # seconds
    labels: {
      env: "staging"
      region: "us-east"
      team: "kids"
    }
  }

  # Alert thresholds
  alerts: {
    cpu_threshold: 77
    memory_threshold: 88
    latency_p99_ms: 1107
    error_rate: 0.02
    notification: {
      channel: "#incidents"
      severity: "warning"
    }
  }
}
