Example 7089

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@posh-nectarine.org"
  subject: "Amplus admitto ultio tracto aufero supra."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Synagoga terreo accedo cito caveo somniculosus complectus cruciamentum aggredior thalassinus. Cohaero succurro beatae caries teneo tergo aveho. Arbitro ipsum delectus.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Desino cohaero assumenda calculus calcar tametsi summa. Decens vaco calamitas umbra cognatus. Reiciendis universe crastinus adficio varius patior demulceo.

Visit: {{url}}
"""
}
