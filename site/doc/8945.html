Example 8945

API Routes

REST API route definitions.

{
  base_url: "/api/v1"
  routes: [
    {
      method: "PATCH"
      path: "/posts/:id"
      handler: "createOrders"
      auth: true
    }

    {
      method: "POST"
      path: "/posts/:id"
      handler: "createOrders"
      auth: false
    }

    {
      method: "PATCH"
      path: "/comments"
      handler: "listOrders"
      auth: false
    }

    {
      method: "PUT"
      path: "/orders"
      handler: "deleteProducts"
      auth: false
    }

    {
      method: "GET"
      path: "/users"
      handler: "updateOrders"
      auth: true
    }
  ]
}
