Terraform Resource

Infrastructure as code resource definition.

{
  provider: "azurerm"
  region: "asia-southeast1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.149.0.0/16"
      enable_dns: true
      tags: {
        Name: "relative-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-bf5ff676"
      type: "m5.large"
      key_name: "brush-key"
      monitoring: true
      root_volume: {
        size: 100
        type: "gp3"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-all"
      ingress_port: 8080
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
