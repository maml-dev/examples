Example 8743

API Routes

REST API route definitions.

{
  base_url: "/api/v2"
  routes: [
    {
      method: "PATCH"
      path: "/products/:id"
      handler: "deleteUsers"
      auth: false
    }

    {
      method: "POST"
      path: "/users"
      handler: "getProducts"
      auth: false
    }

    {
      method: "DELETE"
      path: "/posts/:id"
      handler: "updateProducts"
      auth: true
    }

    {
      method: "POST"
      path: "/orders/:id"
      handler: "getOrders"
      auth: false
    }
  ]
}
