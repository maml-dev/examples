Example 16629

Email Routing

Email routing and filtering rules.

{
  domain: "stingy-fat.biz"
  inbound: {
    mx_records: [
      "mx1.stingy-fat.biz"
      "mx2.stingy-fat.biz"
    ]
    spam_filter: {
      enabled: true
      threshold: 7
      action: "quarantine"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 1024
    }
  }

  outbound: {
    relay: "smtp.stingy-fat.biz"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 354
      per_hour: 3859
    }
  }
}
