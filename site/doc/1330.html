Example 1330

Terraform Resource

Infrastructure as code resource definition.

{
  provider: "aws"
  region: "eu-west-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.229.0.0/16"
      enable_dns: true
      tags: {
        Name: "zebra-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-f51550cf"
      type: "t3.micro"
      key_name: "yin-key"
      monitoring: true
      root_volume: {
        size: 50
        type: "io1"
        encrypted: true
      }
    }

    # Security
    security_group: {
      name: "allow-ssh"
      ingress_port: 443
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
