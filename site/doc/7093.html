Example 7093

HTML Templates

Email HTML templates with raw string content.

# Email template configuration
{
  from: "noreply@selfish-fax.net"
  subject: "Sortitus volo tepidus."
  # HTML email body
  template: """
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Welcome</title>
</head>
<body>
  <h1>Hello, {{name}}!</h1>
  <p>Iste commodo neque cavus beatus apto turbo spectaculum cubitum numquam. Depulso esse exercitationem complectus talio vir delectatio. Laboriosam dedecor tutamen tepesco decor calcar defetiscor capio urbs.</p>
  <a href="{{url}}">Get Started</a>
</body>
</html>
"""

  # Plain text fallback
  plain_text: """
Hello, {{name}}!

Sodalitas laboriosam color ad sui. Virgo basium conventus adhaero textus solus quisquam blandior illum. Cum tredecim crepusculum tricesimus.

Visit: {{url}}
"""
}
