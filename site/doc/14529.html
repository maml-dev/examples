Example 14529

Queue Config

Message queue and worker configuration.

{
  broker: "rabbitmq"
  connection: {
    host: "mq.internal"
    port: 9092
    vhost: "/"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 9
      retry: 4
      timeout: 59148 # ms
    }

    priority: {
      concurrency: 5
      retry: 10
      timeout: 1240 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 4
    queue: "dead-letter"
  }
}
