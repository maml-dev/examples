Example 13664

Dashboard Layout

Dashboard widget layout configuration.

{
  title: "Pacocha, Towne and Schneider Dashboard"
  refresh_interval: 60 # seconds
  widgets: [
    {
      type: "chart"
      title: "Traffic"
      source: "/api/metrics/traffic"
      chart_type: "line"
      period: "90d"
    }

    {
      type: "stat"
      title: "Total Users"
      source: "/api/stats/revenue"
      format: "percent"
    }

    {
      type: "table"
      title: "Recent Orders"
      source: "/api/recent/orders"
      limit: 5
    }
  ]

  theme: "light"
}
