Example 4856

Roles & Permissions

Role-based access control definitions.

{
  roles: [
    {
      name: "billing"
      permissions: [
        "export_data"
        "delete"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "manage_billing"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "read"
        "write"
        "manage_users"
      ]
      max_seats: 12
    }
  ]
}
