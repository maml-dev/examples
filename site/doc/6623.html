Example 6623

Backup Strategy

Backup schedule and retention policy.

{
  source: {
    type: "postgresql"
    host: "primary.db.internal"
    databases: [
      "deduction"
      "cow"
    ]
  }

  # Backup schedule
  schedule: {
    full: "0 2 * * 0" # weekly Sunday 2am
    incremental: "0 2 * * 1-6" # daily except Sunday
    snapshot: "0 */4 * * *" # every 4 hours
  }

  retention: {
    daily: 11
    weekly: 4
    monthly: 11
  }

  destination: {
    type: "gcs"
    path: "/mnt/backups/couch"
    encryption: true
    compression: "zstd"
  }
}
