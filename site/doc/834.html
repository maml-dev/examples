Example 834

Monitoring Config

Application monitoring and alerting setup.

{
  service: "channels"
  # Prometheus metrics
  metrics: {
    endpoint: "/metrics"
    interval: 30 # seconds
    labels: {
      env: "prod"
      region: "ap-south"
      team: "jewelry"
    }
  }

  # Alert thresholds
  alerts: {
    cpu_threshold: 84
    memory_threshold: 90
    latency_p99_ms: 1628
    error_rate: 0.02
    notification: {
      channel: "#ops-alerts"
      severity: "critical"
    }
  }
}
