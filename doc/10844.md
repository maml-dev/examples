# 10844 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "read"
        "view_analytics"
        "manage_billing"
        "delete"
      ]
      max_seats: 37
    }

    {
      name: "editor"
      permissions: [
        "delete"
        "view_analytics"
        "manage_billing"
        "read"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "read"
        "manage_users"
        "write"
        "export_data"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "delete"
        "read"
      ]
      max_seats: 20
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "read"
      ]
      max_seats: 14
    }
  ]
}
```
