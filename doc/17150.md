# 17150 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "read"
        "view_analytics"
        "export_data"
        "manage_users"
        "delete"
      ]
      max_seats: 12
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "read"
        "view_analytics"
        "export_data"
      ]
      max_seats: 27
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "manage_billing"
        "view_analytics"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "read"
        "delete"
        "manage_billing"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "manage_users"
        "view_analytics"
      ]
      max_seats: 4
    }
  ]
}
```
