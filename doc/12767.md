# 12767 - Docker Compose

```maml
{
  version: "3.8"
  services: {
    app: {
      image: "node:18-alpine"

        "3333:3362"
      ]

        "db"
        "redis"
      ]
    }

    db: {

        "6379:6379"
      ]
    }
  }

  volumes: {
    db_data: {}
  }
}
```
