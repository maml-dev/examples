# 14871 - Docker Compose

```maml
{
  version: "3.8"
  services: {
    app: {
      image: "node:20-slim"

        "3953:3983"

        "db"
        "redis"
      ]
    }

        "6379:6379"
      ]
    }
  }

  volumes: {
    db_data: {}
  }
}
```
