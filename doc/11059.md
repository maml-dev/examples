# 11059 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "manage_billing"
        "write"
        "view_analytics"
      ]
      max_seats: 2
    }

    {
      name: "viewer"
      permissions: [
        "write"
        "delete"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "read"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "write"
        "delete"
      ]
      max_seats: 41
    }

    {
      name: "viewer"
      permissions: [
        "read"
        "manage_users"
        "write"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "view_analytics"
        "manage_users"
        "delete"
      ]
      max_seats: null
    }
  ]
}
```
