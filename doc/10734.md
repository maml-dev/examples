# 10734 - API Gateway

API gateway routing and middleware configuration. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  gateway: {
    host: "api.drab-bench.net"
    port: 443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "15m"
    max_requests: 3642
    by: "ip"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [
      "https://exhausted-shadowbox.net"
      "https://app.puzzled-pendant.name"

      "GET"

      "PUT"
      "DELETE"
    ]
    max_age: 7200
  }

  cache: {
    enabled: false
    ttl: 1953
    store: "redis"
  }
}
```
