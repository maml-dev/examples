# 14154 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "export_data"
        "delete"
        "read"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "manage_users"
        "manage_billing"
        "write"
        "read"
      ]
      max_seats: 44
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "view_analytics"
        "manage_billing"
        "read"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "read"
        "manage_users"
        "manage_billing"
      ]
      max_seats: 6
    }

    {
      name: "billing"
      permissions: [
        "delete"
        "export_data"
      ]
      max_seats: null
    }
  ]
}
```
