# 5358 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "velvety-flat.info"
  records: [
    {
      type: "A"
      name: "*"
      value: "17.64.246.115"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "api"
      value: "wide-brush.org"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "mail"
      value: "124.244.51.6"
      ttl: 86400
    }

    {
      type: "A"
      name: "staging"
      value: "v=spf1 include:_spf.puny-guide.org ~all"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "staging"
      value: "slow-bowler.org"
      ttl: 86400
    }

    {
      type: "NS"
      name: "www"
      value: "v=spf1 include:_spf.total-husband.org ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "@"
      value: "optimistic-tarragon.com"
      ttl: 86400
    }

    {
      type: "MX"
      name: "*"
      value: "v=spf1 include:_spf.squeaky-promise.name ~all"
      ttl: 300
    }
  ]
}
```
