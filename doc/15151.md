# 15151 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "export_data"
        "manage_users"
        "write"
        "manage_billing"
      ]
      max_seats: 42
    }

    {
      name: "support"
      permissions: [
        "manage_billing"
        "write"
        "export_data"
        "view_analytics"
        "read"
      ]
      max_seats: 49
    }

    {
      name: "billing"
      permissions: [
        "write"
        "read"
        "manage_billing"
      ]
      max_seats: 31
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "read"
      ]
      max_seats: 37
    }

    {
      name: "viewer"
      permissions: [
        "manage_billing"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "write"
        "read"
        "delete"
      ]
      max_seats: null
    }
  ]
}
```
