# 8746 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "manage_billing"
        "write"
        "delete"
        "export_data"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "delete"
        "manage_users"
        "view_analytics"
        "write"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "manage_billing"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "read"
        "manage_users"
        "write"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "view_analytics"
      ]
      max_seats: 48
    }
  ]
}
```
