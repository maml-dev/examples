# 9641 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "delete"
        "write"
        "manage_billing"
      ]
      max_seats: 47
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "export_data"
        "read"
        "write"
      ]
      max_seats: 8
    }

    {
      name: "support"
      permissions: [
        "read"
        "manage_billing"
        "write"
        "export_data"
        "delete"
      ]
      max_seats: 6
    }

    {
      name: "admin"
      permissions: [
        "write"
        "read"
        "manage_billing"
        "view_analytics"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "read"
        "manage_billing"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "manage_users"
      ]
      max_seats: null
    }
  ]
}
```
