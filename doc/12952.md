# 12952 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "manage_billing"
        "manage_users"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "export_data"
        "write"
        "read"
        "manage_billing"
      ]
      max_seats: 45
    }

    {
      name: "viewer"
      permissions: [
        "read"
        "write"
        "manage_users"
        "manage_billing"
      ]
      max_seats: 8
    }

    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "read"
        "manage_billing"
        "export_data"
        "write"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "read"
        "manage_billing"
        "write"
        "export_data"
        "delete"
      ]
      max_seats: 5
    }

    {
      name: "editor"
      permissions: [
        "write"
        "view_analytics"
        "export_data"
      ]
      max_seats: null
    }
  ]
}
```
