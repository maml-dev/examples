# 6146 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "delete"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "read"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "write"
        "read"
        "export_data"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "write"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "write"
        "manage_users"
        "manage_billing"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "manage_users"
        "read"
      ]
      max_seats: 31
    }
  ]
}
```
