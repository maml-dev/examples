# 6923 - Backup Strategy

Backup schedule and retention policy. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  source: {
    type: "filesystem"
    host: "localhost"
    databases: [
      "derby"
      "disadvantage"
    ]
  }

  # Backup schedule
  schedule: {
    full: "0 2 * * 0" # weekly Sunday 2am
    incremental: "0 2 * * 1-6" # daily except Sunday
    snapshot: "0 */4 * * *" # every 4 hours
  }

  retention: {
    daily: 11
    weekly: 11
    monthly: 23
  }

  destination: {
    type: "local"
    path: "/mnt/backups/awareness"
    encryption: true
    compression: "gzip"
  }
}
```
