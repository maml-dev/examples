# 12548 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "slight-vol.net"
  records: [
    {
      type: "CNAME"
      name: "api"
      value: "alive-pigpen.info"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "www"
      value: "shrill-ostrich.net"
      ttl: 300
    }

    {
      type: "MX"
      name: "mail"
      value: "pushy-wriggler.net"
      ttl: 300
    }

    {
      type: "A"
      name: "cdn"
      value: "v=spf1 include:_spf.husky-suspension.net ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "api"
      value: "junior-lamp.com"
      ttl: 300
    }

    {
      type: "TXT"
      name: "*"
      value: "back-godparent.info"
      ttl: 3600
    }

    {
      type: "NS"
      name: "www"
      value: "grimy-premise.net"
      ttl: 300
    }

    {
      type: "A"
      name: "@"
      value: "bright-tributary.info"
      ttl: 300
    }
  ]
}
```
