# 15944 - DNS Records

  records: [

      type: "MX"
      name: "staging"
      value: "swift-ice-cream.net"
      ttl: 86400

      type: "TXT"
      name: "api"
      value: "45.185.173.30"
      ttl: 300
    }

    {
      type: "AAAA"

      ttl: 86400

      name: "cdn"
      value: "55.239.43.12"
      ttl: 3600
    }
