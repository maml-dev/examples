# 15053 - Cron Jobs

Scheduled task definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  timezone: "Europe/London"
  jobs: [
    {
      name: "send-reports"
      schedule: "*/15 * * * *"
      command: "node scripts/cleanup.js"
      enabled: false
    }

    {
      name: "send-reports"
      schedule: "30 6 * * 1-5"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "cleanup-temp"
      schedule: "0 0 * * *"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }

    {
      name: "sync-data"
      schedule: "30 6 * * 1-5"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "send-reports"
      schedule: "0 2 * * 0"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }

    {
      name: "backup-db"
      schedule: "30 6 * * 1-5"
      command: "node scripts/cleanup.js"
      enabled: true
    }
  ]
}
```
