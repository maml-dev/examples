# 5324 - Terraform Resource

Infrastructure as code resource definition. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  provider: "digitalocean"
  region: "eu-west-1"
  resources: {
    # Network
    vpc: {
      cidr_block: "10.221.0.0/16"
      enable_dns: true
      tags: {
        Name: "detective-vpc"
        Environment: "staging"
      }
    }

    # Compute
    instance: {
      ami: "ami-fbeecfcd"
      type: "t3.small"
      key_name: "pharmacopoeia-key"
      monitoring: true
      root_volume: {
        size: 100
        type: "gp3"
        encrypted: true

    }

    # Security
    security_group: {
      name: "allow-all"
      ingress_port: 8080
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
```
