# 5857 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "jagged-king.net"
  records: [
    {
      type: "A"
      name: "staging"
      value: "43.237.74.248"
      ttl: 300
    }

    {
      type: "TXT"
      name: "mail"
      value: "172.241.204.100"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "www"
      value: "v=spf1 include:_spf.unkempt-puritan.org ~all"
      ttl: 300
    }

    {
      type: "NS"
      name: "www"
      value: "important-whack.info"
      ttl: 300
    }

    {
      type: "A"
      name: "staging"
      value: "v=spf1 include:_spf.wilted-guacamole.net ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "staging"
      value: "v=spf1 include:_spf.nimble-chiffonier.name ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "mail"
      value: "fair-brush.info"
      ttl: 86400
    }
  ]
}
```
