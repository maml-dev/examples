# 12249 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "fruitful-creator.com"
  records: [
    {
      type: "NS"
      name: "mail"
      value: "189.130.247.162"
      ttl: 300
    }

    {
      type: "TXT"
      name: "mail"
      value: "nutritious-editor.biz"
      ttl: 3600
    }

    {
      type: "MX"
      name: "api"
      value: "lumpy-zebra.name"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "api"
      value: "unwieldy-accompanist.org"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "api"
      value: "18.87.176.125"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "@"
      value: "30.149.63.249"
      ttl: 86400
    }

    {
      type: "NS"
      name: "staging"
      value: "classic-suitcase.name"
      ttl: 86400
    }

    {
      type: "MX"
      name: "mail"
      value: "v=spf1 include:_spf.petty-bathhouse.net ~all"
      ttl: 86400
    }
  ]
}
```
