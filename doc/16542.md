# 16542 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "manage_users"
        "read"
        "delete"
        "export_data"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "delete"
        "write"
      ]
      max_seats: 12
    }

    {
      name: "admin"
      permissions: [
        "write"
        "view_analytics"
        "manage_users"
        "export_data"
        "delete"
      ]
      max_seats: 1
    }

    {
      name: "billing"
      permissions: [
        "manage_billing"
        "export_data"
        "write"
        "manage_users"
      ]
      max_seats: 6
    }

    {
      name: "support"
      permissions: [
        "delete"
        "view_analytics"
        "manage_billing"
        "export_data"
      ]
      max_seats: null
    }
  ]
}
```
