# 9057 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "manage_users"
        "write"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "delete"
        "write"
      ]
      max_seats: 14
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "manage_billing"
        "export_data"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "export_data"
        "write"
      ]
      max_seats: 19
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "delete"
        "write"
        "view_analytics"
        "read"
      ]
      max_seats: 31
    }

    {
      name: "viewer"
      permissions: [
        "write"
        "view_analytics"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 9
    }
  ]
}
```
