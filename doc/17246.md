# 17246 - Cron Jobs

Scheduled task definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  timezone: "UTC"
  jobs: [
    {
      name: "send-reports"
      schedule: "30 6 * * 1-5"
      command: "python manage.py report"
      enabled: true
    }

    {
      name: "rotate-logs"
      schedule: "*/15 * * * *"
      command: "node scripts/cleanup.js"
      enabled: false
    }

    {
      name: "rotate-logs"
      schedule: "0 0 * * *"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "cleanup-temp"
      schedule: "30 6 * * 1-5"
      command: "./bin/sync --all"
      enabled: false
    }

    {
      name: "send-reports"
      schedule: "30 6 * * 1-5"
      command: "python manage.py report"
      enabled: true
    }

    {
      name: "send-reports"
      schedule: "0 2 * * 0"
      command: "./bin/sync --all"
      enabled: false
    }
  ]
}
```
