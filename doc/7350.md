# 7350 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "monstrous-privilege.org"
  records: [
    {
      type: "A"
      name: "staging"
      value: "tight-appliance.com"
      ttl: 300
    }

    {
      type: "A"
      name: "mail"
      value: "v=spf1 include:_spf.granular-recovery.info ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "api"
      value: "v=spf1 include:_spf.ornate-juggernaut.biz ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "mail"
      value: "likely-subexpression.biz"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "mail"
      value: "v=spf1 include:_spf.valuable-crocodile.name ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "cdn"
      value: "v=spf1 include:_spf.sparse-armoire.net ~all"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "staging"
      value: "v=spf1 include:_spf.faint-prohibition.org ~all"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "api"
      value: "well-documented-consistency.info"
      ttl: 3600
    }
  ]
}
```
