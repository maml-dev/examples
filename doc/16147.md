# 16147 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "measly-thigh.org"
  records: [
    {
      type: "CNAME"
      name: "cdn"
      value: "v=spf1 include:_spf.measly-moment.com ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "staging"
      value: "v=spf1 include:_spf.silky-rim.com ~all"
      ttl: 3600
    }

    {
      type: "NS"
      name: "@"
      value: "subtle-language.name"
      ttl: 300
    }

    {
      type: "MX"
      name: "www"
      value: "55.92.5.178"
      ttl: 300
    }

    {
      type: "TXT"
      name: "mail"
      value: "worthless-draft.biz"
      ttl: 3600
    }

    {
      type: "NS"
      name: "*"
      value: "v=spf1 include:_spf.arid-litter.info ~all"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "*"

    }

    {

  ]
}
```
