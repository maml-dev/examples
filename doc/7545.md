# 7545 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "damaged-precedent.name"
  records: [
    {
      type: "NS"
      name: "www"
      value: "unused-postbox.com"
      ttl: 3600
    }

    {
      type: "MX"
      name: "www"
      value: "v=spf1 include:_spf.puny-catalyst.com ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "*"
      value: "170.77.69.73"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "staging"
      value: "productive-lava.name"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "staging"
      value: "v=spf1 include:_spf.probable-cap.biz ~all"
      ttl: 300
    }

    {
      type: "TXT"
      name: "staging"
      value: "blond-mantua.biz"
      ttl: 86400
    }

    {
      type: "A"
      name: "www"
      value: "sparkling-gazebo.biz"
      ttl: 86400
    }
  ]
}
```
