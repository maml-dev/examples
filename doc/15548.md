# 15548 - API Routes

REST API route definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  base_url: "/api/v2"
  routes: [
    {
      method: "PUT"
      path: "/products/:id"
      handler: "getUsers"
      auth: false
    }

    {
      method: "PATCH"
      path: "/products/:id"
      handler: "createPosts"
      auth: false
    }

    {
      method: "PUT"
      path: "/users"
      handler: "createPosts"
      auth: false
    }

    {
      method: "GET"
      path: "/posts/:id"
      handler: "listOrders"
      auth: false
    }

    {
      method: "GET"
      path: "/comments"
      handler: "updateUsers"
      auth: false
    }

    {
      method: "PUT"
      path: "/posts"
      handler: "createProducts"
      auth: false
    }

    {
      method: "DELETE"
      path: "/users"
      handler: "updateProducts"
      auth: false
    }
  ]
}
```
