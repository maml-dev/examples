# 11041 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "super-coliseum.org"
  records: [
    {
      type: "MX"
      name: "mail"
      value: "242.8.129.164"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "www"
      value: "v=spf1 include:_spf.fair-heating.org ~all"
      ttl: 300
    }

    {
      type: "TXT"
      name: "cdn"
      value: "bogus-babushka.info"
      ttl: 300
    }

    {
      type: "TXT"
      name: "www"
      value: "v=spf1 include:_spf.vivid-cook.com ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "mail"
      value: "responsible-minor.net"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "*"
      value: "homely-co-producer.com"
      ttl: 86400
    }

    {
      type: "A"
      name: "api"
      value: "200.46.24.104"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "api"
      value: "193.113.44.125"
      ttl: 3600
    }
  ]
}
```
