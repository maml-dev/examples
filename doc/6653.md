# 6653 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "editor"
      permissions: [
        "export_data"
        "delete"
        "read"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "manage_users"
        "delete"
        "export_data"
        "read"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "view_analytics"
        "delete"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "export_data"
        "write"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "manage_users"
        "export_data"
        "write"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "read"
        "manage_billing"
        "delete"
      ]
      max_seats: null
    }
  ]
}
```
