# 13142 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "manage_users"
        "delete"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "view_analytics"
        "manage_users"
        "write"
      ]
      max_seats: 12
    }

    {
      name: "billing"
      permissions: [
        "manage_users"
        "export_data"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "write"
        "export_data"
      ]
      max_seats: 47
    }

    {
      name: "billing"
      permissions: [
        "manage_users"
        "write"
        "read"
        "export_data"
        "view_analytics"
      ]
      max_seats: 9
    }

    {
      name: "billing"
      permissions: [
        "manage_billing"
        "export_data"
        "read"
        "manage_users"
        "view_analytics"
      ]
      max_seats: null
    }
  ]
}
```
