# 7441 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "editor"
      permissions: [
        "delete"
        "export_data"
        "manage_users"
        "read"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "write"
        "read"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "view_analytics"
        "manage_users"
        "export_data"
      ]
      max_seats: 38
    }

    {
      name: "support"
      permissions: [
        "delete"
        "export_data"
        "manage_billing"
        "write"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "delete"
        "manage_billing"
        "view_analytics"
        "write"
        "manage_users"
      ]
      max_seats: 42
    }

    {
      name: "editor"
      permissions: [
        "delete"
        "manage_users"
        "read"
      ]
      max_seats: null
    }
  ]
}
```
