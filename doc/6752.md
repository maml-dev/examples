# 6752 - Cron Jobs

Scheduled task definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  timezone: "America/New_York"
  jobs: [
    {
      name: "backup-db"
      schedule: "30 6 * * 1-5"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "backup-db"
      schedule: "*/15 * * * *"
      command: "node scripts/cleanup.js"
      enabled: false
    }

    {
      name: "rotate-logs"
      schedule: "0 * * * *"
      command: "node scripts/cleanup.js"
      enabled: true
    }

    {
      name: "send-reports"
      schedule: "30 6 * * 1-5"
      command: "./bin/sync --all"
      enabled: false
    }

    {
      name: "cleanup-temp"
      schedule: "30 6 * * 1-5"
      command: "python manage.py report"
      enabled: false
    }

    {
      name: "refresh-cache"
      schedule: "0 * * * *"
      command: "node scripts/cleanup.js"
      enabled: false
    }
  ]
}
```
