# 15957 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "amused-vein.com"
  records: [
    {
      type: "MX"
      name: "www"
      value: "155.133.88.108"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "*"
      value: "206.26.196.156"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "@"
      value: "giving-draft.info"
      ttl: 3600
    }

    {
      type: "NS"
      name: "staging"
      value: "v=spf1 include:_spf.mild-membership.net ~all"
      ttl: 300
    }

    {
      type: "TXT"
      name: "www"
      value: "strong-design.com"
      ttl: 86400
    }

    {
      type: "A"
      name: "cdn"
      value: "26.7.143.54"
      ttl: 86400
    }

    {
      type: "MX"
      name: "mail"
      value: "v=spf1 include:_spf.new-tenement.net ~all"
      ttl: 300
    }
  ]
}
```
