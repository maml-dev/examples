# 12421 - Queue Config

Message queue and worker configuration. This is an example of a [MAML](https://maml.dev) document.

{
  broker: "redis"
  connection: {
    host: "localhost"
    port: 6379
    vhost: "/staging"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 1
      retry: 3
      timeout: 44619 # ms
    }

    priority: {
      concurrency: 20
      retry: 6
      timeout: 1373 # ms
    }
  }

  dead_letter: {
    enabled: true
    max_retries: 8
    queue: "dead-letter"
  }
}
