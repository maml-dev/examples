# 17141 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "manage_users"
        "view_analytics"
        "delete"
        "write"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "write"
        "export_data"
        "delete"
      ]
      max_seats: 24
    }

    {
      name: "admin"
      permissions: [
        "manage_users"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "export_data"
        "write"
      ]
      max_seats: 45
    }

    {
      name: "support"
      permissions: [
        "delete"
        "read"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 19
    }

    {
      name: "billing"
      permissions: [
        "write"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 3
    }
  ]
}
```
