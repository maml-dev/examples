# 10942 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "editor"
      permissions: [
        "export_data"
        "write"
        "delete"
        "view_analytics"
        "manage_billing"
      ]
      max_seats: 22
    }

    {
      name: "support"
      permissions: [
        "manage_users"
        "delete"
        "manage_billing"
      ]
      max_seats: 29
    }

    {
      name: "support"
      permissions: [
        "write"
        "manage_users"
      ]
      max_seats: 31
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "read"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "manage_billing"
        "write"
        "read"
        "export_data"
      ]
      max_seats: 40
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "delete"
      ]
      max_seats: 22
    }
  ]
}
```
