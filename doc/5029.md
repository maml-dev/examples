# 5029 - Queue Config

{
  broker: "redis"
  connection: {
    host: "mq.internal"
    port: 6379
    vhost: "/production"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 4
      retry: 4

      concurrency: 16
      retry: 4
      timeout: 5979 # ms

  }
}
