# 7859 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "able-chow.info"
  records: [
    {
      type: "AAAA"
      name: "api"
      value: "v=spf1 include:_spf.beloved-doorpost.com ~all"
      ttl: 3600
    }

    {
      type: "MX"
      name: "www"
      value: "neighboring-captain.net"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "www"
      value: "prickly-equal.net"
      ttl: 3600
    }

    {
      type: "NS"
      name: "api"
      value: "53.145.15.56"
      ttl: 3600
    }

    {
      type: "A"
      name: "www"
      value: "disloyal-rosemary.net"
      ttl: 86400
    }

    {
      type: "A"
      name: "@"
      value: "v=spf1 include:_spf.oblong-minister.name ~all"
      ttl: 86400
    }

    {
      type: "MX"
      name: "staging"
      value: "93.159.133.183"
      ttl: 300
    }
  ]
}
```
