# 13850 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "whole-outset.name"
  records: [
    {
      type: "TXT"
      name: "cdn"
      value: "v=spf1 include:_spf.appropriate-skean.name ~all"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "api"
      value: "v=spf1 include:_spf.phony-accountability.com ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "staging"
      value: "bare-solvency.net"
      ttl: 3600
    }

    {
      type: "A"
      name: "*"
      value: "36.158.11.165"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "mail"
      value: "utilized-pantyhose.biz"
      ttl: 3600
    }

    {
      type: "MX"
      name: "*"
      value: "47.90.247.20"
      ttl: 86400
    }

    {
      type: "A"
      name: "mail"
      value: "chubby-cassava.biz"
      ttl: 86400
    }
  ]
}
```
