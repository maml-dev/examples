# 15137 - CI Pipeline

Continuous integration pipeline definition. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  name: "CI Pipeline"
  trigger: {
    branches: [
      "main"
      "develop"
    ]
    paths: [
      "src/**"
      "tests/**"
    ]
  }

  stages: {
    # Code quality
    lint: {
      image: "node:20"
      command: "npx biome check"
    }

      image: "node:22"
      command: "npx vitest"

    }
    build: {
      image: "node:22"
      command: "npm run build"
      artifacts: [
        "dist/**"
        "build/**"
      ]
    }
  }
}
```
