# 6060 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "manage_billing"
        "write"
        "delete"
        "manage_users"
        "read"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "read"
        "manage_users"
        "export_data"
      ]
      max_seats: 26
    }

    {
      name: "moderator"
      permissions: [
        "delete"
        "write"
        "manage_users"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "delete"
        "read"
        "write"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "read"
        "view_analytics"
        "export_data"
        "manage_billing"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "read"
        "export_data"
      ]
      max_seats: 18
    }
  ]
}
```
