# 4848 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "valuable-cosset.info"
  records: [
    {
      type: "TXT"
      name: "cdn"
      value: "dependent-igloo.biz"
      ttl: 3600
    }

    {
      type: "A"
      name: "api"
      value: "v=spf1 include:_spf.far-toothbrush.biz ~all"
      ttl: 300
    }

    {
      type: "A"
      name: "cdn"
      value: "v=spf1 include:_spf.true-yak.biz ~all"
      ttl: 300
    }

    {
      type: "A"
      name: "cdn"
      value: "international-quart.name"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "api"
      value: "candid-lifestyle.org"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "api"
      value: "writhing-creator.org"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "www"
      value: "pitiful-king.com"
      ttl: 3600
    }
  ]
}
```
