# 7745 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "billing"
      permissions: [
        "export_data"
        "view_analytics"
        "manage_billing"
        "write"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "read"
        "delete"
        "view_analytics"
      ]
      max_seats: 35
    }

    {
      name: "billing"
      permissions: [
        "manage_billing"
        "delete"
      ]
      max_seats: 8
    }

    {
      name: "billing"
      permissions: [
        "delete"
        "manage_users"
        "view_analytics"
        "read"
        "write"
      ]
      max_seats: 49
    }

    {
      name: "billing"
      permissions: [
        "write"
        "export_data"
        "manage_users"
        "delete"
        "read"
      ]
      max_seats: 26
    }
  ]
}
```
