# 13045 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "delete"
        "read"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "view_analytics"
        "write"
        "manage_billing"
        "export_data"
      ]
      max_seats: 26
    }

    {
      name: "viewer"
      permissions: [
        "delete"
        "write"
        "export_data"
      ]
      max_seats: 43
    }

    {
      name: "admin"
      permissions: [
        "export_data"
        "write"
        "view_analytics"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "manage_billing"
        "view_analytics"
        "manage_users"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "manage_users"
        "write"
      ]
      max_seats: 44
    }
  ]
}
```
