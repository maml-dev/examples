# 11859 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "delete"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "write"
        "delete"
        "manage_billing"
        "view_analytics"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "write"
        "delete"
        "view_analytics"
        "manage_billing"
      ]
      max_seats: 6
    }

    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "export_data"
        "read"
      ]
      max_seats: 46
    }

    {
      name: "viewer"
      permissions: [
        "read"
        "view_analytics"
        "export_data"
      ]
      max_seats: 22
    }

    {
      name: "editor"
      permissions: [
        "delete"
        "manage_users"
        "export_data"
        "read"
      ]
      max_seats: 28
    }
  ]
}
```
