# 12442 - DNS Records

```maml
{
  zone: "sarcastic-ribbon.com"
  records: [
    {
      type: "MX"
      name: "*"
      value: "helpless-forage.name"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "@"
      value: "109.161.96.243"
      ttl: 300
    }

    {
      type: "TXT"
      name: "staging"
      value: "250.72.87.94"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "@"
      value: "considerate-allocation.net"
      ttl: 300
    }

    {
      type: "MX"
      name: "api"
      value: "v=spf1 include:_spf.quick-witted-representation.info ~all"

      ttl: 300
    }

    {
      type: "A"
      name: "mail"
      value: "175.49.59.49"
      ttl: 86400
    }
  ]
}
```
