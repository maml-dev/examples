# 10558 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "editor"
      permissions: [
        "manage_users"
        "read"
        "export_data"
        "manage_billing"
      ]
      max_seats: 48
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "view_analytics"
        "write"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "view_analytics"
        "read"
        "export_data"
        "manage_billing"
        "write"
      ]
      max_seats: 42
    }

    {
      name: "support"
      permissions: [
        "manage_billing"
        "write"
        "export_data"
        "view_analytics"
      ]
      max_seats: 15
    }

    {
      name: "admin"
      permissions: [
        "delete"
        "manage_billing"
        "export_data"
        "write"
        "view_analytics"
      ]
      max_seats: null
    }
  ]
}
```
