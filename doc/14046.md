# 14046 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "read"
        "write"
        "view_analytics"
        "export_data"
        "manage_billing"
      ]
      max_seats: 45
    }

    {
      name: "moderator"
      permissions: [
        "read"
        "manage_users"
        "delete"
        "view_analytics"
        "write"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "manage_billing"
        "manage_users"
        "read"
        "write"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "write"
        "export_data"
      ]
      max_seats: 38
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "manage_users"
        "write"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "delete"
        "manage_billing"
        "view_analytics"
        "write"
        "read"
      ]
      max_seats: 2
    }
  ]
}
```
