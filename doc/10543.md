# 10543 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "billing"
      permissions: [
        "read"
        "write"
        "manage_billing"
        "view_analytics"
        "delete"
      ]
      max_seats: 28
    }

    {
      name: "editor"
      permissions: [
        "read"
        "write"
        "export_data"
      ]
      max_seats: 44
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "delete"
      ]
      max_seats: 25
    }

    {
      name: "admin"
      permissions: [
        "export_data"
        "manage_users"
        "view_analytics"
        "delete"
        "read"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "write"
        "manage_billing"
        "read"
      ]
      max_seats: 20
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "delete"
        "write"
        "export_data"
        "read"
      ]
      max_seats: 21
    }
  ]
}
```
