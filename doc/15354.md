# 15354 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "delete"
        "read"
        "write"
        "manage_billing"
        "export_data"
      ]
      max_seats: 3
    }

    {
      name: "billing"
      permissions: [
        "manage_users"
        "read"
        "export_data"
        "write"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "delete"
        "read"
      ]
      max_seats: 8
    }

    {
      name: "support"
      permissions: [
        "delete"
        "write"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "write"
        "delete"
        "read"
      ]
      max_seats: 45
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "view_analytics"
      ]
      max_seats: 42
    }
  ]
}
```
