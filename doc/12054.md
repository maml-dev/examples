# 12054 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "export_data"
        "manage_users"
      ]
      max_seats: 8
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "manage_billing"
        "read"
        "view_analytics"
      ]
      max_seats: 45
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "write"
        "delete"
        "read"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "manage_billing"
        "write"
        "delete"
        "view_analytics"
        "read"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "delete"
        "export_data"
      ]
      max_seats: 26
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "delete"
        "manage_users"
      ]
      max_seats: null
    }
  ]
}
```
