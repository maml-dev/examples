# 12553 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "hot-stock.com"
  records: [
    {
      type: "NS"
      name: "@"
      value: "108.120.207.10"
      ttl: 300
    }

    {
      type: "A"
      name: "staging"
      value: "250.178.122.86"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "*"
      value: "lazy-window.biz"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "api"
      value: "questionable-hovercraft.biz"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "staging"
      value: "pointless-feather.org"
      ttl: 300
    }

    {
      type: "A"
      name: "mail"
      value: "80.151.250.100"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "staging"
      value: "bruised-pick.name"
      ttl: 86400
    }

    {
      type: "MX"
      name: "staging"
      value: "v=spf1 include:_spf.blue-object.com ~all"
      ttl: 86400
    }
  ]
}
```
