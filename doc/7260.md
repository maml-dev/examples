# 7260 - API Routes

REST API route definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  base_url: "/api/v2"
  routes: [
    {
      method: "POST"
      path: "/products/:id"
      handler: "deleteUsers"
      auth: true
    }

    {
      method: "PATCH"
      path: "/orders"
      handler: "deleteProducts"
      auth: true
    }

    {
      method: "DELETE"
      path: "/comments"
      handler: "updateProducts"
      auth: true
    }

    {
      method: "PATCH"
      path: "/orders/:id"
      handler: "createUsers"
      auth: true
    }

    {
      method: "DELETE"
      path: "/products/:id"
      handler: "deletePosts"
      auth: true
    }

    {
      method: "PUT"
      path: "/posts"
      handler: "deleteProducts"
      auth: false
    }

    {
      method: "PATCH"
      path: "/products/:id"
      handler: "createPosts"
      auth: false
    }
  ]
}
```
