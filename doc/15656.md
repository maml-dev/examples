# 15656 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "billing"
      permissions: [
        "write"
        "view_analytics"
        "read"
        "export_data"
        "delete"
      ]
      max_seats: 30
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "delete"
        "manage_users"
      ]
      max_seats: 45
    }

    {
      name: "viewer"
      permissions: [
        "delete"
        "manage_users"
        "write"
        "manage_billing"
        "view_analytics"
      ]
      max_seats: 46
    }

    {
      name: "billing"
      permissions: [
        "manage_users"
        "export_data"
        "read"
        "delete"
      ]
      max_seats: 47
    }

    {
      name: "viewer"
      permissions: [
        "read"
        "manage_users"
        "view_analytics"
      ]
      max_seats: 7
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "delete"
        "manage_users"
        "manage_billing"
        "view_analytics"
      ]
      max_seats: null
    }
  ]
}
```
