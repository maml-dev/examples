```maml
  name: "handsome Theme"
  # Color palette
  colors: {
    primary: "#66ee13"
    secondary: "#2b0fc8"
    accent: "#aeec18"

    font_family: "system-ui, sans-serif"

      4

      24
      32
      48
    ]
  }
  border_radius: 12
}
```
