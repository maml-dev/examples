# 6342 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "arid-brook.com"
  records: [
    {
      type: "MX"
      name: "staging"
      value: "40.48.37.183"
      ttl: 300
    }

    {
      type: "TXT"
      name: "*"
      value: "41.232.177.173"
      ttl: 300
    }

    {
      type: "A"
      name: "@"
      value: "v=spf1 include:_spf.faraway-barracks.name ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "cdn"
      value: "5.53.101.165"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "mail"
      value: "elegant-fork.info"
      ttl: 3600
    }

    {
      type: "NS"
      name: "www"
      value: "216.50.199.142"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "mail"
      value: "icy-director.org"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "www"
      value: "65.247.229.119"
      ttl: 86400
    }
  ]
}
```
