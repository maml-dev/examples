# 11543 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "punctual-paintwork.com"
  records: [
    {
      type: "TXT"
      name: "staging"
      value: "agitated-marathon.biz"
      ttl: 300
    }

    {
      type: "MX"
      name: "mail"
      value: "v=spf1 include:_spf.unhealthy-transparency.name ~all"
      ttl: 3600
    }

    {
      type: "MX"
      name: "@"
      value: "faraway-hygienic.com"
      ttl: 3600
    }

    {
      type: "A"
      name: "www"
      value: "158.120.175.59"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "@"
      value: "172.236.7.171"
      ttl: 300
    }

    {
      type: "MX"
      name: "mail"
      value: "243.242.190.45"
      ttl: 3600
    }

    {
      type: "MX"
      name: "*"
      value: "154.213.69.34"
      ttl: 3600
    }
  ]
}
```
