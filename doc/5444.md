# 5444 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "billing"
      permissions: [
        "delete"
        "view_analytics"
        "manage_users"
        "write"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "read"
        "write"
        "export_data"
      ]
      max_seats: 46
    }

    {
      name: "editor"
      permissions: [
        "manage_billing"
        "manage_users"
      ]
      max_seats: 15
    }

    {
      name: "editor"
      permissions: [
        "view_analytics"
        "manage_billing"
        "manage_users"
        "read"
        "delete"
      ]
      max_seats: 7
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "write"
        "read"
        "delete"
        "manage_users"
      ]
      max_seats: 40
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "view_analytics"
        "read"
        "delete"
      ]
      max_seats: null
    }
  ]
}
```
