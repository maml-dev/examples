# 10260 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "manage_users"
        "write"
        "manage_billing"
        "delete"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "write"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 41
    }

    {
      name: "support"
      permissions: [
        "view_analytics"
        "export_data"
        "delete"
        "read"
      ]
      max_seats: 35
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "view_analytics"
        "write"
        "read"
        "export_data"
      ]
      max_seats: 25
    }

    {
      name: "billing"
      permissions: [
        "view_analytics"
        "export_data"
        "read"
        "write"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "delete"
        "read"
        "write"
        "manage_users"
        "export_data"
      ]
      max_seats: null
    }
  ]
}
```
