# 13241 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "exotic-programme.info"
  records: [
    {
      type: "NS"
      name: "*"
      value: "183.17.20.103"
      ttl: 86400
    }

    {
      type: "NS"
      name: "mail"
      value: "separate-postbox.net"
      ttl: 3600
    }

    {
      type: "MX"
      name: "cdn"
      value: "v=spf1 include:_spf.punctual-roundabout.com ~all"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "*"
      value: "colorful-electronics.biz"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "mail"
      value: "179.48.232.94"
      ttl: 86400
    }

    {
      type: "MX"
      name: "api"
      value: "v=spf1 include:_spf.nautical-casket.info ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "staging"
      value: "purple-receptor.name"
      ttl: 86400
    }
  ]
}
```
