# 10441 - API Routes

REST API route definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  base_url: "/api/v2"
  routes: [
    {
      method: "PUT"
      path: "/orders/:id"
      handler: "listProducts"
      auth: false
    }

    {
      method: "DELETE"
      path: "/products"
      handler: "listUsers"
      auth: true
    }

    {
      method: "GET"
      path: "/products"
      handler: "updateOrders"
      auth: true
    }

    {
      method: "PUT"
      path: "/comments"
      handler: "deleteUsers"
      auth: true
    }

    {
      method: "PUT"
      path: "/posts/:id"
      handler: "deleteProducts"
      auth: true
    }

    {
      method: "PATCH"
      path: "/products"
      handler: "getProducts"
      auth: true
    }

    {
      method: "GET"
      path: "/posts/:id"
      handler: "updatePosts"
      auth: true
    }
  ]
}
```
