# 10746 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "moderator"
      permissions: [
        "write"
        "delete"
        "manage_billing"
      ]
      max_seats: 17
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "manage_billing"
        "read"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "write"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "read"
        "manage_users"
        "manage_billing"
        "view_analytics"
        "delete"
      ]
      max_seats: 26
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "export_data"
        "manage_billing"
        "delete"
      ]
      max_seats: 38
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "view_analytics"
      ]
      max_seats: null
    }
  ]
}
```
