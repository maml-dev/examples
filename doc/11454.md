# 11454 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "billing"
      permissions: [
        "read"
        "manage_users"
      ]
      max_seats: 20
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "delete"
        "manage_users"
        "write"
        "view_analytics"
      ]
      max_seats: 48
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "manage_billing"
        "read"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "export_data"
        "manage_billing"
        "manage_users"
        "write"
        "delete"
      ]
      max_seats: 20
    }

    {
      name: "editor"
      permissions: [
        "manage_billing"
        "export_data"
        "manage_users"
        "view_analytics"
      ]
      max_seats: 14
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "manage_billing"
        "manage_users"
        "read"
        "delete"
      ]
      max_seats: null
    }
  ]
}
```
