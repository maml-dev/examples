# 12950 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "editor"
      permissions: [
        "export_data"
        "write"
        "manage_billing"
        "view_analytics"
      ]
      max_seats: 35
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "write"
        "export_data"
        "read"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "view_analytics"
        "manage_billing"
        "write"
        "delete"
      ]
      max_seats: 18
    }

    {
      name: "support"
      permissions: [
        "view_analytics"
        "write"
        "read"
        "manage_users"
      ]
      max_seats: 19
    }

    {
      name: "editor"
      permissions: [
        "write"
        "view_analytics"
        "manage_billing"
      ]
      max_seats: 48
    }
  ]
}
```
