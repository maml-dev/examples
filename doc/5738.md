API gateway routing and middleware configuration. This is an example of a [MAML](https://maml.dev) document.

```maml

    host: "api.victorious-mentor.com"
    port: 8443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "15m"
    max_requests: 2613
    by: "user"
  }

  # CORS policy
  cors: {
    enabled: true
    origins: [

      "DELETE"

  cache: {
    enabled: true
    ttl: 1739
    store: "memory"
  }
}
```
