# 8457 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "view_analytics"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "manage_billing"
        "delete"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "manage_billing"
        "export_data"
        "view_analytics"
        "read"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "write"
        "delete"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "write"
        "manage_billing"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "export_data"
        "write"
        "manage_users"
        "read"
      ]
      max_seats: null
    }
  ]
}
```
