# 7660 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "billing"
      permissions: [
        "write"
        "delete"
        "manage_users"
      ]
      max_seats: 16
    }

    {
      name: "billing"
      permissions: [
        "manage_billing"
        "delete"
        "view_analytics"
        "write"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "export_data"
        "write"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "read"
        "manage_billing"
        "delete"
        "view_analytics"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "manage_users"
        "read"
        "write"
      ]
      max_seats: 23
    }

    {
      name: "support"
      permissions: [
        "view_analytics"
        "manage_users"
        "read"
      ]
      max_seats: 18
    }
  ]
}
```
