# 14943 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "moderator"
      permissions: [
        "delete"
        "export_data"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "manage_billing"
        "write"
        "read"
        "delete"
        "view_analytics"
      ]
      max_seats: 9
    }

    {
      name: "editor"
      permissions: [
        "delete"
        "view_analytics"
        "write"
      ]
      max_seats: 26
    }

    {
      name: "viewer"
      permissions: [
        "read"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 10
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "write"
        "export_data"
      ]
      max_seats: 38
    }
  ]
}
```
