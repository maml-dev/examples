# 16946 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "moderator"
      permissions: [
        "delete"
        "manage_users"
        "read"
        "write"
      ]
      max_seats: 49
    }

    {
      name: "support"
      permissions: [
        "read"
        "manage_users"
        "export_data"
        "delete"
        "manage_billing"
      ]
      max_seats: 2
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "view_analytics"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "read"
        "write"
        "manage_billing"
        "delete"
      ]
      max_seats: 40
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "manage_users"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "manage_billing"
        "view_analytics"
      ]
      max_seats: null
    }
  ]
}
```
