# 15441 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "manage_billing"
        "delete"
        "read"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "manage_billing"
        "read"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "delete"
        "write"
      ]
      max_seats: 37
    }

    {
      name: "admin"
      permissions: [
        "write"
        "manage_users"
      ]
      max_seats: 24
    }

    {
      name: "editor"
      permissions: [
        "view_analytics"
        "manage_users"
        "write"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "delete"
        "export_data"
        "manage_users"
      ]
      max_seats: 1
    }
  ]
}
```
