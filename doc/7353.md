# 7353 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "ripe-yeast.org"
  records: [
    {
      type: "MX"
      name: "cdn"
      value: "100.151.166.117"
      ttl: 300
    }

    {
      type: "MX"
      name: "@"
      value: "afraid-ecliptic.info"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "*"
      value: "v=spf1 include:_spf.clueless-toothpick.net ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "www"
      value: "rapid-boyfriend.net"
      ttl: 300
    }

    {
      type: "MX"
      name: "staging"
      value: "winged-rim.net"
      ttl: 86400
    }

    {
      type: "MX"
      name: "staging"
      value: "v=spf1 include:_spf.sleepy-plugin.com ~all"
      ttl: 86400
    }

    {
      type: "A"
      name: "www"
      value: "cultivated-thongs.net"
      ttl: 300
    }
  ]
}
```
