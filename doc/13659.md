# 13659 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "read"
        "delete"
        "write"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 29
    }

    {
      name: "viewer"
      permissions: [
        "delete"
        "manage_users"
        "write"
        "read"
        "export_data"
      ]
      max_seats: 20
    }

    {
      name: "editor"
      permissions: [
        "write"
        "delete"
        "view_analytics"
      ]
      max_seats: 40
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "delete"
        "read"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "read"
        "delete"
      ]
      max_seats: null
    }
  ]
}
```
