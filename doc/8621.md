```maml
  provider: "digitalocean"
  region: "us-east-1"

    # Network
    vpc: {
      cidr_block: "10.190.0.0/16"
      enable_dns: true
      tags: {
        Name: "citizen-vpc"
        Environment: "staging"

      ami: "ami-ce59f5ca"
      type: "t3.micro"
      key_name: "legend-key"
      monitoring: true

        size: 100

    security_group: {
      name: "allow-all"
      ingress_port: 443
      ingress_cidr: "0.0.0.0/0"
    }
  }
}
```
