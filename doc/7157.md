# 7157 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "glittering-omelet.name"
  records: [
    {
      type: "A"
      name: "@"
      value: "v=spf1 include:_spf.plain-designation.com ~all"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "cdn"
      value: "133.55.98.82"
      ttl: 300
    }

    {
      type: "NS"
      name: "cdn"
      value: "alarmed-daughter.biz"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "@"
      value: "49.40.72.128"
      ttl: 86400
    }

    {
      type: "MX"
      name: "@"
      value: "74.1.215.120"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "@"
      value: "lazy-commodity.name"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "staging"
      value: "v=spf1 include:_spf.enlightened-designation.name ~all"
      ttl: 3600
    }
  ]
}
```
