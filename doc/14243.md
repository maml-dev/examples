# 14243 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "read"
        "view_analytics"
        "delete"
        "export_data"
      ]
      max_seats: 40
    }

    {
      name: "viewer"
      permissions: [
        "write"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "manage_billing"
        "delete"
        "export_data"
        "write"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "view_analytics"
        "manage_billing"
        "export_data"
        "delete"
        "read"
      ]
      max_seats: 45
    }

    {
      name: "viewer"
      permissions: [
        "manage_billing"
        "manage_users"
      ]
      max_seats: 9
    }
  ]
}
```
