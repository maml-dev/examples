# 6956 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "second-adult.info"
  records: [
    {
      type: "NS"
      name: "www"
      value: "insidious-heartbeat.com"
      ttl: 300
    }

    {
      type: "A"
      name: "cdn"
      value: "golden-cauliflower.org"
      ttl: 86400
    }

    {
      type: "NS"
      name: "www"
      value: "purple-taxicab.name"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "mail"
      value: "pale-boulevard.info"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "mail"
      value: "v=spf1 include:_spf.lazy-deduction.net ~all"
      ttl: 86400
    }

    {
      type: "A"
      name: "www"
      value: "v=spf1 include:_spf.general-gray.biz ~all"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "staging"
      value: "134.181.188.249"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "*"
      value: "v=spf1 include:_spf.ruddy-jogging.biz ~all"
      ttl: 3600
    }
  ]
}
```
