# 12349 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "red-coast.name"
  records: [
    {
      type: "NS"
      name: "*"
      value: "v=spf1 include:_spf.secret-wafer.com ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "mail"
      value: "sardonic-elevation.com"
      ttl: 86400
    }

    {
      type: "MX"
      name: "www"
      value: "v=spf1 include:_spf.pleasant-carboxyl.name ~all"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "cdn"
      value: "defenseless-onset.biz"
      ttl: 3600
    }

    {
      type: "A"
      name: "www"
      value: "v=spf1 include:_spf.icy-deployment.info ~all"
      ttl: 300
    }

    {
      type: "TXT"
      name: "cdn"
      value: "v=spf1 include:_spf.frail-pulse.com ~all"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "*"
      value: "63.13.240.141"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "mail"
      value: "164.84.5.152"
      ttl: 3600
    }
  ]
}
```
