# 7448 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "cloudy-scrap.biz"
  records: [
    {
      type: "CNAME"
      name: "www"
      value: "200.79.144.23"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "@"
      value: "v=spf1 include:_spf.haunting-teammate.info ~all"
      ttl: 3600
    }

    {
      type: "NS"
      name: "api"
      value: "123.199.8.12"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "api"
      value: "v=spf1 include:_spf.stale-jury.org ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "@"
      value: "154.158.207.198"
      ttl: 3600
    }

    {
      type: "MX"
      name: "staging"
      value: "v=spf1 include:_spf.ornery-tail.com ~all"
      ttl: 300
    }

    {
      type: "NS"
      name: "*"
      value: "63.211.194.245"
      ttl: 86400
    }
  ]
}
```
