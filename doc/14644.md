# 14644 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "smug-pomelo.info"
  records: [
    {
      type: "TXT"
      name: "*"
      value: "tense-birdcage.name"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "api"
      value: "v=spf1 include:_spf.admired-knitting.info ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "cdn"
      value: "v=spf1 include:_spf.lucky-governance.biz ~all"
      ttl: 300
    }

    {
      type: "MX"
      name: "*"
      value: "133.240.41.98"
      ttl: 3600
    }

    {
      type: "MX"
      name: "mail"
      value: "51.141.140.163"
      ttl: 3600
    }

    {
      type: "A"
      name: "@"
      value: "v=spf1 include:_spf.well-off-foodstuffs.com ~all"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "mail"
      value: "v=spf1 include:_spf.tiny-typeface.org ~all"
      ttl: 300
    }
  ]
}
```
