# 9959 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "billing"
      permissions: [
        "write"
        "view_analytics"
      ]
      max_seats: 27
    }

    {
      name: "admin"
      permissions: [
        "export_data"
        "delete"
      ]
      max_seats: 50
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "manage_billing"
      ]
      max_seats: 3
    }

    {
      name: "editor"
      permissions: [
        "delete"
        "export_data"
        "write"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "manage_billing"
        "write"
        "read"
        "delete"
        "view_analytics"
      ]
      max_seats: 4
    }

    {
      name: "editor"
      permissions: [
        "read"
        "manage_billing"
        "export_data"
        "delete"
      ]
      max_seats: 20
    }
  ]
}
```
