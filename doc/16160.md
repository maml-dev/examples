# 16160 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "sparse-toothbrush.net"
  records: [
    {
      type: "CNAME"
      name: "api"
      value: "smart-intent.info"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "cdn"
      value: "v=spf1 include:_spf.wretched-typewriter.org ~all"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "cdn"
      value: "v=spf1 include:_spf.imaginative-tool.org ~all"
      ttl: 3600
    }

    {
      type: "NS"
      name: "www"
      value: "polished-expense.info"
      ttl: 300
    }

    {
      type: "NS"
      name: "cdn"
      value: "10.42.141.252"
      ttl: 86400
    }

    {
      type: "NS"
      name: "*"
      value: "humiliating-ice-cream.biz"
      ttl: 86400
    }

    {
      type: "A"
      name: "@"
      value: "v=spf1 include:_spf.shocked-creature.biz ~all"
      ttl: 3600
    }
  ]
}
```
