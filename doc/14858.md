# 14858 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "read"
        "manage_users"
      ]
      max_seats: 31
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "export_data"
        "delete"
        "read"
      ]
      max_seats: 1
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "delete"
        "write"
        "manage_users"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "view_analytics"
        "write"
        "delete"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "write"
        "delete"
        "export_data"
        "manage_billing"
        "read"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "delete"
        "view_analytics"
        "manage_users"
      ]
      max_seats: null
    }
  ]
}
```
