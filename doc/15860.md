# 15860 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "manage_billing"
        "manage_users"
        "read"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "delete"
        "view_analytics"
      ]
      max_seats: 7
    }

    {
      name: "support"
      permissions: [
        "export_data"
        "delete"
        "view_analytics"
        "write"
      ]
      max_seats: 20
    }

    {
      name: "moderator"
      permissions: [
        "delete"
        "write"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "write"
        "manage_billing"
        "delete"
        "read"
      ]
      max_seats: 37
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "manage_users"
      ]
      max_seats: 17
    }
  ]
}
```
