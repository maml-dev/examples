# 10855 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "export_data"
        "view_analytics"
        "write"
        "delete"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "delete"
        "read"
        "manage_users"
      ]
      max_seats: 7
    }

    {
      name: "billing"
      permissions: [
        "manage_users"
        "read"
        "delete"
        "view_analytics"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "view_analytics"
        "read"
        "write"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "write"
        "manage_billing"
        "export_data"
        "read"
        "manage_users"
      ]
      max_seats: 44
    }

    {
      name: "support"
      permissions: [
        "write"
        "manage_users"
        "view_analytics"
      ]
      max_seats: null
    }
  ]
}
```
