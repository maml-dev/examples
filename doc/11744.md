# 11744 - API Routes

REST API route definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  base_url: "/api/v2"
  routes: [
    {
      method: "GET"
      path: "/orders"
      handler: "deleteOrders"
      auth: true
    }

    {
      method: "PUT"
      path: "/orders/:id"
      handler: "listUsers"
      auth: false
    }

    {
      method: "GET"
      path: "/posts/:id"
      handler: "getUsers"
      auth: true
    }

    {
      method: "PUT"
      path: "/products"
      handler: "getOrders"
      auth: true
    }

    {
      method: "DELETE"
      path: "/users/:id"
      handler: "updatePosts"
      auth: true
    }

    {
      method: "DELETE"
      path: "/comments"
      handler: "deleteProducts"
      auth: true
    }

    {
      method: "PUT"
      path: "/orders/:id"
      handler: "listProducts"
      auth: true
    }
  ]
}
```
