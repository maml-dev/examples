# 8357 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "glorious-hovel.biz"
  records: [
    {
      type: "A"
      name: "www"
      value: "v=spf1 include:_spf.imaginative-junior.net ~all"
      ttl: 86400
    }

    {
      type: "MX"
      name: "*"
      value: "multicolored-filter.org"
      ttl: 300
    }

    {
      type: "MX"
      name: "www"
      value: "afraid-fog.biz"
      ttl: 300
    }

    {
      type: "NS"
      name: "staging"
      value: "pleasing-maintainer.info"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "staging"
      value: "184.190.165.27"
      ttl: 3600
    }

    {
      type: "MX"
      name: "@"
      value: "159.227.63.83"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "mail"
      value: "v=spf1 include:_spf.bruised-sock.biz ~all"
      ttl: 86400
    }

    {
      type: "MX"
      name: "*"
      value: "v=spf1 include:_spf.charming-pop.net ~all"
      ttl: 86400
    }
  ]
}
```
