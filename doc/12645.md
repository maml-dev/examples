# 12645 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "slimy-nougat.biz"
  records: [
    {
      type: "NS"
      name: "staging"
      value: "colorless-popularity.net"
      ttl: 300
    }

    {
      type: "TXT"
      name: "api"
      value: "v=spf1 include:_spf.acceptable-pleasure.org ~all"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "@"
      value: "v=spf1 include:_spf.ill-schnitzel.org ~all"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "@"
      value: "v=spf1 include:_spf.quixotic-making.com ~all"
      ttl: 300
    }

    {
      type: "A"
      name: "staging"
      value: "228.179.88.138"
      ttl: 300
    }

    {
      type: "MX"
      name: "staging"
      value: "58.59.51.184"
      ttl: 300
    }

    {
      type: "NS"
      name: "cdn"
      value: "v=spf1 include:_spf.second-hand-suitcase.name ~all"
      ttl: 86400
    }
  ]
}
```
