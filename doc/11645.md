# 11645 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "delete"
        "export_data"
        "manage_billing"
        "write"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "write"
        "read"
        "view_analytics"
        "manage_billing"
      ]
      max_seats: 15
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "delete"
        "manage_users"
        "write"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "read"
        "delete"
        "view_analytics"
        "export_data"
        "manage_billing"
      ]
      max_seats: 1
    }
  ]
}
```
