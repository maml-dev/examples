# 12945 - API Routes

REST API route definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  base_url: "/api/v2"
  routes: [
    {
      method: "PATCH"
      path: "/products/:id"
      handler: "createPosts"
      auth: true
    }

    {
      method: "DELETE"
      path: "/comments/:id"
      handler: "deleteOrders"
      auth: false
    }

    {
      method: "DELETE"
      path: "/comments/:id"
      handler: "deletePosts"
      auth: true
    }

    {
      method: "DELETE"
      path: "/posts"
      handler: "deleteUsers"
      auth: false
    }

    {
      method: "GET"
      path: "/orders"
      handler: "deleteOrders"
      auth: false
    }

    {
      method: "POST"
      path: "/posts/:id"
      handler: "createPosts"
      auth: false
    }

    {
      method: "PUT"
      path: "/comments"
      handler: "getProducts"
      auth: true
    }
  ]
}
```
