{
  service: "solutions"
  # Prometheus metrics
  metrics: {
    endpoint: "/metrics"
    interval: 15 # seconds
    labels: {
      env: "prod"
      region: "eu-west"
      team: "jewelry"

  alerts: {
    cpu_threshold: 79

    latency_p99_ms: 701

      channel: "#oncall"
      severity: "warning"
    }
  }
}
