# 16957 - Cron Jobs

Scheduled task definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  timezone: "America/New_York"
  jobs: [
    {
      name: "backup-db"
      schedule: "0 0 * * *"
      command: "node scripts/cleanup.js"
      enabled: true
    }

    {
      name: "cleanup-temp"
      schedule: "30 6 * * 1-5"
      command: "python manage.py report"
      enabled: true
    }

    {
      name: "sync-data"
      schedule: "0 0 * * *"
      command: "python manage.py report"
      enabled: true
    }

    {
      name: "backup-db"
      schedule: "30 6 * * 1-5"
      command: "node scripts/cleanup.js"
      enabled: true
    }

    {
      name: "cleanup-temp"
      schedule: "30 6 * * 1-5"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }

    {
      name: "rotate-logs"
      schedule: "*/15 * * * *"
      command: "./bin/sync --all"
      enabled: true
    }
  ]
}
```
