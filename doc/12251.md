# 12251 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "victorious-worth.net"
  records: [
    {
      type: "NS"
      name: "mail"
      value: "v=spf1 include:_spf.variable-brook.info ~all"
      ttl: 300
    }

    {
      type: "MX"
      name: "www"
      value: "uneven-pilot.biz"
      ttl: 300
    }

    {
      type: "A"
      name: "cdn"
      value: "spiffy-folklore.org"
      ttl: 86400
    }

    {
      type: "A"
      name: "api"
      value: "v=spf1 include:_spf.handy-strait.biz ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "api"
      value: "limping-sanity.info"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "api"
      value: "92.147.217.20"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "*"
      value: "expert-captain.org"
      ttl: 300
    }
  ]
}
```
