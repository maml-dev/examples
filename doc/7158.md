# 7158 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "manage_billing"
        "export_data"
        "read"
      ]
      max_seats: 24
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "write"
        "view_analytics"
        "delete"
        "read"
      ]
      max_seats: 39
    }

    {
      name: "billing"
      permissions: [
        "read"
        "delete"
        "manage_users"
        "view_analytics"
      ]
      max_seats: 34
    }

    {
      name: "support"
      permissions: [
        "manage_users"
        "write"
      ]
      max_seats: 42
    }

    {
      name: "viewer"
      permissions: [
        "delete"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "read"
        "view_analytics"
        "manage_users"
      ]
      max_seats: null
    }
  ]
}
```
