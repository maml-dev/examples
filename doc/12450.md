# 12450 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "grumpy-saw.net"
  records: [
    {
      type: "MX"
      name: "staging"
      value: "152.84.228.208"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "staging"
      value: "v=spf1 include:_spf.thick-phrase.info ~all"
      ttl: 3600
    }

    {
      type: "NS"
      name: "cdn"
      value: "v=spf1 include:_spf.brilliant-suitcase.name ~all"
      ttl: 300
    }

    {
      type: "MX"
      name: "api"
      value: "v=spf1 include:_spf.right-loyalty.net ~all"
      ttl: 300
    }

    {
      type: "MX"
      name: "cdn"
      value: "extroverted-mobility.org"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "@"
      value: "184.146.247.125"
      ttl: 300
    }

    {
      type: "MX"
      name: "cdn"
      value: "v=spf1 include:_spf.hateful-casket.name ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "www"
      value: "v=spf1 include:_spf.nice-cappelletti.com ~all"
      ttl: 3600
    }
  ]
}
```
