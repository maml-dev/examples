# 15660 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "export_data"
        "manage_billing"
        "manage_users"
        "read"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "read"
        "export_data"
      ]
      max_seats: 25
    }

    {
      name: "moderator"
      permissions: [
        "read"
        "manage_users"
        "view_analytics"
        "delete"
      ]
      max_seats: 19
    }

    {
      name: "moderator"
      permissions: [
        "write"
        "delete"
        "manage_users"
        "manage_billing"
      ]
      max_seats: 36
    }

    {
      name: "billing"
      permissions: [
        "delete"
        "read"
      ]
      max_seats: 25
    }

    {
      name: "editor"
      permissions: [
        "read"
        "manage_billing"
      ]
      max_seats: 35
    }
  ]
}
```
