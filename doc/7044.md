# 7044 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "ample-resolve.name"
  records: [
    {
      type: "CNAME"
      name: "staging"
      value: "red-mousse.org"
      ttl: 3600
    }

    {
      type: "NS"
      name: "*"
      value: "227.132.55.99"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "@"
      value: "enchanted-bump.info"
      ttl: 300
    }

    {
      type: "A"
      name: "www"
      value: "spirited-presume.name"
      ttl: 3600
    }

    {
      type: "NS"
      name: "www"
      value: "v=spf1 include:_spf.clueless-luck.info ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "staging"
      value: "v=spf1 include:_spf.stiff-bookcase.biz ~all"
      ttl: 300
    }

    {
      type: "TXT"
      name: "www"
      value: "113.34.239.144"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "@"
      value: "v=spf1 include:_spf.grubby-backburn.org ~all"
      ttl: 300
    }
  ]
}
```
