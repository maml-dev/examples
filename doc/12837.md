# 12837 - Email Routing

Email routing and filtering rules. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  domain: "untimely-futon.org"
  inbound: {
    mx_records: [
      "mx1.untimely-futon.org"
      "mx2.untimely-futon.org"
    ]
    spam_filter: {
      enabled: true
      threshold: 8.5
      action: "tag"
    }

    dkim: {
      enabled: true
      selector: "default"
      key_size: 2048
    }
  }

  outbound: {
    relay: "smtp.untimely-futon.org"
    port: 587
    tls: true
    rate_limit: {
      per_minute: 326
      per_hour: 5537
    }
  }
}
```
