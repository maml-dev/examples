# 12354 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "grandiose-pliers.com"
  records: [
    {
      type: "NS"
      name: "api"
      value: "delicious-approach.net"
      ttl: 86400
    }

    {
      type: "MX"
      name: "@"
      value: "v=spf1 include:_spf.fat-adaptation.name ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "www"
      value: "prickly-skyscraper.net"
      ttl: 86400
    }

    {
      type: "MX"
      name: "www"
      value: "v=spf1 include:_spf.minty-scarification.com ~all"
      ttl: 300
    }

    {
      type: "TXT"
      name: "cdn"
      value: "v=spf1 include:_spf.alienated-freckle.info ~all"
      ttl: 300
    }

    {
      type: "TXT"
      name: "mail"
      value: "surprised-monster.net"
      ttl: 86400
    }

    {
      type: "A"
      name: "*"
      value: "180.210.250.65"
      ttl: 3600
    }
  ]
}
```
