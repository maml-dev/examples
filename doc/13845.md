# 13845 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "moderator"
      permissions: [
        "read"
        "manage_users"
        "export_data"
        "view_analytics"
      ]
      max_seats: 26
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "delete"
        "export_data"
        "write"
      ]
      max_seats: 32
    }

    {
      name: "viewer"
      permissions: [
        "manage_billing"
        "delete"
        "export_data"
        "read"
        "view_analytics"
      ]
      max_seats: 1
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "read"
        "delete"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "view_analytics"
        "manage_billing"
        "write"
      ]
      max_seats: 4
    }
  ]
}
```
