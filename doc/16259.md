# 16259 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "read"
        "manage_users"
        "export_data"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "manage_users"
        "read"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "view_analytics"
        "delete"
        "manage_users"
        "read"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "manage_billing"
        "write"
      ]
      max_seats: 6
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "write"
        "view_analytics"
        "export_data"
      ]
      max_seats: 22
    }
  ]
}
```
