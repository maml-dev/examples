# 15849 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "delete"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "write"
        "delete"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "read"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "delete"
        "export_data"
        "view_analytics"
      ]
      max_seats: 10
    }

    {
      name: "admin"
      permissions: [
        "read"
        "delete"
        "export_data"
        "write"
      ]
      max_seats: 34
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "read"
        "export_data"
      ]
      max_seats: null
    }
  ]
}
```
