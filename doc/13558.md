# 13558 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "weird-newsprint.net"
  records: [
    {
      type: "CNAME"
      name: "www"
      value: "v=spf1 include:_spf.remarkable-devil.name ~all"
      ttl: 86400
    }

    {
      type: "A"
      name: "staging"
      value: "45.255.211.250"
      ttl: 300
    }

    {
      type: "NS"
      name: "api"
      value: "buttery-disk.name"
      ttl: 3600
    }

    {
      type: "NS"
      name: "mail"
      value: "indelible-poppy.net"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "www"
      value: "stormy-sprinkles.name"
      ttl: 300
    }

    {
      type: "A"
      name: "mail"
      value: "70.242.13.130"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "@"
      value: "sandy-obedience.name"
      ttl: 3600
    }

    {
      type: "A"
      name: "staging"
      value: "selfish-populist.org"

}
```
