# 16945 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "billing"
      permissions: [
        "export_data"
        "manage_billing"
      ]
      max_seats: 49
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "read"
        "write"
        "manage_users"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "view_analytics"
        "delete"
        "write"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "write"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "read"
        "export_data"
        "delete"
        "view_analytics"
      ]
      max_seats: 33
    }
  ]
}
```
