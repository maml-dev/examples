# 6347 - API Routes

REST API route definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  base_url: "/api/v1"
  routes: [
    {
      method: "GET"
      path: "/comments/:id"
      handler: "getPosts"
      auth: false
    }

    {
      method: "PUT"
      path: "/users/:id"
      handler: "deleteOrders"
      auth: true
    }

    {
      method: "PATCH"
      path: "/orders/:id"
      handler: "createPosts"
      auth: false
    }

    {
      method: "PUT"
      path: "/comments"
      handler: "createUsers"
      auth: true
    }

    {
      method: "PUT"
      path: "/users/:id"
      handler: "updateOrders"
      auth: true
    }

    {
      method: "POST"
      path: "/products"
      handler: "listPosts"
      auth: false
    }

    {
      method: "POST"
      path: "/users"
      handler: "updatePosts"
      auth: false
    }
  ]
}
```
