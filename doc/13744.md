# 13744 - Cron Jobs

Scheduled task definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  timezone: "America/New_York"
  jobs: [
    {
      name: "rotate-logs"
      schedule: "0 2 * * 0"
      command: "node scripts/cleanup.js"
      enabled: true
    }

    {
      name: "refresh-cache"
      schedule: "0 * * * *"
      command: "./bin/sync --all"
      enabled: true
    }

    {
      name: "backup-db"
      schedule: "*/15 * * * *"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "sync-data"
      schedule: "0 * * * *"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }

    {
      name: "cleanup-temp"
      schedule: "*/15 * * * *"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }

    {
      name: "refresh-cache"
      schedule: "30 6 * * 1-5"
      command: "node scripts/cleanup.js"
      enabled: true
    }
  ]
}
```
