# 7255 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "great-paintwork.info"
  records: [
    {
      type: "A"
      name: "staging"
      value: "timely-kettledrum.com"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "api"
      value: "v=spf1 include:_spf.junior-offset.com ~all"
      ttl: 3600
    }

    {
      type: "NS"
      name: "@"
      value: "87.159.81.51"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "@"
      value: "outrageous-bell.net"
      ttl: 300
    }

    {
      type: "TXT"
      name: "*"
      value: "183.121.35.173"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "*"
      value: "v=spf1 include:_spf.delectable-masterpiece.net ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "mail"
      value: "aware-tenant.net"
      ttl: 86400
    }
  ]
}
```
