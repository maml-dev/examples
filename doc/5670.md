# 5670 - Docker Compose

```maml
{
  version: "3.8"
  services: {
    app: {
      image: "node:22"
      ports: [
        "3556:3144"

      environment: {
        NODE_ENV: "production"

        "db"
        "redis"
      ]
    }

    db: {
      image: "postgres:16"

      image: "redis:7-alpine"

        "6379:6379"
      ]
    }
  }

  volumes: {
    db_data: {}
  }
}
```
