# 7246 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "delete"
        "export_data"
        "view_analytics"
        "manage_users"
      ]
      max_seats: 47
    }

    {
      name: "moderator"
      permissions: [
        "read"
        "delete"
        "view_analytics"
        "manage_users"
        "write"
      ]
      max_seats: 1
    }

    {
      name: "support"
      permissions: [
        "read"
        "view_analytics"
        "write"
      ]
      max_seats: 22
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "delete"
        "read"
        "manage_billing"
      ]
      max_seats: 45
    }

    {
      name: "moderator"
      permissions: [
        "delete"
        "manage_billing"
      ]
      max_seats: 30
    }
  ]
}
```
