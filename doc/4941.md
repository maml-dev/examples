# 4941 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "write"
        "export_data"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 32
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "read"
        "write"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "delete"
        "read"
      ]
      max_seats: 20
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "export_data"
        "manage_users"
        "delete"
        "write"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "write"
        "delete"
        "read"
        "manage_users"
      ]
      max_seats: 25
    }
  ]
}
```
