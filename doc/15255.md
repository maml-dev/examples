# 15255 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "manage_billing"
        "export_data"
      ]
      max_seats: 23
    }

    {
      name: "viewer"
      permissions: [
        "manage_users"
        "write"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "export_data"
        "manage_billing"
        "manage_users"
        "write"
        "view_analytics"
      ]
      max_seats: 42
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "manage_users"
        "read"
        "write"
      ]
      max_seats: 35
    }

    {
      name: "viewer"
      permissions: [
        "manage_users"
        "read"
        "export_data"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "read"
        "view_analytics"
        "delete"
      ]
      max_seats: 22
    }
  ]
}
```
