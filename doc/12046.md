# 12046 - Cron Jobs

Scheduled task definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  timezone: "America/New_York"
  jobs: [
    {
      name: "send-reports"
      schedule: "*/15 * * * *"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "backup-db"
      schedule: "0 2 * * 0"
      command: "./bin/sync --all"
      enabled: false
    }

    {
      name: "cleanup-temp"
      schedule: "0 2 * * 0"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }

    {
      name: "sync-data"
      schedule: "30 6 * * 1-5"
      command: "node scripts/cleanup.js"
      enabled: false
    }

    {
      name: "rotate-logs"
      schedule: "0 0 * * *"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }

    {
      name: "sync-data"
      schedule: "0 2 * * 0"
      command: "node scripts/cleanup.js"
      enabled: true
    }
  ]
}
```
