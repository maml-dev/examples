# 6348 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "editor"
      permissions: [
        "manage_users"
        "read"
      ]
      max_seats: 49
    }

    {
      name: "viewer"
      permissions: [
        "write"
        "read"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "export_data"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 9
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "delete"
        "manage_billing"
        "read"
        "export_data"
      ]
      max_seats: 13
    }

    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "delete"
        "manage_users"
        "export_data"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "manage_users"
        "delete"
        "manage_billing"
        "view_analytics"
        "export_data"
      ]
      max_seats: null
    }
  ]
}
```
