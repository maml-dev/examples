# 15721 - Queue Config

{
  broker: "kafka"
  connection: {
    host: "localhost"
    port: 5672
    vhost: "/staging"

  # Queue definitions

      concurrency: 7
      retry: 1

      retry: 10
      timeout: 1393 # ms

  }
}
