# 6554 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "read"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "view_analytics"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "export_data"
        "read"
        "delete"
      ]
      max_seats: 31
    }

    {
      name: "moderator"
      permissions: [
        "manage_users"
        "write"
        "read"
        "export_data"
      ]
      max_seats: 26
    }

    {
      name: "moderator"
      permissions: [
        "read"
        "manage_users"
        "delete"
        "write"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "read"
        "delete"
        "manage_billing"
      ]
      max_seats: 30
    }
  ]
}
```
