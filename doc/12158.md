# 12158 - Cron Jobs

Scheduled task definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  timezone: "Europe/London"
  jobs: [
    {
      name: "sync-data"
      schedule: "*/15 * * * *"
      command: "node scripts/cleanup.js"
      enabled: true
    }

    {
      name: "backup-db"
      schedule: "30 6 * * 1-5"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }

    {
      name: "sync-data"
      schedule: "0 0 * * *"
      command: "python manage.py report"
      enabled: true
    }

    {
      name: "cleanup-temp"
      schedule: "*/15 * * * *"
      command: "pg_dump $DATABASE_URL"
      enabled: false
    }

    {
      name: "send-reports"
      schedule: "0 2 * * 0"
      command: "python manage.py report"
      enabled: true
    }

    {
      name: "sync-data"
      schedule: "0 2 * * 0"
      command: "./bin/sync --all"
      enabled: true
    }
  ]
}
```
