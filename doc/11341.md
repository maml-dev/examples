# 11341 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "viewer"
      permissions: [
        "export_data"
        "read"
      ]
      max_seats: 24
    }

    {
      name: "viewer"
      permissions: [
        "manage_billing"
        "manage_users"
        "view_analytics"
        "delete"
        "read"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "write"
        "view_analytics"
      ]
      max_seats: 35
    }

    {
      name: "support"
      permissions: [
        "write"
        "manage_users"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 27
    }

    {
      name: "editor"
      permissions: [
        "write"
        "export_data"
        "delete"
        "view_analytics"
        "manage_billing"
      ]
      max_seats: 36
    }
  ]
}
```
