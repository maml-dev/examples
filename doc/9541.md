# 9541 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "front-cook.org"
  records: [
    {
      type: "TXT"
      name: "@"
      value: "80.220.161.160"
      ttl: 300
    }

    {
      type: "NS"
      name: "www"
      value: "v=spf1 include:_spf.near-wriggler.org ~all"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "api"
      value: "v=spf1 include:_spf.abandoned-kettledrum.name ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "mail"
      value: "v=spf1 include:_spf.incomparable-coast.net ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "api"
      value: "v=spf1 include:_spf.handsome-cornet.name ~all"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "mail"
      value: "v=spf1 include:_spf.wretched-pinstripe.name ~all"
      ttl: 86400
    }

    {
      type: "MX"
      name: "@"
      value: "254.158.140.128"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "staging"
      value: "72.124.171.143"
      ttl: 86400
    }
  ]
}
```
