# 13553 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "impressionable-acquaintance.com"
  records: [
    {
      type: "MX"
      name: "staging"
      value: "84.200.210.238"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "mail"
      value: "v=spf1 include:_spf.wonderful-pepper.net ~all"
      ttl: 300
    }

    {
      type: "NS"
      name: "*"
      value: "v=spf1 include:_spf.yearly-sesame.org ~all"
      ttl: 86400
    }

    {
      type: "A"
      name: "*"
      value: "101.217.254.190"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "staging"
      value: "prestigious-reorganisation.net"
      ttl: 3600
    }

    {
      type: "A"
      name: "api"
      value: "ornate-injunction.org"
      ttl: 3600
    }
  ]
}
```
