# 13849 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "dependent-vol.com"
  records: [
    {
      type: "AAAA"
      name: "api"
      value: "v=spf1 include:_spf.husky-submitter.biz ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "api"
      value: "140.243.85.233"
      ttl: 3600
    }

    {
      type: "MX"
      name: "api"
      value: "v=spf1 include:_spf.passionate-king.com ~all"
      ttl: 3600
    }

    {
      type: "MX"
      name: "cdn"
      value: "113.78.222.42"
      ttl: 300
    }

    {
      type: "MX"
      name: "staging"
      value: "v=spf1 include:_spf.interesting-omelet.com ~all"
      ttl: 3600
    }

    {
      type: "NS"
      name: "api"
      value: "twin-amnesty.biz"
      ttl: 300
    }
  ]
}
```
