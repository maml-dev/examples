# 11743 - API Routes

REST API route definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  base_url: "/api/v2"
  routes: [
    {
      method: "PATCH"
      path: "/users"
      handler: "deleteUsers"
      auth: true
    }

    {
      method: "POST"
      path: "/comments"
      handler: "updateUsers"
      auth: true
    }

    {
      method: "GET"
      path: "/comments/:id"
      handler: "deleteProducts"
      auth: false
    }

    {
      method: "PUT"
      path: "/orders/:id"
      handler: "deleteOrders"
      auth: false
    }

    {
      method: "PATCH"
      path: "/users"
      handler: "listPosts"
      auth: true
    }

    {
      method: "PATCH"
      path: "/products/:id"
      handler: "deleteOrders"
      auth: true
    }

    {
      method: "GET"
      path: "/users/:id"
      handler: "deleteUsers"
      auth: true
    }
  ]
}
```
