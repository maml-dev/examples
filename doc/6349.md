# 6349 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "soft-soybean.name"
  records: [
    {
      type: "AAAA"
      name: "cdn"
      value: "v=spf1 include:_spf.decisive-handover.net ~all"
      ttl: 3600
    }

    {
      type: "NS"
      name: "*"
      value: "v=spf1 include:_spf.deadly-packaging.net ~all"
      ttl: 300
    }

    {
      type: "A"
      name: "cdn"
      value: "172.244.90.25"
      ttl: 86400
    }

    {
      type: "NS"
      name: "staging"
      value: "official-rosemary.info"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "www"
      value: "170.196.11.189"
      ttl: 3600
    }

    {
      type: "A"
      name: "mail"
      value: "tedious-whack.org"
      ttl: 3600
    }

    {
      type: "NS"
      name: "cdn"
      value: "151.62.9.24"
      ttl: 86400
    }
  ]
}
```
