# 17051 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "simple-brief.com"
  records: [
    {
      type: "NS"
      name: "@"
      value: "16.16.203.1"
      ttl: 300
    }

    {
      type: "NS"
      name: "www"
      value: "v=spf1 include:_spf.amused-ad.org ~all"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "staging"
      value: "admired-thigh.net"
      ttl: 300
    }

    {
      type: "TXT"
      name: "cdn"
      value: "v=spf1 include:_spf.aged-instance.com ~all"
      ttl: 3600
    }

    {
      type: "NS"
      name: "staging"
      value: "190.2.6.65"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "cdn"
      value: "v=spf1 include:_spf.ill-fated-bungalow.org ~all"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "mail"

    }

    {
      type: "CNAME"
      name: "cdn"

  ]
```
