# 8446 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "view_analytics"
        "delete"
        "manage_billing"
        "manage_users"
      ]
      max_seats: 23
    }

    {
      name: "support"
      permissions: [
        "read"
        "manage_users"
        "view_analytics"
        "delete"
      ]
      max_seats: 34
    }

    {
      name: "billing"
      permissions: [
        "manage_billing"
        "manage_users"
        "view_analytics"
        "delete"
        "read"
      ]
      max_seats: 27
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "view_analytics"
        "manage_users"
        "export_data"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "read"
        "export_data"
        "write"
      ]
      max_seats: 33
    }

    {
      name: "admin"
      permissions: [
        "view_analytics"
        "manage_billing"
        "write"
        "export_data"
      ]
      max_seats: null
    }
  ]
}
```
