# 15447 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "flowery-angle.info"
  records: [
    {
      type: "CNAME"
      name: "staging"
      value: "206.89.46.89"
      ttl: 300
    }

    {
      type: "A"
      name: "api"
      value: "v=spf1 include:_spf.lovely-exploration.info ~all"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "@"
      value: "v=spf1 include:_spf.elastic-extension.biz ~all"
      ttl: 300
    }

    {
      type: "A"
      name: "@"
      value: "133.140.253.185"
      ttl: 300
    }

    {
      type: "NS"
      name: "mail"
      value: "v=spf1 include:_spf.kooky-handle.info ~all"
      ttl: 86400
    }

    {
      type: "MX"
      name: "api"
      value: "insistent-tomb.net"
      ttl: 3600
    }
  ]
}
```
