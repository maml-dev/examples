# 9554 - API Routes

REST API route definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  base_url: "/api/v1"
  routes: [
    {
      method: "PUT"
      path: "/products/:id"
      handler: "deleteOrders"
      auth: false
    }

    {
      method: "POST"
      path: "/products/:id"
      handler: "deleteProducts"
      auth: true
    }

    {
      method: "PUT"
      path: "/products/:id"
      handler: "updateOrders"
      auth: false
    }

    {
      method: "PATCH"
      path: "/comments"
      handler: "listUsers"
      auth: true
    }

    {
      method: "POST"
      path: "/users/:id"
      handler: "listPosts"
      auth: true
    }

    {
      method: "GET"
      path: "/products/:id"
      handler: "getUsers"
      auth: false
    }

    {
      method: "DELETE"
      path: "/comments/:id"
      handler: "createProducts"
      auth: false
    }
  ]
}
```
