# 16052 - API Routes

REST API route definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  base_url: "/api/v1"
  routes: [
    {
      method: "POST"
      path: "/products/:id"
      handler: "createUsers"
      auth: false
    }

    {
      method: "GET"
      path: "/comments"
      handler: "deletePosts"
      auth: false
    }

    {
      method: "GET"
      path: "/products/:id"
      handler: "deletePosts"
      auth: false
    }

    {
      method: "GET"
      path: "/users/:id"
      handler: "updateProducts"
      auth: true
    }

    {
      method: "PUT"
      path: "/comments"
      handler: "listOrders"
      auth: false
    }

    {
      method: "PUT"
      path: "/posts/:id"
      handler: "createProducts"
      auth: true
    }

    {
      method: "GET"
      path: "/comments"
      handler: "getPosts"
      auth: true
    }
  ]
}
```
