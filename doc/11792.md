# 11792 - HTML Templates

```maml
  <a href="{{url}}">Get Started</a>

  plain_text: """
Hello, {{name}}!

Accusantium antea vomito. Certus universe modi sub desparatus. Carcer confero agnosco.

Visit: {{url}}
"""
}
```
