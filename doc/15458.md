# 15458 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "manage_billing"
        "write"
        "manage_users"
      ]
      max_seats: 2
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "write"
        "read"
      ]
      max_seats: 39
    }

    {
      name: "admin"
      permissions: [
        "delete"
        "read"
        "write"
      ]
      max_seats: 21
    }

    {
      name: "viewer"
      permissions: [
        "export_data"
        "manage_users"
        "manage_billing"
        "read"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "write"
        "delete"
        "read"
        "view_analytics"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "manage_users"
        "read"
      ]
      max_seats: null
    }
  ]
}
```
