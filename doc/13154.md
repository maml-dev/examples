# 13154 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "manage_users"
        "delete"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "read"
        "manage_billing"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "view_analytics"
        "export_data"
        "read"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "view_analytics"
        "delete"
        "read"
        "manage_billing"
        "export_data"
      ]
      max_seats: 48
    }

    {
      name: "admin"
      permissions: [
        "export_data"
        "view_analytics"
        "read"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "write"
        "view_analytics"
      ]
      max_seats: null
    }
  ]
}
```
