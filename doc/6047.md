# 6047 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "support"
      permissions: [
        "manage_users"
        "delete"
        "read"
        "export_data"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "view_analytics"
        "export_data"
        "manage_billing"
        "read"
        "write"
      ]
      max_seats: 49
    }

    {
      name: "support"
      permissions: [
        "write"
        "manage_billing"
        "read"
      ]
      max_seats: 10
    }

    {
      name: "support"
      permissions: [
        "manage_users"
        "write"
        "delete"
      ]
      max_seats: 9
    }

    {
      name: "viewer"
      permissions: [
        "manage_billing"
        "read"
        "write"
      ]
      max_seats: 40
    }
  ]
}
```
