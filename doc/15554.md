# 15554 - API Routes

REST API route definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  base_url: "/api/v2"
  routes: [
    {
      method: "PUT"
      path: "/comments"
      handler: "getPosts"
      auth: false
    }

    {
      method: "GET"
      path: "/orders/:id"
      handler: "listUsers"
      auth: true
    }

    {
      method: "PATCH"
      path: "/products"
      handler: "createProducts"
      auth: true
    }

    {
      method: "POST"
      path: "/comments"
      handler: "createUsers"
      auth: false
    }

    {
      method: "POST"
      path: "/products/:id"
      handler: "createProducts"
      auth: true
    }

    {
      method: "POST"
      path: "/products/:id"
      handler: "updateProducts"
      auth: false
    }

    {
      method: "PUT"
      path: "/comments/:id"
      handler: "deleteUsers"
      auth: true
    }
  ]
}
```
