# 11726 - Queue Config

{
  broker: "redis"
  connection: {
    host: "mq.internal"
    port: 9092
    vhost: "/production"
  }

  # Queue definitions
  queues: {
    default: {
      concurrency: 4
      retry: 4
      timeout: 21397 # ms

      concurrency: 9
      retry: 7
      timeout: 3366 # ms

    enabled: true
    max_retries: 4

  }
}
