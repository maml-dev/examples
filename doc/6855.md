# 6855 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "editor"
      permissions: [
        "export_data"
        "view_analytics"
        "read"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "read"
        "view_analytics"
      ]
      max_seats: 15
    }

    {
      name: "support"
      permissions: [
        "manage_billing"
        "read"
        "export_data"
      ]
      max_seats: 27
    }

    {
      name: "admin"
      permissions: [
        "read"
        "delete"
        "export_data"
        "write"
      ]
      max_seats: 25
    }

    {
      name: "billing"
      permissions: [
        "read"
        "write"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "manage_billing"
        "export_data"
      ]
      max_seats: 22
    }
  ]
}
```
