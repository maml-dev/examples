# 15046 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "moderator"
      permissions: [
        "read"
        "manage_users"
        "manage_billing"
        "view_analytics"
      ]
      max_seats: 22
    }

    {
      name: "billing"
      permissions: [
        "delete"
        "export_data"
        "read"
        "manage_billing"
      ]
      max_seats: 24
    }

    {
      name: "admin"
      permissions: [
        "read"
        "delete"
        "view_analytics"
      ]
      max_seats: 42
    }

    {
      name: "editor"
      permissions: [
        "manage_users"
        "write"
        "view_analytics"
        "read"
        "export_data"
      ]
      max_seats: 30
    }

    {
      name: "editor"
      permissions: [
        "delete"
        "read"
        "write"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "manage_users"
        "view_analytics"
        "read"
        "delete"
        "export_data"
      ]
      max_seats: null
    }
  ]
}
```
