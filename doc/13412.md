# 13412 - SSH Config

{
  host: "staging"
  hostname: "200.128.3.182"
  user: "silvia.collier"
  port: 22
  identity_file: "~/.ssh/id_rsa"

  forward_agent: false
  server_alive_interval: 30
  compression: true
}
