# 9455 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "delete"
        "view_analytics"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "manage_billing"
        "write"
      ]
      max_seats: 27
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "manage_users"
        "manage_billing"
        "read"
        "write"
      ]
      max_seats: 22
    }

    {
      name: "admin"
      permissions: [
        "view_analytics"
        "manage_users"
        "delete"
      ]
      max_seats: 6
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "write"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "manage_users"
        "view_analytics"
        "read"
        "write"
      ]
      max_seats: null
    }
  ]
}
```
