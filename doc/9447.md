# 9447 - Cron Jobs

Scheduled task definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  timezone: "UTC"
  jobs: [
    {
      name: "cleanup-temp"
      schedule: "0 * * * *"
      command: "node scripts/cleanup.js"
      enabled: false
    }

    {
      name: "backup-db"
      schedule: "30 6 * * 1-5"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }

    {
      name: "send-reports"
      schedule: "*/15 * * * *"
      command: "pg_dump $DATABASE_URL"
      enabled: true
    }

    {
      name: "refresh-cache"
      schedule: "0 2 * * 0"
      command: "./bin/sync --all"
      enabled: false
    }

    {
      name: "sync-data"
      schedule: "30 6 * * 1-5"
      command: "./bin/sync --all"
      enabled: false
    }

    {
      name: "backup-db"
      schedule: "*/15 * * * *"
      command: "node scripts/cleanup.js"
      enabled: true
    }
  ]
}
```
