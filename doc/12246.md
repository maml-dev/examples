# 12246 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "fussy-insolence.org"
  records: [
    {
      type: "A"
      name: "@"
      value: "v=spf1 include:_spf.slushy-noon.name ~all"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "api"
      value: "102.215.39.153"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "*"
      value: "v=spf1 include:_spf.serene-account.info ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "staging"
      value: "152.113.73.152"
      ttl: 300
    }

    {
      type: "A"
      name: "www"
      value: "severe-effector.biz"
      ttl: 300
    }

    {
      type: "MX"
      name: "@"
      value: "v=spf1 include:_spf.aching-scenario.biz ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "api"
      value: "59.201.147.100"
      ttl: 300
    }
  ]
}
```
