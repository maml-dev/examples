# 5042 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "super-sarong.biz"
  records: [
    {
      type: "A"
      name: "cdn"
      value: "v=spf1 include:_spf.minor-guard.org ~all"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "www"
      value: "205.83.191.109"
      ttl: 3600
    }

    {
      type: "MX"
      name: "api"
      value: "v=spf1 include:_spf.nocturnal-flat.com ~all"
      ttl: 300
    }

    {
      type: "TXT"
      name: "mail"
      value: "v=spf1 include:_spf.plump-bid.net ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "cdn"
      value: "last-obedience.name"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "@"
      value: "176.128.159.45"
      ttl: 86400
    }

    {
      type: "NS"
      name: "@"
      value: "232.143.76.160"
      ttl: 86400
    }
  ]
}
```
