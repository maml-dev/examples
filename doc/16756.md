# 16756 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "likable-velocity.info"
  records: [
    {
      type: "TXT"
      name: "@"
      value: "v=spf1 include:_spf.secret-conversation.com ~all"
      ttl: 86400
    }

    {
      type: "NS"
      name: "*"
      value: "blond-nucleotidase.name"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "cdn"
      value: "v=spf1 include:_spf.grim-colon.name ~all"
      ttl: 3600
    }

    {
      type: "MX"
      name: "cdn"
      value: "202.227.86.126"
      ttl: 300
    }

    {
      type: "MX"
      name: "www"
      value: "closed-gallery.info"
      ttl: 3600
    }
  ]
}
```
