# 7043 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "joyous-venom.info"
  records: [
    {
      type: "A"
      name: "api"
      value: "209.174.106.80"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "@"
      value: "v=spf1 include:_spf.utter-harp.com ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "*"
      value: "v=spf1 include:_spf.thrifty-unit.org ~all"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "staging"
      value: "v=spf1 include:_spf.best-descent.name ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "*"
      value: "v=spf1 include:_spf.elastic-pliers.info ~all"
      ttl: 300
    }

    {
      type: "MX"
      name: "www"
      value: "15.62.186.138"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "@"
      value: "v=spf1 include:_spf.sorrowful-outlaw.info ~all"
      ttl: 86400
    }
  ]
}
```
