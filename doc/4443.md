# 4443 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "editor"
      permissions: [
        "export_data"
        "view_analytics"
      ]
      max_seats: 1
    }

    {
      name: "admin"
      permissions: [
        "write"
        "manage_billing"
        "delete"
        "export_data"
      ]
      max_seats: 37
    }

    {
      name: "billing"
      permissions: [
        "view_analytics"
        "export_data"
        "delete"
        "manage_users"
      ]
      max_seats: 17
    }

    {
      name: "editor"
      permissions: [
        "write"
        "read"
        "manage_billing"
        "manage_users"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "read"
        "manage_users"
        "delete"
        "write"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "delete"
        "view_analytics"
        "export_data"
        "read"
        "manage_billing"
      ]
      max_seats: null
    }
  ]
}
```
