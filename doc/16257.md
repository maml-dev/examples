# 16257 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "moderator"
      permissions: [
        "manage_billing"
        "manage_users"
        "view_analytics"
        "read"
      ]
      max_seats: 50
    }

    {
      name: "support"
      permissions: [
        "manage_billing"
        "manage_users"
        "write"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "read"
        "manage_billing"
      ]
      max_seats: 11
    }

    {
      name: "support"
      permissions: [
        "read"
        "delete"
        "view_analytics"
        "write"
      ]
      max_seats: 17
    }

    {
      name: "support"
      permissions: [
        "view_analytics"
        "write"
        "export_data"
        "manage_billing"
        "delete"
      ]
      max_seats: null
    }
  ]
}
```
