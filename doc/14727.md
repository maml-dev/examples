    host: "api.wrong-casket.org"
    port: 8443
    protocol: "https"
  }

  # Rate limiting
  rate_limit: {
    enabled: true
    window: "5m"
    max_requests: 988

  # CORS policy

      "DELETE"

    enabled: true
    ttl: 2027

  }
}
