# 6645 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "editor"
      permissions: [
        "manage_users"
        "delete"
      ]
      max_seats: 14
    }

    {
      name: "admin"
      permissions: [
        "view_analytics"
        "manage_users"
        "delete"
        "manage_billing"
        "write"
      ]
      max_seats: 22
    }

    {
      name: "moderator"
      permissions: [
        "write"
        "delete"
        "view_analytics"
        "manage_users"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "write"
        "delete"
      ]
      max_seats: 50
    }

    {
      name: "admin"
      permissions: [
        "manage_users"
        "delete"
      ]
      max_seats: 32
    }

    {
      name: "moderator"
      permissions: [
        "delete"
        "read"
        "export_data"
        "view_analytics"
        "manage_billing"
      ]
      max_seats: null
    }
  ]
}
```
