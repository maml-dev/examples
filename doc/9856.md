# 9856 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "tangible-conservation.info"
  records: [
    {
      type: "NS"
      name: "staging"
      value: "2.203.193.108"
      ttl: 300
    }

    {
      type: "AAAA"
      name: "mail"
      value: "201.39.191.214"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "@"
      value: "v=spf1 include:_spf.busy-overload.biz ~all"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "staging"
      value: "giving-jungle.name"
      ttl: 300
    }

    {
      type: "TXT"
      name: "cdn"
      value: "35.113.216.154"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "cdn"
      value: "120.13.197.1"
      ttl: 3600
    }

    {
      type: "NS"
      name: "*"
      value: "v=spf1 include:_spf.honorable-castanet.net ~all"
      ttl: 3600
    }
  ]
}
```
