# 7753 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "sizzling-optimal.info"
  records: [
    {
      type: "CNAME"
      name: "api"
      value: "squeaky-brush.name"
      ttl: 300
    }

    {
      type: "A"
      name: "www"
      value: "v=spf1 include:_spf.superior-account.biz ~all"
      ttl: 86400
    }

    {
      type: "A"
      name: "api"
      value: "v=spf1 include:_spf.kaleidoscopic-king.name ~all"
      ttl: 86400
    }

    {
      type: "MX"
      name: "www"
      value: "idolized-paintwork.biz"
      ttl: 3600
    }

    {
      type: "A"
      name: "www"
      value: "runny-place.name"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "api"
      value: "lucky-wear.org"
      ttl: 3600
    }

    {
      type: "NS"
      name: "mail"
      value: "v=spf1 include:_spf.edible-celsius.net ~all"
      ttl: 86400
    }
  ]
}
```
