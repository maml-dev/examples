# 15248 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "fuzzy-cleaner.name"
  records: [
    {
      type: "MX"
      name: "@"
      value: "unique-manner.biz"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "@"
      value: "42.135.151.128"
      ttl: 300
    }

    {
      type: "TXT"
      name: "api"
      value: "222.62.53.37"
      ttl: 3600
    }

    {
      type: "NS"
      name: "*"
      value: "246.207.74.100"
      ttl: 300
    }

    {
      type: "MX"
      name: "*"
      value: "v=spf1 include:_spf.monthly-cauliflower.biz ~all"
      ttl: 300
    }

    {
      type: "TXT"
      name: "@"
      value: "putrid-gradient.org"
      ttl: 3600
    }

    {
      type: "A"
      name: "api"
      value: "sour-steeple.info"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "mail"
      value: "v=spf1 include:_spf.pleasant-pine.biz ~all"
      ttl: 86400
    }
  ]
}
```
