# 8554 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "admin"
      permissions: [
        "read"
        "manage_users"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "write"
        "delete"
        "manage_users"
      ]
      max_seats: null
    }

    {
      name: "moderator"
      permissions: [
        "export_data"
        "manage_billing"
        "view_analytics"
        "write"
      ]
      max_seats: null
    }

    {
      name: "editor"
      permissions: [
        "export_data"
        "read"
        "manage_users"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "viewer"
      permissions: [
        "delete"
        "export_data"
        "write"
        "read"
      ]
      max_seats: 22
    }
  ]
}
```
