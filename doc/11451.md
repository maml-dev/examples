# 11451 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "monstrous-marten.org"
  records: [
    {
      type: "CNAME"
      name: "staging"
      value: "46.73.68.155"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "api"
      value: "blushing-saw.name"
      ttl: 300
    }

    {
      type: "TXT"
      name: "api"
      value: "64.177.235.204"
      ttl: 300
    }

    {
      type: "NS"
      name: "www"
      value: "v=spf1 include:_spf.chubby-veto.biz ~all"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "mail"
      value: "v=spf1 include:_spf.indelible-vista.name ~all"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "@"
      value: "trusty-hope.info"
      ttl: 300
    }

    {
      type: "NS"
      name: "mail"
      value: "assured-appliance.org"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "cdn"
      value: "5.61.52.38"
      ttl: 300
    }
  ]
}
```
