# 13860 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "enlightened-obedience.net"
  records: [
    {
      type: "AAAA"
      name: "mail"
      value: "animated-thigh.org"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "@"
      value: "v=spf1 include:_spf.buzzing-shipper.org ~all"
      ttl: 300
    }

    {
      type: "MX"
      name: "cdn"
      value: "12.214.249.125"
      ttl: 86400
    }

    {
      type: "A"
      name: "www"
      value: "silver-fedora.biz"
      ttl: 300
    }

    {
      type: "CNAME"
      name: "mail"
      value: "far-begonia.net"
      ttl: 300
    }

    {
      type: "A"
      name: "staging"
      value: "memorable-saloon.com"
      ttl: 86400
    }

    {
      type: "TXT"
      name: "api"
      value: "56.218.197.120"
      ttl: 86400
    }

    {
      type: "MX"
      name: "www"
      value: "22.218.180.201"
      ttl: 300
    }
  ]
}
```
