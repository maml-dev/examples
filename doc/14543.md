# 14543 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "powerful-solvency.name"
  records: [
    {
      type: "NS"
      name: "staging"
      value: "shimmering-guacamole.info"
      ttl: 300
    }

    {
      type: "TXT"
      name: "cdn"
      value: "v=spf1 include:_spf.easy-fedora.biz ~all"
      ttl: 300
    }

    {
      type: "TXT"
      name: "*"
      value: "172.9.168.51"
      ttl: 300
    }

    {
      type: "NS"
      name: "*"
      value: "delirious-colon.org"
      ttl: 3600
    }

    {
      type: "AAAA"
      name: "*"
      value: "sparkling-celsius.com"
      ttl: 300
    }

    {
      type: "TXT"
      name: "*"
      value: "131.209.163.206"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "@"
      value: "v=spf1 include:_spf.ornery-plugin.org ~all"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "@"
      value: "heavenly-desk.net"
      ttl: 300
    }
  ]
}
```
