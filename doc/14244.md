# 14244 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "bright-dependency.name"
  records: [
    {
      type: "AAAA"
      name: "cdn"
      value: "213.89.212.3"
      ttl: 3600
    }

    {
      type: "TXT"
      name: "*"
      value: "misguided-disposer.info"
      ttl: 86400
    }

    {
      type: "AAAA"
      name: "*"
      value: "enraged-someplace.biz"
      ttl: 300
    }

    {
      type: "NS"
      name: "@"
      value: "v=spf1 include:_spf.nippy-airline.net ~all"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "www"
      value: "62.91.55.86"
      ttl: 3600
    }

    {
      type: "MX"
      name: "api"
      value: "close-switchboard.org"
      ttl: 86400
    }

    {
      type: "CNAME"
      name: "*"
      value: "v=spf1 include:_spf.insecure-euphonium.com ~all"
      ttl: 3600
    }
  ]
}
```
