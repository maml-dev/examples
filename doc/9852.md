# 9852 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "billing"
      permissions: [
        "view_analytics"
        "manage_billing"
        "write"
        "delete"
        "manage_users"
      ]
      max_seats: 50
    }

    {
      name: "editor"
      permissions: [
        "manage_billing"
        "export_data"
        "write"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "export_data"
        "view_analytics"
        "read"
        "delete"
        "write"
      ]
      max_seats: 18
    }

    {
      name: "admin"
      permissions: [
        "delete"
        "write"
        "manage_users"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "manage_billing"
        "manage_users"
        "delete"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "read"
        "delete"
      ]
      max_seats: null
    }
  ]
}
```
