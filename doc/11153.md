# 11153 - DNS Records

DNS zone record entries. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  zone: "sniveling-giant.org"
  records: [
    {
      type: "MX"
      name: "staging"
      value: "v=spf1 include:_spf.quarterly-compromise.name ~all"
      ttl: 3600
    }

    {
      type: "CNAME"
      name: "api"
      value: "unwelcome-bug.com"
      ttl: 300
    }

    {
      type: "A"
      name: "www"
      value: "8.184.197.242"
      ttl: 300
    }

    {
      type: "TXT"
      name: "staging"
      value: "intrepid-embossing.name"
      ttl: 3600
    }

    {
      type: "MX"
      name: "api"
      value: "138.237.107.205"
      ttl: 3600
    }

    {
      type: "NS"
      name: "@"
      value: "ill-aircraft.com"
      ttl: 86400
    }
  ]
}
```
