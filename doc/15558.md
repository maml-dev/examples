# 15558 - Roles & Permissions

Role-based access control definitions. This is an example of a [MAML](https://maml.dev) document.

```maml
{
  roles: [
    {
      name: "editor"
      permissions: [
        "view_analytics"
        "write"
        "delete"
        "export_data"
        "manage_billing"
      ]
      max_seats: 38
    }

    {
      name: "billing"
      permissions: [
        "view_analytics"
        "manage_users"
        "manage_billing"
      ]
      max_seats: null
    }

    {
      name: "billing"
      permissions: [
        "export_data"
        "write"
        "manage_billing"
        "manage_users"
        "view_analytics"
      ]
      max_seats: null
    }

    {
      name: "admin"
      permissions: [
        "write"
        "delete"
        "read"
        "manage_billing"
        "export_data"
      ]
      max_seats: null
    }

    {
      name: "support"
      permissions: [
        "write"
        "view_analytics"
      ]
      max_seats: null
    }
  ]
}
```
